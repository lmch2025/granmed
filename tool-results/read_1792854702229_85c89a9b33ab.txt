   149→export default function SequenceMemoryGame() {
   150→  const t = useT()
   151→  const lang = useLang()
   152→  const reduce = useReducedMotion()
   153→  const addPlay = useProfileStore((s) => s.addPlay)
   154→  const addWin = useProfileStore((s) => s.addWin)
   155→  const addXp = useProfileStore((s) => s.addXp)
   156→  const setBest = useProfileStore((s) => s.setBest)
   157→  const unlockBadge = useProfileStore((s) => s.unlockBadge)
   158→  const recordsRef = useProfileStore((s) => s.profile.games[gameId]?.records)
   159→  const records = recordsRef ?? {}
   160→
   161→  const [mode, setMode] = useState<Mode>('classic')
   162→  const [phase, setPhase] = useState<Phase>('idle')
   163→  const [level, setLevel] = useState(1)
   164→  const [inputIdx, setInputIdx] = useState(0)
   165→  const [lit, setLit] = useState<number | null>(null)
   166→  const [tapFlash, setTapFlash] = useState<number | null>(null)
   167→  const [shakeIdx, setShakeIdx] = useState<number | null>(null)
   168→  const [over, setOver] = useState<OverInfo | null>(null)
   169→  const [showOver, setShowOver] = useState(false)
   170→  const [turnFlash, setTurnFlash] = useState(false)
   171→  const [confetti, setConfetti] = useState(0)
   172→
   173→  const seqRef = useRef<number[]>([])
   174→  const timersRef = useRef<number[]>([])
   175→  const inputIdxRef = useRef(0)
   176→  const phaseRef = useRef<Phase>('idle')
   177→  const modeRef = useRef<Mode>('classic')
   178→  const pausedRef = useRef(false)
   179→  const lastStartRef = useRef(0)
   180→
   181→  const L = {
   182→    fr: {
   183→      board: 'Plateau de séquences',
   184→      watchTitle: 'Observez la séquence…',

   201→    en: {

   204→      yourTurn: 'Your turn!',

   206→      repeat: 'Repeat',
   207→      levelWord: 'Level',
   208→      bestClassic: 'Best — Classic',
   209→      bestFast: 'Best — Fast',
   210→      reached: 'sequence reproduced',
   211→      classicMode: 'Classic',
   212→      fastMode: 'Fast',
   213→      speedConst: 'constant speed',
   214→      speedAcc: 'speeds up every 5 levels',
   215→      sequence: 'The expected sequence',
   216→      expected: 'Expected',
   217→      played: 'Played',
   218→      startHint: 'Four pads, four tones: memorize the order and repeat it.',
   219→    },
   220→  }[lang]
   221→
   222→  function clearTimers() {
   223→    for (const id of timersRef.current) window.clearTimeout(id)
   224→    timersRef.current = []
   225→  }
   226→
   227→  function goPhase(p: Phase) {
   228→    phaseRef.current = p
   229→    setPhase(p)
   230→  }
   231→
   232→  function speedFor(len: number, m: Mode): { lit: number; gap: number } {
   233→    if (m === 'classic') return { lit: 400, gap: 200 }
   234→    const lit = Math.max(150, 300 - Math.floor((len - 1) / 5) * 10)
   235→    return { lit, gap: Math.round(lit / 2) }
   236→  }
   237→
   238→  /** Rejoue la séquence complète depuis le début (allume chaque pastille + ton).
   239→   *  Tous les minuteurs passent par timersRef (jamais d'effet qui fuit). */
   240→  function playSequence() {
   241→    clearTimers()
   242→    setLit(null)
   243→    setTurnFlash(false)
   244→    goPhase('showing')
   245→    inputIdxRef.current = 0
   246→    setInputIdx(0)
   247→    const seq = seqRef.current
   248→    const { lit: litMs, gap } = speedFor(seq.length, modeRef.current)
   249→    // Ton légèrement raccourci quand le mode Rapide accélère, pour éviter le brouillage.
   250→    const toneDur = Math.max(0.12, Math.min(0.3, litMs / 1000 - 0.04))
   251→    let at = 380
   252→    for (const pad of seq) {
   253→      timersRef.current.push(
   254→        window.setTimeout(() => {
   255→          setLit(pad)
   256→          sfx.tone(PADS[pad].tone, toneDur)
   257→        }, at),
   258→      )
   259→      timersRef.current.push(window.setTimeout(() => setLit(null), at + litMs))
   260→      at += litMs + gap
   261→    }
   262→    timersRef.current.push(
   263→      window.setTimeout(() => {
   264→        goPhase('input')
   265→        // Signal « À vous ! » les premiers niveaux, le temps de prendre le rythme.
   266→        if (seqRef.current.length <= 3) {
   267→          setTurnFlash(true)
   268→          timersRef.current.push(window.setTimeout(() => setTurnFlash(false), 1000))
   269→        }
   270→      }, at + 90),
   271→    )
   272→  }
   273→
   274→  function start(m: Mode) {
   275→    const now = Date.now()
   276→    if (now - lastStartRef.current < 350) return // anti double-tap (double addPlay)
   277→    lastStartRef.current = now
   278→    clearTimers()
   279→    pausedRef.current = false
   280→    modeRef.current = m
   281→    setMode(m)
   282→    seqRef.current = [Math.floor(Math.random() * 4)]
   283→    setLevel(1)
   284→    setOver(null)
   285→    setShowOver(false)
   286→    setShakeIdx(null)
   287→    setTapFlash(null)
   288→    setTurnFlash(false)
   289→    addPlay(gameId)
   290→    sfx.play('sweep', { volume: 0.5 })
   291→    playSequence()
   292→  }
   293→
   294→  /** Fin de partie : records, succès, écran de fin. Appelée depuis le handler
   295→   *  de saisie (jamais depuis un effet — convention React 19 du portail). */
   296→  function failRun(failedAt: number, played: number) {
   297→    const seq = seqRef.current
   298→    const score = Math.max(0, seq.length - 1)
   299→    const metric = modeRef.current === 'classic' ? 'best-sequence-classic' : 'best-sequence-fast'
   300→    const isRecord = score > 0 ? setBest(gameId, metric, score) : false
   301→    if (score >= 10) unlockBadge('simon-10')
   302→    if (score >= 20) unlockBadge('simon-20')
   303→    if (score >= 5) addWin(gameId)
   304→    goPhase('over')
   305→    setOver({ score, isRecord, failedAt, played, seq: [...seq] })
   306→    setShowOver(false)
   307→    // Laisse d'abord voir la pastille ratée (secousse rouge) avant l'écran de fin.
   308→    timersRef.current.push(window.setTimeout(() => sfx.play('lose', { volume: 0.55 }), 450))
   309→    timersRef.current.push(
   310→      window.setTimeout(() => {
   311→        setShowOver(true)
   312→        setShakeIdx(null)
   313→      }, 900),
   314→    )
   315→  }
   316→
   317→  function onPad(i: number) {
   318→    if (phaseRef.current !== 'input') return
   319→    sfx.tone(PADS[i].tone, 0.32)
   320→    setTapFlash(i)
   321→    timersRef.current.push(window.setTimeout(() => setTapFlash(null), 220))
   322→    const seq = seqRef.current
   323→    const idx = inputIdxRef.current
   324→    if (seq[idx] === i) {
   325→      navigator.vibrate?.(10)
   326→      if (idx + 1 >= seq.length) {
   327→        // Séquence complète : niveau suivant, la séquence s'allonge d'une couleur.
   328→        sfx.play('coin', { volume: 0.6 })
   329→        addXp(5)
   330→        if (seq.length % 5 === 0) {
   331→          sfx.play('levelup', { volume: 0.7 })
   332→          setConfetti((c) => c + 1)
   333→        }
   334→        seqRef.current = [...seq, Math.floor(Math.random() * 4)]
   335→        setLevel(seqRef.current.length)
   336→        goPhase('showing')
   337→        timersRef.current.push(window.setTimeout(() => playSequence(), 650))
   338→      } else {
   339→        inputIdxRef.current = idx + 1
   340→        setInputIdx(idx + 1)
   341→      }
   342→    } else {
   343→      sfx.play('wrong', { volume: 0.5 })
   344→      navigator.vibrate?.(10)
   345→      setShakeIdx(i)
   346→      failRun(idx, i)
   347→    }
   348→  }
   349→
   350→  /* Pause : si l'onglet est masqué pendant la lecture, on rejoue la séquence au retour. */
   351→  useEffect(() => {
   352→    const onVis = () => {
   353→      if (document.hidden) {
   354→        if (phaseRef.current === 'showing') {
   355→          clearTimers()
   356→          setLit(null)
   357→          pausedRef.current = true
   358→        }
   359→      } else if (pausedRef.current) {
   360→        pausedRef.current = false
   361→        playSequence()
   362→      }
   363→    }
   364→    document.addEventListener('visibilitychange', onVis)
   365→    return () => document.removeEventListener('visibilitychange', onVis)
   366→  }, [])
   367→
   368→  /* Nettoyage de tous les minuteurs au démontage. */
   369→  useEffect(() => () => clearTimers(), [])
   370→
   371→  /* Clavier : touches 1-4 pendant la phase de saisie (appui maintenu ignoré). */
   372→  useEffect(() => {
   373→    const onKey = (e: KeyboardEvent) => {
   374→      if (e.repeat) return
   375→      const i = ['1', '2', '3', '4'].indexOf(e.key)
   376→      if (i >= 0 && phaseRef.current === 'input') {
   377→        e.preventDefault()
   378→        onPad(i)
   379→      }
   380→    }
   381→    window.addEventListener('keydown', onKey)
   382→    return () => window.removeEventListener('keydown', onKey)
   383→  }, [])
   384→
   385→  const best = records[mode === 'classic' ? 'best-sequence-classic' : 'best-sequence-fast']?.value
   386→
   387→  return (
   388→    <div className="relative select-none">
   389→      {/* ---- HUD : une seule ligne compacte, au-dessus des pads ---- */}
   390→      <div className="mb-2.5 flex min-h-10 flex-wrap items-center gap-1.5">
   391→        {phase === 'idle' || phase === 'over' ? (
   392→          <>
   393→            <div className="flex flex-wrap gap-1.5" role="group" aria-label={t('common.mode')}>
   394→              {(['classic', 'fast'] as Mode[]).map((m) => (
   395→                <button
   396→                  key={m}
   397→                  type="button"
   398→                  aria-pressed={mode === m}
   399→                  onClick={() => {
   400→                    sfx.play('click', { volume: 0.5 })
   401→                    modeRef.current = m
   402→                    setMode(m)
   403→                  }}
   404→                  className={cn(
   405→                    'h-9 rounded-full border px-3.5 text-xs font-bold transition-colors',
   406→                    mode === m
   407→                      ? 'border-primary bg-primary/10 text-primary'
   408→                      : 'border-border bg-card text-muted-foreground hover:border-primary/40 hover:text-foreground',
   409→                  )}
   410→                >
   411→                  {m === 'classic' ? L.classicMode : L.fastMode}
   412→                </button>
   413→              ))}
   414→            </div>
   415→            {best != null ? (
   416→              <span
   417→                title={mode === 'classic' ? L.bestClassic : L.bestFast}
   418→                className="flex h-9 items-center gap-1.5 rounded-full bg-secondary/70 px-3 text-xs font-bold"
   419→              >
   420→                <Trophy className="h-3.5 w-3.5 text-amber-500" aria-hidden="true" />
   421→                <span className="tabular-nums">{best}</span>
   422→              </span>
   423→            ) : null}
   424→            <span className="hidden text-xs text-muted-foreground sm:inline">
   425→              {mode === 'classic' ? L.speedConst : L.speedAcc}
   426→            </span>
   427→            <Button size="sm" className="anim-glow ms-auto h-9 gap-1.5 rounded-full px-4" onClick={() => start(mode)}>
   428→              <Play className="h-4 w-4" aria-hidden="true" />
   429→              {phase === 'over' ? t('common.playAgain') : t('common.start')}
   430→            </Button>
   431→          </>
   432→        ) : (
   433→          <>
   434→            <span
   435→              key={level}
   436→              className={cn(
   437→                'flex h-8 items-center gap-1.5 rounded-full bg-secondary/70 px-2.5 text-xs font-bold',
   438→                !reduce && 'anim-pop-in',
   439→              )}
   440→            >
   441→              <Zap className="h-3.5 w-3.5 text-amber-500" aria-hidden="true" />
   442→              <span className="text-muted-foreground">{L.levelWord}</span>
   443→              <span className="tabular-nums">{level}</span>
   444→            </span>
   445→            {phase === 'showing' ? (
