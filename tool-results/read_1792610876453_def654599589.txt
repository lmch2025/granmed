   142→/** Série du jour : 6 vraies + 6 fausses, déterministe par date (graine contractuelle). */
   143→function buildDailySeries(dateKey: string): QuizStatement[] {
   144→  const rand = mulberry32(seedFromString('health-true-false:' + dateKey))
   145→  const trues = shuffle(QUIZ_STATEMENTS.filter((s) => s.answer === 'true'), rand).slice(0, 6)
   146→  const falses = shuffle(QUIZ_STATEMENTS.filter((s) => s.answer === 'false'), rand).slice(0, 6)
   147→  return shuffle([...trues, ...falses], rand)
   148→}
   149→
   150→/** Série rapide : 6 vraies + 6 fausses au hasard, en évitant les 20 dernières jouées. */
   151→function buildQuickSeries(recentIds: string[]): { series: QuizStatement[]; recent: string[] } {
   152→  const recent = new Set(recentIds)
   153→  let pool = QUIZ_STATEMENTS.filter((s) => !recent.has(s.id))
   154→  if (pool.length < SERIES_LEN) pool = QUIZ_STATEMENTS
   155→  const shuffled = shuffle(pool)
   156→  const trues = shuffled.filter((s) => s.answer === 'true').slice(0, 6)
   157→  const falses = shuffled.filter((s) => s.answer === 'false').slice(0, 6)
   158→  const chosen = [...trues, ...falses]
   159→  if (chosen.length < SERIES_LEN) {
   160→    const chosenIds = new Set(chosen.map((s) => s.id))
   161→    for (const s of shuffled) {
   162→      if (chosen.length >= SERIES_LEN) break
   163→      if (!chosenIds.has(s.id)) chosen.push(s)
   164→    }
   165→  }
   166→  const series = shuffle(chosen)
   167→  const nextRecent = [...series.map((s) => s.id), ...recentIds].slice(0, RECENT_LIMIT)
   168→  return { series, recent: nextRecent }
   169→}
