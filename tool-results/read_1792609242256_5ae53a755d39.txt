   174→/** mulberry32 local avec état sérialisable (défi du jour déterministe, reprise incluse). */
   175→function makeRng(seed: number) {
   176→  let a = seed >>> 0
   177→  return {
   178→    next: () => {
   179→      a |= 0
   180→      a = (a + 0x6d2b79f5) | 0
   181→      let t = Math.imul(a ^ (a >>> 15), 1 | a)
   182→      t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t
   183→      return ((t ^ (t >>> 14)) >>> 0) / 4294967296
   184→    },
   185→    state: () => a >>> 0,
   186→  }
   187→}

   189→function drawShape(pool: Shape[], rand: () => number): number {
   190→  const total = pool.reduce((acc, s) => acc + s.weight, 0)
   191→  let r = rand() * total
   192→  for (const s of pool) {
   193→    r -= s.weight
   194→    if (r <= 0) return SHAPES.indexOf(s)
   195→  }
   196→  return SHAPES.indexOf(pool[pool.length - 1])
   197→}

   199→function canPlaceAt(board: number[], s: Shape, r: number, c: number): boolean {
   200→  if (r < 0 || c < 0 || r + s.h > 10 || c + s.w > 10) return false
   201→  for (let i = 0; i < s.h; i++) {
   202→    for (let j = 0; j < s.w; j++) {
   203→      if (s.grid[i][j] && board[(r + i) * 10 + (c + j)] !== 0) return false
   204→    }
   205→  }
   206→  return true
   207→}
   208→
   209→function fitsAnywhere(board: number[], s: Shape): boolean {
   210→  for (let r = 0; r <= 10 - s.h; r++) {
   211→    for (let c = 0; c <= 10 - s.w; c++) {
   212→      if (canPlaceAt(board, s, r, c)) return true
   213→    }
   214→  }
   215→  return false
   216→}

   403→  /* --- Placement d'une pièce (tap sur une case) --- */
   404→  function placeAt(r: number, c: number) {
   405→    if (!game || game.over || lockRef.current || selected === null) return
   406→    const sid = game.slots[selected]
   407→    if (sid == null) return

   410→    if (!canPlaceAt(game.board, s, r, c)) {
   411→      sfx.play('wrong', { volume: 0.3 })
   412→      setFlash((f) => f + 1)
   413→      window.setTimeout(() => setFlash(0), 430)
   414→      return
   415→    }
   416→
   417→    lockRef.current = true
   418→    setSelected(null)
   419→    setGhost(null)
   420→
   421→    // Pose de la pièce
   422→    const board = [...game.board]
   423→    let placed = 0
   424→    for (let i = 0; i < s.h; i++) {
   425→      for (let j = 0; j < s.w; j++) {
   426→        if (s.grid[i][j]) {
   427→          board[(r + i) * 10 + (c + j)] = s.fam
   428→          placed++
   429→        }
   430→      }
   431→    }
   432→
   433→    // Lignes et colonnes pleines (effacées simultanément)
   434→    const fullRows: number[] = []
   435→    const fullCols: number[] = []
   436→    for (let rr = 0; rr < 10; rr++) {
   437→      let full = true
   438→      for (let cc = 0; cc < 10; cc++) {
   439→        if (board[rr * 10 + cc] === 0) {
   440→          full = false
   441→          break
   442→        }
   443→      }
   444→      if (full) fullRows.push(rr)
   445→    }
   446→    for (let cc = 0; cc < 10; cc++) {
   447→      let full = true
   448→      for (let rr = 0; rr < 10; rr++) {
   449→        if (board[rr * 10 + cc] === 0) {
   450→          full = false
   451→          break
   452→        }
   453→      }
   454→      if (full) fullCols.push(cc)
   455→    }
   456→    const nLines = fullRows.length + fullCols.length
   457→    const mult = nLines >= 2 ? nLines : 1
   458→    const newScore = game.score + placed + 10 * nLines * mult
   459→
   460→    // Consommation de l'emplacement + redistribution si les 3 sont vides
   461→    let slots: (number | null)[] = [...game.slots]
   462→    slots[selected] = null
   463→    const rng = makeRng(game.rng)
   464→    const pool = poolFor(game.mode)
   465→    if (slots.every((x) => x == null)) {
   466→      slots = [drawShape(pool, rng.next), drawShape(pool, rng.next), drawShape(pool, rng.next)]
   467→    }
   468→
   469→    const cleared = new Set<number>()
   470→    for (const rr of fullRows) for (let cc = 0; cc < 10; cc++) cleared.add(rr * 10 + cc)
   471→    for (const cc of fullCols) for (let rr = 0; rr < 10; rr++) cleared.add(rr * 10 + cc)
   472→    const clearedBoard = cleared.size > 0 ? board.map((v, i) => (cleared.has(i) ? 0 : v)) : board
   473→
   474→    const over = !slots.some((id) => id != null && fitsAnywhere(clearedBoard, SHAPES[id]))
   475→
   476→    sfx.play('pop')
   477→    if (nLines > 0) {
   478→      sfx.play('sweep', { volume: 0.8 })
   479→      if (nLines >= 2) sfx.play('coin')
   480→    }
   481→    if (nLines >= 4) unlockBadge('block-combo4')
   482→    if (newScore >= 5000 && game.score < 5000) unlockBadge('block-5000')
   483→
   484→    const wasDaily = game.daily
   485→    if (cleared.size > 0) {
   486→      // Flash des cases effacées, puis effacement réel
   487→      setGame({ ...game, board, slots, score: newScore, lines: game.lines + nLines, rng: rng.state() })
   488→      setClearing(cleared)
   489→      if (nLines >= 2) {
   490→        setComboFx({ n: nLines, pts: 10 * nLines * mult, key: Date.now() })
   491→        window.setTimeout(() => setComboFx(null), 950)
   492→      }
   493→      window.setTimeout(() => {
   494→        setClearing(new Set())
   495→        lockRef.current = false
   496→        setGame((g) => (g ? { ...g, board: clearedBoard, over } : g))
   497→        if (over) finish(newScore, wasDaily)
   498→      }, 300)
   499→    } else {
   500→      setGame({ ...game, board: clearedBoard, slots, score: newScore, lines: game.lines + nLines, rng: rng.state(), over })
   501→      lockRef.current = false
   502→      if (over) finish(newScore, wasDaily)
   503→    }
   504→  }
   505→
   506→  const selectedShapeId = game && selected !== null ? (game.slots[selected] ?? null) : null
   507→  const ghostFam = selectedShapeId != null ? SHAPES[selectedShapeId].fam : 1
   508→  const playing = game !== null && !game.over
   509→
   510→  return (
   511→    <div className="relative select-none">
   512→      {/* Bandeau de reprise */}
   513→      {saved && !game ? (
   514→        <div className="mb-4 flex flex-wrap items-center gap-3 rounded-2xl border-2 border-amber-300/60 bg-amber-50 px-4 py-3 dark:border-amber-500/25 dark:bg-amber-500/10">
   515→          <span className="text-2xl" aria-hidden="true">🧱</span>
   516→          <span className="min-w-0 flex-1 text-sm">
   517→            <span className="block font-bold text-amber-700 dark:text-amber-300">
   518→              {L.savedGame} — {saved.score} {L.points}
   519→            </span>
   520→            <span className="block text-xs text-muted-foreground">
   521→              {saved.mode === 'relax' ? L.relax : L.classic} · {saved.lines} {L.lines.toLowerCase()} · {saved.daily ? t('common.daily') : '—'}
   522→            </span>
   523→          </span>
   524→          <Button size="lg" className="h-11 rounded-full" onClick={resumeGame}>
   525→            {t('game.resume')}
   526→          </Button>
   527→          <Button size="lg" variant="secondary" className="h-11 rounded-full" onClick={() => startGame(saved.mode, false)}>
   528→            {t('game.newGame')}
   529→          </Button>
   530→        </div>
   531→      ) : null}
   532→
   533→      {/* Barre du haut : stats en jeu / configuration sinon */}
   534→      {playing ? (
   535→        <div className="mb-3 flex flex-wrap items-center gap-2">
   536→          <StatPill icon={<Zap className="h-4 w-4" aria-hidden="true" />} label={L.score} value={game.score} />
   537→          <StatPill icon={<Sparkles className="h-4 w-4" aria-hidden="true" />} label={L.lines} value={game.lines} />
   538→          <StatPill icon={<Trophy className="h-4 w-4" aria-hidden="true" />} label={t('common.best')} value={bestScore ?? 0} />
   539→          {game.daily ? (
   540→            <span className="flex h-11 items-center gap-1.5 rounded-full bg-amber-100 px-3.5 text-sm font-bold text-amber-700 dark:bg-amber-500/20 dark:text-amber-300">
   541→              <CalendarCheck className="h-4 w-4" aria-hidden="true" />
   542→              {t('common.daily')}
   543→            </span>
   544→          ) : null}
   545→          <Button
   546→            variant="secondary"
   547→            size="lg"
   548→            className="ms-auto h-11 rounded-full"
   549→            onClick={() => {
   550→              sfx.play('click')
   551→              startGame(game.mode, game.daily)
   552→            }}
   553→          >
   554→            <RotateCcw className="h-4 w-4" aria-hidden="true" />
   555→            {t('common.restart')}
   556→          </Button>
   557→        </div>
   558→      ) : (
   559→        <div className="mb-3 flex flex-wrap items-center gap-2">
   560→          <div className="flex flex-wrap gap-1.5" role="group" aria-label={t('common.mode')}>
   561→            <button
   562→              type="button"
   563→              aria-pressed={mode === 'classic'}
   564→              onClick={() => {
   565→                sfx.play('click', { volume: 0.5 })
   566→                setMode('classic')
   567→              }}
   568→              className={cn(
   569→                'flex h-11 items-center gap-1.5 rounded-full border px-3.5 text-sm font-semibold transition-all',
   570→                mode === 'classic'
   571→                  ? 'border-primary bg-primary/10 text-primary'
   572→                  : 'border-border bg-card text-muted-foreground hover:border-primary/40 hover:text-foreground',
   573→              )}
   574→            >
   575→              <Grid2x2 className="h-4 w-4" aria-hidden="true" />
   576→              {L.classic}
   577→            </button>
   578→            <button
   579→              type="button"
   580→              aria-pressed={mode === 'relax'}
   581→              onClick={() => {
   582→                sfx.play('click', { volume: 0.5 })
   583→                setMode('relax')
   584→              }}
   585→              className={cn(
   586→                'flex h-11 items-center gap-1.5 rounded-full border px-3.5 text-sm font-semibold transition-all',
   587→                mode === 'relax'
   588→                  ? 'border-primary bg-primary/10 text-primary'
   589→                  : 'border-border bg-card text-muted-foreground hover:border-primary/40 hover:text-foreground',
   590→              )}
   591→            >
   592→              <Flower2 className="h-4 w-4" aria-hidden="true" />
   593→              {L.relax}
   594→            </button>
   595→          </div>
   596→          {mode === 'relax' ? (
   597→            <span className="hidden text-xs text-muted-foreground sm:inline">{L.relaxHint}</span>
   598→          ) : (
   599→            <span className="hidden text-xs text-muted-foreground sm:inline">{L.classicHint}</span>
   600→          )}
   601→          {!dailyDone ? (
   602→            <Button
   603→              variant="outline"
   604→              size="lg"
   605→              className="h-11 rounded-full"
   606→              title={L.dailyHint}
   607→              onClick={() => startGame('classic', true)}
   608→            >
   609→              <CalendarCheck className="h-4 w-4 text-amber-500" aria-hidden="true" />
   610→              {t('common.daily')}
   611→            </Button>
   612→          ) : null}
   613→          <Button size="lg" className="ms-auto h-11 rounded-full anim-glow" onClick={() => startGame(mode, false)}>
   614→            <Play className="h-4 w-4" aria-hidden="true" />
   615→            {t('common.playNow')}
   616→          </Button>
   617→        </div>
   618→      )}
   619→
   620→      {/* Plateau */}
   621→      {game ? (
   622→        <div className="relative mx-auto w-[min(92vw,420px)]">
   623→          <div
   624→            role="grid"
   625→            aria-label={L.board}
   626→            onPointerLeave={() => setGhost(null)}
   627→            className={cn(
   628→              'grid aspect-square grid-cols-10 gap-[2px] rounded-xl border-2 bg-card p-[3px] shadow-inner',
   629→              flash > 0 ? 'anim-shake border-red-400 ring-2 ring-red-400/60' : 'border-border/70',
   630→            )}
   631→          >
   632→            {game.board.map((v, i) => {
   633→              const isGhostCell = ghost != null && ghost.includes(i)
   634→              return (
   635→                <div
   636→                  key={i}
   637→                  onPointerEnter={(e) => {
   638→                    if (e.pointerType === 'mouse') hoverCell(i)
   639→                  }}
   640→                  onClick={() => placeAt(Math.floor(i / 10), i % 10)}
   641→                  className={cn(
   642→                    'relative aspect-square rounded-[3px] transition-colors duration-75',
   643→                    v > 0 ? cn('bg-gradient-to-br shadow-sm', FAM_GRAD[v] ?? FAM_GRAD[1]) : 'bg-foreground/[0.06]',
   644→                    isGhostCell && cn('bg-gradient-to-br opacity-50', FAM_GRAD[ghostFam]),
   645→                    selectedShapeId != null && 'cursor-pointer',
   646→                  )}
   647→                >
   648→                  {clearing.has(i) ? (
   649→                    <span className="anim-pop-in absolute inset-0 rounded-[3px] bg-white/80" aria-hidden="true" />
   650→                  ) : null}
   651→                </div>
   652→              )
   653→            })}
   654→          </div>
   655→
   656→          {/* Combo flottant */}
   657→          {comboFx ? (
   658→            <div
   659→              key={comboFx.key}
   660→              className="anim-score pointer-events-none absolute left-1/2 top-1/2 z-20 -translate-x-1/2 -translate-y-1/2 whitespace-nowrap rounded-full bg-amber-500/95 px-4 py-2 font-display text-lg font-extrabold text-white shadow-lg"
   661→              aria-hidden="true"
   662→            >
   663→              COMBO ×{comboFx.n} +{comboFx.pts}
   664→            </div>
   665→          ) : null}
   666→
   667→          {/* Écran de fin de partie */}
   668→          {game.over ? (
   669→            <div className="absolute inset-0 z-20 flex flex-col items-center justify-center gap-3 rounded-xl bg-card/95 p-4 text-center shadow-lg backdrop-blur-sm" role="status">
   670→              <span className="text-4xl" aria-hidden="true">🧱</span>
   671→              <p className="font-display text-xl font-extrabold">{L.gameOver}</p>
   672→              <div className="flex flex-wrap justify-center gap-2">
   673→                <StatPill icon={<Zap className="h-4 w-4" aria-hidden="true" />} label={L.score} value={game.score} />
   674→                <StatPill icon={<Trophy className="h-4 w-4" aria-hidden="true" />} label={t('common.best')} value={Math.max(bestScore ?? 0, game.score)} />
   675→              </div>
   676→              {(bestScore ?? 0) > 0 && game.score >= (bestScore ?? 0) ? (
   677→                <span className="flex h-11 items-center gap-1.5 rounded-full bg-amber-100 px-3.5 text-sm font-bold text-amber-700 dark:bg-amber-500/20 dark:text-amber-300">
   678→                  <Trophy className="h-4 w-4" aria-hidden="true" /> {t('common.newRecord')}
   679→                </span>
   680→              ) : null}
   681→              <Button size="lg" className="rounded-full" onClick={() => startGame(game.mode, false)}>
   682→                <RotateCcw className="h-4 w-4" aria-hidden="true" />
   683→                {t('common.playAgain')}
   684→              </Button>
   685→            </div>
   686→          ) : null}
   687→
   688→          {/* Emplacements de pièces */}
   689→          <div className="mt-3 grid grid-cols-3 gap-2" role="group" aria-label={L.slots}>
   690→            {game.slots.map((sid, i) =>
   691→              sid == null ? (
   692→                <div
   693→                  key={i}
   694→                  className="flex min-h-[76px] items-center justify-center rounded-xl border-2 border-dashed border-border/60 bg-secondary/30 text-xs text-muted-foreground"
   695→                >
   696→                  {L.empty}
   697→                </div>
   698→              ) : (
   699→                <button
   700→                  key={i}
   701→                  type="button"
   702→                  aria-pressed={selected === i}
   703→                  aria-label={`${L.slots} ${i + 1}`}
   704→                  onClick={() => {
   705→                    sfx.play('click', { volume: 0.5 })
   706→                    setGhost(null)
   707→                    setSelected((cur) => (cur === i ? null : i))
   708→                  }}
   709→                  className={cn(
   710→                    'flex min-h-[76px] items-center justify-center rounded-xl border-2 bg-card transition-all',
   711→                    selected === i
   712→                      ? '-translate-y-1 border-primary shadow-md ring-2 ring-primary/40'
   713→                      : 'border-border hover:border-primary/40',
   714→                  )}
   715→                >
   716→                  <MiniPiece shape={SHAPES[sid]} />
   717→                </button>
   718→              ),
   719→            )}
   720→          </div>
   721→          {selectedShapeId == null && !game.over ? (
   722→            <p className="mt-2 text-center text-xs text-muted-foreground">{L.selectPiece}</p>
   723→          ) : null}
   724→        </div>
   725→      ) : (
   726→        <div className="checkerboard flex min-h-[320px] flex-col items-center justify-center gap-4 rounded-2xl border p-6 text-center">
   727→          <div className="flex items-end gap-4" aria-hidden="true">
   728→            <MiniPiece shape={SHAPES[1]} />
   729→            <MiniPiece shape={SHAPES[6]} />
   730→            <MiniPiece shape={SHAPES[10]} />
   731→          </div>
   732→          <p className="max-w-md text-sm text-muted-foreground">{content.tagline[lang]}</p>
   733→        </div>
   734→      )}
   735→    </div>
   736→  )
   737→}
