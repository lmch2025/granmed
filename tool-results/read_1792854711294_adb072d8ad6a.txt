   219→   219→function pickSpeciesIndex(band: number): number {
   220→   220→  const pool: { idx: number; w: number }[] = []
   221→   221→  SPECIES.forEach((s, i) => {
   222→   222→    if (s.band !== band) return
   223→   223→    pool.push({ idx: i, w: s.rarity === 'common' ? 1 : s.rarity === 'rare' ? 0.24 : 0.06 })
   224→   224→  })
   225→   225→  const total = pool.reduce((a, o) => a + o.w, 0)
   226→   226→  let r = Math.random() * total
   227→   227→  for (const o of pool) {
   228→   228→    r -= o.w
   229→   229→    if (r <= 0) return o.idx
   230→   230→  }
   231→   231→  return pool[pool.length - 1].idx
   232→   232→}

   234→   234→function spawnFish(band: number, fromEdge: boolean, W: number): Fish {
   235→   235→  const dir: 1 | -1 = Math.random() < 0.5 ? 1 : -1
   236→   236→  const spIdx = pickSpeciesIndex(band)
   237→   237→  return {
   238→   238→    sp: spIdx,
   239→   239→    x: fromEdge ? (dir === 1 ? -50 : W + 50) : Math.random() * W,
   240→   240→    yFrac: 0.08 + Math.random() * 0.84,
   241→   241→    dir,
   242→   242→    speed: SPECIES[spIdx].speed * (0.85 + Math.random() * 0.3),
   243→   243→    bobA: 5 + Math.random() * 9,
   244→   244→    bobF: 0.5 + Math.random() * 0.8,
   245→   245→    phase: Math.random() * Math.PI * 2,
   246→   246→    hooked: false,
   247→   247→    slot: 0,
   248→   248→  }
   249→   249→}

   260→   260→  }
   261→   261→  return w
   262→   262→}

   269→   269→function drawWater(ctx: CanvasRenderingContext2D, W: number, H: number, surfaceY: number) {
   270→   270→  // Ciel doux au-dessus de l'eau
   271→   271→  const sky = ctx.createLinearGradient(0, 0, 0, surfaceY)
   272→   272→  sky.addColorStop(0, '#ffedd2')
   273→   273→  sky.addColorStop(1, '#f9dcb0')
   274→   274→  ctx.fillStyle = sky
   275→   275→  ctx.fillRect(0, 0, W, surfaceY)
   276→   276→  // Eau vert-teal (vert dominant, jamais bleu)
   277→   277→  const water = ctx.createLinearGradient(0, surfaceY, 0, H)
   278→   278→  water.addColorStop(0, '#9fdcc0')
   279→   279→  water.addColorStop(0.3, '#52ab86')
   280→   280→  water.addColorStop(0.65, '#2a7c60')
   281→   281→  water.addColorStop(1, '#175243')
   282→   282→  ctx.fillStyle = water
   283→   283→  ctx.fillRect(0, surfaceY, W, H - surfaceY)
   284→   284→  // Assombrissement des profondeurs sur les bords
   285→   285→  const vg = ctx.createLinearGradient(0, surfaceY, 0, H)
   286→   286→  vg.addColorStop(0, 'rgba(10,48,36,0)')
   287→   287→  vg.addColorStop(1, 'rgba(8,40,30,0.45)')
   288→   288→  ctx.fillStyle = vg
   289→   289→  ctx.fillRect(0, surfaceY, W, H - surfaceY)
   290→   290→  // Fond marin sableux
   291→   291→  ctx.fillStyle = '#c9a36a'
   292→   292→  ctx.beginPath()
   293→   293→  ctx.moveTo(0, H)
   294→   294→  ctx.lineTo(0, H - 16)
   295→   295→  ctx.quadraticCurveTo(W * 0.25, H - 30, W * 0.5, H - 16)
   296→   296→  ctx.quadraticCurveTo(W * 0.75, H - 4, W, H - 20)
   297→   297→  ctx.lineTo(W, H)
   298→   298→  ctx.closePath()
   299→   299→  ctx.fill()
   300→   300→}

   302→   302→function drawBoat(ctx: CanvasRenderingContext2D, W: number, surfaceY: number, t: number, reduce: boolean) {
   303→   303→  const bob = reduce ? 0 : Math.sin(t * 0.9) * 3
   304→   304→  const by = surfaceY - 9 + bob
   305→   305→  const cx = W / 2
   306→   306→  // Coque
   307→   307→  const g = ctx.createLinearGradient(0, by, 0, by + 20)
   308→   308→  g.addColorStop(0, '#b97a45')
   309→   309→  g.addColorStop(1, '#8a5a30')
   310→   310→  ctx.fillStyle = g
   311→   311→  ctx.beginPath()
   312→   312→  ctx.moveTo(cx - 58, by)
   313→   313→  ctx.lineTo(cx + 58, by)
   314→   314→  ctx.quadraticCurveTo(cx + 52, by + 20, cx + 34, by + 20)
   315→   315→  ctx.lineTo(cx - 34, by + 20)
   316→   316→  ctx.quadraticCurveTo(cx - 52, by + 20, cx - 58, by)
   317→   317→  ctx.closePath()
   318→   318→  ctx.fill()
   319→   319→  ctx.fillStyle = 'rgba(255,240,214,0.4)'
   320→   320→  ctx.fillRect(cx - 56, by + 1.5, 112, 2)
   321→   321→  ctx.strokeStyle = 'rgba(90,58,30,0.4)'
   322→   322→  ctx.lineWidth = 1
   323→   323→  ctx.beginPath()
   324→   324→  ctx.moveTo(cx - 44, by + 9)
   325→   325→  ctx.lineTo(cx + 44, by + 9)
   326→   326→  ctx.stroke()
   327→   327→  // Canne
   328→   328→  ctx.strokeStyle = '#6b4a2f'
   329→   329→  ctx.lineWidth = 2.5
   330→   330→  ctx.beginPath()
   331→   331→  ctx.moveTo(cx + 16, by + 2)

   751→   751→            const fy = fishY(f)
