   170→   170→function buildDeck(theme: ThemeId, pairs: number, rand: () => number): Card[] {
   171→   171→  const chosen = shuffle(THEMES[theme].emojis, rand).slice(0, pairs)
   172→   172→  const deck = [...chosen, ...chosen].map((emoji, i) => ({ id: i, emoji }))
   173→   173→  return shuffle(deck, rand)

   176→   176→/* ================= Composant principal ================= */
   177→   177→export default function MemoryMatchGame() {
   178→   178→  const t = useT()
   179→   179→  const lang = useLang()
   180→   180→  const reduce = useReducedMotion()
   181→   181→  const dailyDone = useIsDailyDone(gameId)
   182→   182→  const addPlay = useProfileStore((s) => s.addPlay)
   183→   183→  const addWin = useProfileStore((s) => s.addWin)
   184→   184→  const addXp = useProfileStore((s) => s.addXp)
   185→   185→  const setBest = useProfileStore((s) => s.setBest)
   186→   186→  const unlockBadge = useProfileStore((s) => s.unlockBadge)
   187→   187→  const completeDaily = useProfileStore((s) => s.completeDaily)
   188→   188→
   189→   189→  const [theme, setTheme] = useState<ThemeId>('fruits')
   190→   190→  const [size, setSize] = useState<SizeId>('8')
   191→   191→  const [dailyMode, setDailyMode] = useState(false)
   192→   192→  const [showTimer, setShowTimer] = useState(true)
   193→   193→  const [deck, setDeck] = useState<Card[]>([])
   194→   194→  const [phase, setPhase] = useState<'idle' | 'playing' | 'won'>('idle')
   195→   195→  const [flipped, setFlipped] = useState<number[]>([])
   196→   196→  const [matched, setMatched] = useState<Set<number>>(new Set())
   197→   197→  const [moves, setMoves] = useState(0)
   198→   198→  const [mistakes, setMistakes] = useState(0)
   199→   199→  const [elapsed, setElapsed] = useState(0)
   200→   200→  const [won, setWon] = useState<{ time: number; moves: number; isRecord: boolean } | null>(null)
   201→   201→  const [confettiKey, setConfettiKey] = useState(0)
   202→   202→  const lockRef = useRef(false)
   203→   203→  const startRef = useRef(0)
   204→   204→
   205→   205→  /* --- Défi du jour : thème + taille déterministes --- */
   206→   206→  const dailyConfig = useMemo(() => {
   207→   207→    const rand = mulberry32(seedFromString(`memory:${todayKey()}`))
   208→   208→    const themes: ThemeId[] = ['fruits', 'nature', 'nutrition', 'animals', 'retro']
   209→   209→    const sizes: SizeId[] = ['6', '8', '12', '15']
   210→   210→    return { theme: themes[Math.floor(rand() * themes.length)], size: sizes[Math.floor(rand() * sizes.length)] }
   211→   211→  }, [])
   212→   212→
   213→   213→  const startGame = useCallback(
   214→   214→    (opts?: { daily?: boolean }) => {
   215→   215→      const isDaily = opts?.daily ?? false
   216→   216→      const gTheme = isDaily ? dailyConfig.theme : theme
   217→   217→      const gSize = isDaily ? dailyConfig.size : size
   218→   218→      const rand = isDaily ? mulberry32(seedFromString(`memory-deck:${todayKey()}`)) : Math.random
   219→   219→      setDeck(buildDeck(gTheme, SIZES[gSize].pairs, rand))
   220→   220→      setDailyMode(isDaily)
   221→   221→      setPhase('playing')
   222→   222→      setFlipped([])
   223→   223→      setMatched(new Set())
   224→   224→      setMoves(0)
   225→   225→      setMistakes(0)
   226→   226→      setElapsed(0)
   227→   227→      setWon(null)
   228→   228→      startRef.current = Date.now()
   229→   229→      addPlay(gameId)
   230→   230→    },
   231→   231→    [theme, size, dailyConfig, addPlay],
   232→   232→  )
   233→   233→
   234→   234→  /* --- Victoire (appelée depuis le handler, jamais depuis un effet) --- */
   235→   235→  const finishWin = useCallback(
   236→   236→    (finalMoves: number, finalMistakes: number) => {
   237→   237→      const time = Date.now() - startRef.current
   238→   238→      const pairs = SIZES[dailyMode ? dailyConfig.size : size].pairs
   239→   239→      const rec1 = setBest(gameId, `best-moves-${pairs}`, finalMoves, true)
   240→   240→      const rec2 = setBest(gameId, `best-time-${pairs}`, Math.round(time / 100) / 10, true)
   241→   241→      const isRecord = rec1 || rec2
   242→   242→      setPhase('won')
   243→   243→      setWon({ time, moves: finalMoves, isRecord })
   244→   244→      setConfettiKey((k) => k + 1)
   245→   245→      sfx.play('win')
   246→   246→      addWin(gameId)
   247→   247→      addXp(15 + pairs * 2 + (isRecord ? 10 : 0))
   248→   248→      if (finalMistakes === 0) unlockBadge('memory-perfect')
   249→   249→      if (pairs === 15) unlockBadge('memory-master')
   250→   250→      if (dailyMode) completeDaily(gameId)
   251→   251→    },
   252→   252→    [dailyMode, dailyConfig.size, size, setBest, addWin, addXp, unlockBadge, completeDaily],
   253→   253→  )
   254→   254→
   255→   255→  /* --- Chronomètre --- */
   256→   256→  useEffect(() => {
   257→   257→    if (phase !== 'playing') return
   258→   258→    const id = window.setInterval(() => setElapsed(Date.now() - startRef.current), 200)
   259→   259→    return () => window.clearInterval(id)
   260→   260→  }, [phase])
   261→   261→
   262→   262→  /* --- Logique de retournement --- */
   263→   263→  const flip = (idx: number) => {
   264→   264→    if (phase !== 'playing' || lockRef.current) return
   265→   265→    if (flipped.includes(idx) || matched.has(idx)) return
   266→   266→    sfx.play('flip')
   267→   267→
   268→   268→    const next = [...flipped, idx]
   269→   269→    setFlipped(next)
   270→   270→
   271→   271→    if (next.length === 2) {
   272→   272→      const newMoves = moves + 1
   273→   273→      setMoves(newMoves)
   274→   274→      lockRef.current = true
   275→   275→      const [a, b] = next
   276→   276→      if (deck[a].emoji === deck[b].emoji) {
   277→   277→        window.setTimeout(() => {
   278→   278→          sfx.play('pair')
   279→   279→          const nextMatched = new Set([...matched, a, b])
   280→   280→          setMatched(nextMatched)
   281→   281→          setFlipped([])
   282→   282→          lockRef.current = false
   283→   283→          if (nextMatched.size === deck.length) {
   284→   284→            finishWin(newMoves, mistakes)
   285→   285→          }
   286→   286→        }, 420)
   287→   287→      } else {
   288→   288→        setMistakes((m) => m + 1)
   289→   289→        window.setTimeout(() => {
   290→   290→          sfx.play('wrong', { volume: 0.35 })
   291→   291→          setFlipped([])
   292→   292→          lockRef.current = false
   293→   293→        }, 850)
   294→   294→      }
   295→   295→    }
   296→   296→  }
   297→   297→
   298→   298→  const pairsTotal = deck.length / 2
   299→   299→  const foundPairs = matched.size / 2
   300→   300→
   301→   301→  /* --- Rendu --- */
   302→   302→  return (
   303→   303→    <div className="relative select-none">
   304→   304→      <Confetti fire={confettiKey} count={reduce ? 0 : 110} />
   305→   305→
   306→   306→      {/* Barre de configuration */}
   307→   307→      <div className="mb-4 flex flex-wrap items-center gap-2">
   308→   308→        {phase === 'idle' || phase === 'won' ? (
   309→   309→          <>
   310→   310→            <div className="flex flex-wrap gap-1.5" role="group" aria-label={t('common.theme_name')}>
   311→   311→              {(Object.keys(THEMES) as ThemeId[]).map((id) => (
   312→   312→                <button
   313→   313→                  key={id}
   314→   314→                  type="button"
   315→   315→                  aria-pressed={theme === id && !dailyMode}
   316→   316→                  onClick={() => {
   317→   317→                    sfx.play('click', { volume: 0.5 })
   318→   318→                    setTheme(id)
   319→   319→                    setDailyMode(false)
   320→   320→                  }}
   321→   321→                  className={cn(
   322→   322→                    'flex h-11 items-center gap-1.5 rounded-full border px-3.5 text-sm font-semibold transition-all',
   323→   323→                    theme === id && !dailyMode
   324→   324→                      ? 'border-primary bg-primary/10 text-primary'
   325→   325→                      : 'border-border bg-card text-muted-foreground hover:border-primary/40 hover:text-foreground',
   326→   326→                  )}
   327→   327→                >
   328→   328→                  <span aria-hidden="true">{THEMES[id].icon}</span>
   329→   329→                  <span className="hidden sm:inline">{THEMES[id].label[lang]}</span>
   330→   330→                </button>
   331→   331→              ))}
   332→   332→            </div>
   333→   333→            <div className="flex flex-wrap gap-1.5" role="group" aria-label={t('common.difficulty')}>
   334→   334→              {(Object.keys(SIZES) as SizeId[]).map((id) => (
   335→   335→                <button
   336→   336→                  key={id}
   337→   337→                  type="button"
   338→   338→                  aria-pressed={size === id && !dailyMode}
   339→   339→                  onClick={() => {
   340→   340→                    sfx.play('click', { volume: 0.5 })
   341→   341→                    setSize(id)
   342→   342→                    setDailyMode(false)
   343→   343→                  }}
   344→   344→                  className={cn(
   345→   345→                    'h-11 rounded-full border px-3.5 text-sm font-bold transition-all',

   537→   537→function fmt(ms: number): string {
   538→   538→  const s = ms / 1000
   539→   539→  if (s < 60) return `${s.toFixed(1)}s`
   540→   540→  const m = Math.floor(s / 60)
   541→   541→  return `${m}:${String(Math.floor(s % 60)).padStart(2, '0')}`
