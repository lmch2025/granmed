   171→/** Générateur pseudo-aléatoire dont l'état (serialisable) permet des parties déterministes. */
   172→function rngStep(s: number): [number, number] {
   173→  const a = (s + 0x6d2b79f5) | 0
   174→  let t = Math.imul(a ^ (a >>> 15), 1 | a)
   175→  t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t
   176→  return [a, ((t ^ (t >>> 14)) >>> 0) / 4294967296]
   177→}
   178→
   179→function makeRng(seed: number): { next: () => number; state: () => number } {
   180→  let s = seed >>> 0
   181→  return {
   182→    next: () => {
   183→      const [ns, v] = rngStep(s)
   184→      s = ns
   185→      return v
   186→    },
   187→    state: () => s,
   188→  }
   189→}

   191→/** Fusionne une ligne de 4 valeurs dans l'ordre du mouvement. */
   192→function slideLine(vals: number[]): { line: number[]; gained: number; merged: boolean[] } {
   193→  const out: number[] = []
   194→  const merged: boolean[] = []
   195→  const items = vals.filter((v) => v !== 0)
   196→  let i = 0
   197→  while (i < items.length) {
   198→    if (i + 1 < items.length && items[i] === items[i + 1]) {
   199→      out.push(items[i] * 2)
   200→      merged.push(true)
   201→      i += 2
   202→    } else {
   203→      out.push(items[i])
   204→      merged.push(false)
   205→      i += 1
   206→    }
   207→  }
   208→  while (out.length < 4) {
   209→    out.push(0)
   210→    merged.push(false)
   211→  }
   212→  let gained = 0
   213→  for (let k = 0; k < 4; k++) if (merged[k]) gained += out[k]
   214→  return { line: out, gained, merged }
   215→}
   216→
   217→/** Les 4 indices de la ligne/colonne `line`, dans l'ordre du mouvement. */
   218→function lineIndices(dir: Dir, line: number): number[] {
   219→  const idx: number[] = []
   220→  for (let k = 0; k < 4; k++) {
   221→    if (dir === 'left') idx.push(line * 4 + k)
   222→    else if (dir === 'right') idx.push(line * 4 + (3 - k))
   223→    else if (dir === 'up') idx.push(k * 4 + line)
   224→    else idx.push((3 - k) * 4 + line)
   225→  }
   226→  return idx
   227→}

   229→function canMove(b: number[]): boolean {
   230→  for (let i = 0; i < 16; i++) {
   231→    if (b[i] === 0) return true
   232→    const r = (i / 4) | 0
   233→    const c = i % 4
   234→    if (c < 3 && b[i] === b[i + 1]) return true
   235→    if (r < 3 && b[i] === b[i + 4]) return true
   236→  }
   237→  return false
   238→}

   376→  /* --- Coup --- */
   377→  function move(dir: Dir) {
   378→    if (!game || game.over || (game.won && !game.ack)) return
   379→    const { board, score } = game
   380→    const next = [...board]
   381→    const mergedIdx: number[] = []
   382→    let gained = 0
   383→    let changed = false
   384→    for (let line = 0; line < 4; line++) {
   385→      const idxs = lineIndices(dir, line)
   386→      const res = slideLine(idxs.map((i) => board[i]))
   387→      for (let k = 0; k < 4; k++) {
   388→        if (next[idxs[k]] !== res.line[k]) {
   389→          changed = true
   390→          next[idxs[k]] = res.line[k]
   391→        }
   392→        if (res.merged[k]) mergedIdx.push(idxs[k])
   393→      }
   394→      gained += res.gained
   395→    }
   396→    if (!changed) return
   397→
   398→    sfx.play('swish', { volume: 0.25 })
   399→    if (mergedIdx.length > 0) sfx.play('merge')
   400→
   401→    const rng = makeRng(game.rngState)
   402→    let spawnIdx = -1
   403→    const empties: number[] = []
   404→    for (let i = 0; i < 16; i++) if (next[i] === 0) empties.push(i)
   405→    if (empties.length > 0) {
   406→      spawnIdx = empties[Math.floor(rng.next() * empties.length)]
   407→      next[spawnIdx] = rng.next() < 0.9 ? 2 : 4
   408→    }
   409→
   410→    const newScore = score + gained
   411→    if (Math.floor(newScore / 500) > Math.floor(score / 500)) sfx.play('coin')
   412→
   413→    setGame({
   414→      ...game,
   415→      board: next,
   416→      score: newScore,
   417→      prev: { board, score },
   418→      rngState: rng.state(),
   419→      moves: game.moves + 1,
   420→    })
   421→    markAnim(mergedIdx, 'merged')
   422→    if (spawnIdx >= 0) markAnim([spawnIdx], 'new')
   423→
   424→    const maxTile = next.reduce((a, b) => Math.max(a, b), 0)
   425→    if (maxTile >= 1024) unlockBadge('n2048-1024')
   426→    if (maxTile >= 2048) unlockBadge('n2048-2048')
   427→    if (maxTile >= 4096) unlockBadge('n2048-4096')
   428→
   429→    if (game.daily && !game.daily256 && maxTile >= 256) {
   430→      setGame((g) => (g ? { ...g, daily256: true } : g))
   431→      sfx.play('achievement')
   432→      completeDaily(gameId)
   433→      toast('🎯', { description: L.dailyDone, duration: 4500 })
   434→    }
   435→
   436→    if (!game.won && maxTile >= 2048) {
   437→      const awarded = awardEnd(newScore, next, game.xpAwarded)
   438→      setGame((g) => (g ? { ...g, won: true, xpAwarded: awarded } : g))
   439→      setConfettiKey((k) => k + 1)
   440→      sfx.play('win')
   441→      addWin(gameId)
   442→    } else if (!canMove(next)) {
   443→      const awarded = awardEnd(newScore, next, game.xpAwarded)
   444→      setGame((g) => (g ? { ...g, over: true, xpAwarded: awarded } : g))
   445→      sfx.play('lose')
   446→      clearSave(gameId)
   447→    }
   448→  }

   450→  /* --- Annulation --- */
   451→  function undo() {
   452→    if (!game || !game.prev || game.undos <= 0) return
   453→    sfx.play('sweep')
   454→    setGame({
   455→      ...game,
   456→      board: game.prev.board,
   457→      score: game.prev.score,
   458→      prev: null,
   459→      undos: game.undos - 1,
   460→      over: false,
   461→    })
   462→    setAnimKind(new Array(16).fill(null))
   463→  }
   464→
   465→  /* --- Clavier : flèches + WASD/ZQSD (ref => toujours à jour) --- */
   466→  useEffect(() => {
   467→    moveRef.current = move
   468→  })
   469→  useEffect(() => {
   470→    const onKey = (e: KeyboardEvent) => {
   471→      if (!game || game.over || (game.won && !game.ack)) return
   472→      const dir = KEYMAP[e.key]
   473→      if (!dir) return
   474→      e.preventDefault()
   475→      moveRef.current(dir)
   476→    }
   477→    window.addEventListener('keydown', onKey)
   478→    return () => window.removeEventListener('keydown', onKey)
   479→  }, [game])
   480→
   481→  /* --- Balayage tactile --- */
   482→  function onPointerDown(e: React.PointerEvent) {
   483→    // Pas de capture quand un voile (victoire / fin) affiche des boutons : le clic doit passer.
   484→    if (game?.over || (game?.won && !game.ack)) return
   485→    swipeRef.current = { x: e.clientX, y: e.clientY }
   486→    try {
   487→      ;(e.currentTarget as HTMLElement).setPointerCapture(e.pointerId)
   488→    } catch {}
   489→  }
   490→
   491→  function onPointerUp(e: React.PointerEvent) {
   492→    const s = swipeRef.current
   493→    swipeRef.current = null
   494→    if (!s) return
   495→    const dx = e.clientX - s.x
   496→    const dy = e.clientY - s.y
   497→    const adx = Math.abs(dx)
   498→    const ady = Math.abs(dy)
   499→    if (Math.max(adx, ady) < 28) return
   500→    if (adx > ady) moveRef.current(dx > 0 ? 'right' : 'left')
   501→    else moveRef.current(dy > 0 ? 'down' : 'up')
   502→  }
   503→
   504→  /* --- Rendu --- */
   505→  if (!game) {
   506→    /* Menu : reprise / nouvelle partie / défi du jour */
   507→    return (
   508→      <div className="checkerboard flex min-h-[360px] select-none flex-col items-center justify-center gap-5 rounded-2xl border p-6 text-center">
   509→        <div className="flex gap-2" aria-hidden="true">
   510→          {['🍒', '🍓', '🍊', '🍉'].map((e, i) => (
   511→            <span key={i} className={cn('text-4xl', !reduce && 'anim-floaty')} style={{ animationDelay: `${i * 0.4}s` }}>
   512→              {e}
   513→            </span>
   514→          ))}
   515→        </div>
   516→        <p className="max-w-md text-sm text-muted-foreground">{content.tagline[lang]}</p>
   517→        {restored ? (
   518→          <Button size="lg" className="rounded-full px-6 anim-glow" onClick={resumeGame}>
   519→            <Play className="h-4 w-4" aria-hidden="true" />
   520→            {t('game.resume')}
   521→          </Button>
   522→        ) : null}
   523→        <div className="flex flex-wrap items-center justify-center gap-2.5">
   524→          <Button size="lg" variant={restored ? 'secondary' : 'default'} className="rounded-full" onClick={() => startNew(false)}>
   525→            <Play className="h-4 w-4" aria-hidden="true" />
   526→            {t('game.newGame')}
   527→          </Button>
   528→          <Button size="lg" variant="outline" className="rounded-full" onClick={() => startNew(true)} disabled={dailyDone}>
   529→            <CalendarCheck className="h-4 w-4" aria-hidden="true" />
   530→            {dailyDone ? `${t('game.startDaily')} ✓` : t('game.startDaily')}
   531→          </Button>
   532→        </div>
   533→        {restored ? (
   534→          <p className="text-xs text-muted-foreground">
   535→            {restored.daily ? L.dailyGame : L.savedGame} — {restored.score} {L.points}
   536→          </p>
   537→        ) : null}
   538→      </div>
   539→    )
   540→  }
   541→
   542→  const winOverlay = game.won && !game.ack && !game.over
   543→  const maxTile = game.board.reduce((a, b) => Math.max(a, b), 0)
   544→
   545→  return (
   546→    <div className="relative select-none">
   547→      <style>{'@keyframes merge-pulse-g { 0% { transform: scale(1); } 45% { transform: scale(1.16); } 100% { transform: scale(1); } } .anim-merge-pulse-g { animation: merge-pulse-g 0.28s ease-out; }'}</style>
   548→      <Confetti fire={confettiKey} count={reduce ? 0 : 110} />
   549→
   550→      {/* Barre d'actions */}
   551→      <div className="mb-3 flex flex-wrap items-center gap-2">
   552→        <StatPill icon={<Hash className="h-4 w-4" aria-hidden="true" />} label={L.score} value={game.score} />
   553→        <StatPill icon={<Timer className="h-4 w-4" aria-hidden="true" />} label={L.moves} value={game.moves} />
   554→        {game.daily ? (
   555→          <span
   556→            className={cn(
   557→              'flex h-11 items-center gap-1.5 rounded-full px-3.5 text-xs font-bold',
   558→              game.daily256
   559→                ? 'bg-emerald-100 text-emerald-700 dark:bg-emerald-500/20 dark:text-emerald-300'
   560→                : 'bg-amber-100 text-amber-700 dark:bg-amber-500/20 dark:text-amber-300',
   561→            )}
   562→            title={L.dailyGoal}
   563→          >
   564→            🎯 256{game.daily256 ? ' ✅' : ''}
   565→          </span>
   566→        ) : null}
   567→        <div className="ms-auto flex flex-wrap items-center gap-2">
   568→          <div className="flex overflow-hidden rounded-full border" role="group" aria-label={t('common.mode')}>
   569→            {(['classic', 'themed'] as Mode[]).map((m) => (
   570→              <button
   571→                key={m}
   572→                type="button"
   573→                aria-pressed={game.mode === m}
   574→                onClick={() => {
   575→                  sfx.play('click', { volume: 0.5 })
   576→                  setGame((g) => (g ? { ...g, mode: m } : g))
   577→                }}
   578→                className={cn(
   579→                  'h-11 px-3.5 text-sm font-bold transition-colors',
   580→                  game.mode === m ? 'bg-primary text-primary-foreground' : 'bg-card text-muted-foreground hover:text-foreground',
   581→                )}
   582→              >
   583→                {m === 'classic' ? L.classic : L.themed}
   584→              </button>
   585→            ))}
   586→          </div>
   587→          <Button
   588→            variant="secondary"
   589→            className="h-11 rounded-full"
   590→            onClick={undo}
   591→            disabled={!game.prev || game.undos <= 0}
   592→            title={L.undo}
   593→          >
   594→            <Undo2 className="h-4 w-4" aria-hidden="true" />
   595→            {game.undos}
   596→          </Button>
   597→          <Button variant="secondary" className="h-11 rounded-full" onClick={() => startNew(game.daily)}>
   598→            <RotateCcw className="h-4 w-4" aria-hidden="true" />
   599→            <span className="hidden sm:inline">{t('game.newGame')}</span>
   600→          </Button>
   601→        </div>
   602→      </div>
   603→
   604→      {/* Plateau */}
   605→      <div
   606→        role="grid"
   607→        aria-label={L.board}
   608→        className="relative mx-auto grid aspect-square w-[min(92vw,420px)] touch-none select-none grid-cols-4 grid-rows-4 gap-2 rounded-2xl border-2 border-amber-200/70 bg-amber-100/50 p-2 dark:border-amber-500/20 dark:bg-secondary/60"
   609→        onPointerDown={onPointerDown}
   610→        onPointerUp={onPointerUp}
   611→        onPointerCancel={() => {
   612→          swipeRef.current = null
   613→        }}
   614→      >
   615→        {game.board.map((v, i) => (
   616→          <div key={i} className="relative">
   617→            {v !== 0 ? (
   618→              <div
   619→                key={`${i}-${bumps[i]}`}
   620→                className={cn(
   621→                  'absolute inset-0 flex flex-col items-center justify-center rounded-xl font-display font-extrabold leading-none shadow-sm',
   622→                  TILE_STYLE[v] ?? TILE_STYLE_BIG,
   623→                  !reduce && animKind[i] === 'new' && 'anim-pop-in',
   624→                  !reduce && animKind[i] === 'merged' && 'anim-merge-pulse-g',
   625→                )}
   626→                aria-label={String(v)}
   627→              >
   628→                {game.mode === 'themed' ? (
   629→                  <>
   630→                    <span className="text-3xl sm:text-4xl" aria-hidden="true">{FOOD[v] ?? '🍽️'}</span>
   631→                    <span className="mt-0.5 text-[11px] font-bold tabular-nums opacity-80">{v}</span>
   632→                  </>
   633→                ) : (
   634→                  <>
   635→                    <span className={cn('tabular-nums', v >= 1024 ? 'text-xl sm:text-2xl' : 'text-2xl sm:text-3xl')}>{v}</span>
   636→                    <span className="absolute end-1 top-1 text-sm sm:text-base" aria-hidden="true">{FOOD[v] ?? '🍽️'}</span>
   637→                  </>
   638→                )}
   639→              </div>
   640→            ) : null}
   641→          </div>
   642→        ))}
   643→
   644→        {/* Bannière de victoire */}
   645→        {winOverlay ? (
   646→          <div className="anim-rise absolute inset-0 z-20 flex flex-col items-center justify-center gap-3 rounded-2xl bg-card/85 p-6 text-center backdrop-blur-sm" role="status">
   647→            <span className="text-5xl" aria-hidden="true">🍉</span>
   648→            <p className="font-display text-2xl font-extrabold text-emerald-600 dark:text-emerald-400">{L.winTitle}</p>
   649→            <p className="max-w-xs text-sm text-muted-foreground">{L.winSub}</p>
   650→            <div className="mt-1 flex flex-wrap justify-center gap-2.5">
   651→              <Button size="lg" className="rounded-full" onClick={() => { sfx.play('click'); setGame((g) => (g ? { ...g, ack: true } : g)) }}>
   652→                {t('common.continue')}
   653→              </Button>
   654→              <Button size="lg" variant="secondary" className="rounded-full" onClick={() => startNew(false)}>
   655→                <RotateCcw className="h-4 w-4" aria-hidden="true" /> {t('game.newGame')}
   656→              </Button>
   657→            </div>
   658→          </div>
   659→        ) : null}
   660→
   661→        {/* Écran de fin */}
   662→        {game.over ? (
   663→          <div className="anim-rise absolute inset-0 z-20 flex flex-col items-center justify-center gap-3 rounded-2xl bg-card/90 p-6 text-center backdrop-blur-sm" role="status">
   664→            <span className="text-5xl" aria-hidden="true">🍽️</span>
   665→            <p className="font-display text-2xl font-extrabold">{L.overTitle}</p>
   666→            <div className="flex flex-wrap justify-center gap-2.5">
   667→              <StatPill icon={<Hash className="h-4 w-4" aria-hidden="true" />} label={L.finalScore} value={game.score} />
   668→              <StatPill icon={<span className="text-sm" aria-hidden="true">🍉</span>} label={L.bestTile} value={maxTile} />
   669→            </div>
   670→            <div className="mt-1 flex flex-wrap justify-center gap-2.5">
   671→              {game.prev && game.undos > 0 ? (
   672→                <Button size="lg" variant="secondary" className="rounded-full" onClick={undo}>
   673→                  <Undo2 className="h-4 w-4" aria-hidden="true" /> {L.undo} ({game.undos})
   674→                </Button>
   675→              ) : null}
   676→              <Button size="lg" className="rounded-full" onClick={() => startNew(game.daily)}>
   677→                <RotateCcw className="h-4 w-4" aria-hidden="true" /> {t('common.playAgain')}
   678→              </Button>
   679→            </div>
   680→          </div>
   681→        ) : null}
   682→      </div>
   683→    </div>
   684→  )
   685→}
   686→
   687→function StatPill({ icon, label, value }: { icon: React.ReactNode; label: string; value: React.ReactNode }) {
   688→  return (
   689→    <span className="flex h-11 items-center gap-2 rounded-full bg-secondary/70 px-3.5 text-sm font-semibold">
   690→      <span className="text-muted-foreground" aria-hidden="true">{icon}</span>
   691→      <span className="text-muted-foreground">{label}</span>
   692→      <span className="tabular-nums">{value}</span>
   693→    </span>
   694→  )
   695→}
