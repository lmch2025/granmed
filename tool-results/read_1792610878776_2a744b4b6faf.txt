   215→function pickSpeciesIndex(band: number): number {
   216→  const pool: { idx: number; w: number }[] = []
   217→  SPECIES.forEach((s, i) => {
   218→    if (s.band !== band) return
   219→    pool.push({ idx: i, w: s.rarity === 'common' ? 1 : s.rarity === 'rare' ? 0.24 : 0.06 })
   220→  })
   221→  const total = pool.reduce((a, o) => a + o.w, 0)
   222→  let r = Math.random() * total
   223→  for (const o of pool) {
   224→    r -= o.w
   225→    if (r <= 0) return o.idx
   226→  }
   227→  return pool[pool.length - 1].idx
   228→}

   230→function spawnFish(band: number, fromEdge: boolean, W: number): Fish {
   231→  const dir: 1 | -1 = Math.random() < 0.5 ? 1 : -1
   232→  const spIdx = pickSpeciesIndex(band)
   233→  return {
   234→    sp: spIdx,
   235→    x: fromEdge ? (dir === 1 ? -50 : W + 50) : Math.random() * W,
   236→    yFrac: 0.08 + Math.random() * 0.84,
   237→    dir,
   238→    speed: SPECIES[spIdx].speed * (0.85 + Math.random() * 0.3),
   239→    bobA: 5 + Math.random() * 9,
   240→    bobF: 0.5 + Math.random() * 0.8,
   241→    phase: Math.random() * Math.PI * 2,
   242→    hooked: false,
   243→    slot: 0,
   244→  }
   245→}

   247→function makeWorld(): World {
   248→  const w: World = {
   249→    W: 320, H: 240, t: 0,
   250→    fishes: [], bubbles: [], flyers: [],
   251→    hookY: 0, hookState: 'top', holding: false,
   252→    castCount: 0, bubbleT: 0, rippleT: 0, rippleY: 0,
   253→  }
   254→  for (let b = 0; b < 3; b++) {
   255→    for (let i = 0; i < 4; i++) w.fishes.push(spawnFish(b, false, w.W))
   256→  }
   257→  return w
   258→}

   261→function rr(ctx: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number) {
   262→  const rad = Math.min(r, w / 2, h / 2)
   263→  ctx.beginPath()
   264→  ctx.moveTo(x + rad, y)
   265→  ctx.arcTo(x + w, y, x + w, y + h, rad)
   266→  ctx.arcTo(x + w, y + h, x, y + h, rad)
   267→  ctx.arcTo(x, y + h, x, y, rad)
   268→  ctx.arcTo(x, y, x + w, y, rad)
   269→  ctx.closePath()
   270→}

   276→function drawWater(ctx: CanvasRenderingContext2D, W: number, H: number, surfaceY: number) {
   277→  // Ciel doux au-dessus de l'eau
   278→  const sky = ctx.createLinearGradient(0, 0, 0, surfaceY)
   279→  sky.addColorStop(0, '#ffedd2')
   280→  sky.addColorStop(1, '#f9dcb0')
   281→  ctx.fillStyle = sky
   282→  ctx.fillRect(0, 0, W, surfaceY)
   283→  // Eau vert-teal (vert dominant, jamais bleu)
   284→  const water = ctx.createLinearGradient(0, surfaceY, 0, H)
   285→  water.addColorStop(0, '#9fdcc0')
   286→  water.addColorStop(0.3, '#52ab86')
   287→  water.addColorStop(0.65, '#2a7c60')
   288→  water.addColorStop(1, '#175243')
   289→  ctx.fillStyle = water
   290→  ctx.fillRect(0, surfaceY, W, H - surfaceY)
   291→  // Assombrissement des profondeurs sur les bords
   292→  const vg = ctx.createLinearGradient(0, surfaceY, 0, H)
   293→  vg.addColorStop(0, 'rgba(10,48,36,0)')
   294→  vg.addColorStop(1, 'rgba(8,40,30,0.45)')
   295→  ctx.fillStyle = vg
   296→  ctx.fillRect(0, surfaceY, W, H - surfaceY)
   297→  // Fond marin sableux
   298→  ctx.fillStyle = '#c9a36a'
   299→  ctx.beginPath()
   300→  ctx.moveTo(0, H)
   301→  ctx.lineTo(0, H - 16)
   302→  ctx.quadraticCurveTo(W * 0.25, H - 30, W * 0.5, H - 16)
   303→  ctx.quadraticCurveTo(W * 0.75, H - 4, W, H - 20)
   304→  ctx.lineTo(W, H)
   305→  ctx.closePath()
   306→  ctx.fill()
   307→}

   309→function drawBoat(ctx: CanvasRenderingContext2D, W: number, surfaceY: number, t: number, reduce: boolean) {
   310→  const bob = reduce ? 0 : Math.sin(t * 0.9) * 3
   311→  const by = surfaceY - 9 + bob
   312→  const cx = W / 2
   313→  // Coque
   314→  const g = ctx.createLinearGradient(0, by, 0, by + 20)
   315→  g.addColorStop(0, '#b97a45')
   316→  g.addColorStop(1, '#8a5a30')
   317→  ctx.fillStyle = g
   318→  ctx.beginPath()
   319→  ctx.moveTo(cx - 58, by)
   320→  ctx.lineTo(cx + 58, by)
   321→  ctx.quadraticCurveTo(cx + 52, by + 20, cx + 34, by + 20)
   322→  ctx.lineTo(cx - 34, by + 20)
   323→  ctx.quadraticCurveTo(cx - 52, by + 20, cx - 58, by)
   324→  ctx.closePath()
   325→  ctx.fill()
   326→  ctx.fillStyle = 'rgba(255,240,214,0.4)'
   327→  ctx.fillRect(cx - 56, by + 1.5, 112, 2)
   328→  ctx.strokeStyle = 'rgba(90,58,30,0.4)'
   329→  ctx.lineWidth = 1
   330→  ctx.beginPath()
   331→  ctx.moveTo(cx - 44, by + 9)
   332→  ctx.lineTo(cx + 44, by + 9)
   333→  ctx.stroke()
   334→  // Canne
   335→  ctx.strokeStyle = '#6b4a2f'
   336→  ctx.lineWidth = 2.5
   337→  ctx.beginPath()
   338→  ctx.moveTo(cx + 16, by + 2)
   339→  ctx.quadraticCurveTo(cx + 14, by - 18, cx, by - 26)
   340→  ctx.stroke()
   341→  // Pêcheur (petit chapeau)
   342→  ctx.fillStyle = '#f2a65a'
   343→  ctx.beginPath()
   344→  ctx.arc(cx - 10, by - 6, 6, 0, Math.PI * 2)
   345→  ctx.fill()
   346→  ctx.fillStyle = '#d9553f'
   347→  ctx.beginPath()
   348→  ctx.arc(cx - 10, by - 8, 6, Math.PI, 0)
   349→  ctx.fill()
   350→}

   392→  useEffect(() => { langRef.current = lang }, [lang])

   399→  )

   401→  /* Chargement de la sauvegarde (setState via timer → conforme) */

   437→      cast: w.castCount,

   494→

   499→      const bottom = surfaceY() + (w.H - surfaceY()) * (band === 0 ? 0.3 : band === 1 ? 0.6 : 0.96)

   504→        return Math.max(surfaceY() + 10, w.hookY - 16 - f.slot * 26)

   512→
   513→    /* ----- entrées ----- */
   514→    const onDown = (e: PointerEvent) => {

   518→      cv.setPointerCapture(e.pointerId)
   519→    }

   524→        e.preventDefault()
   525→        w.holding = true
   526→      }
   527→    }

   529→      if (e.code === 'Space' || e.code === 'ArrowDown') w.holding = false
   530→    }

   534→    cv.addEventListener('pointerup', onUp)
   535→    cv.addEventListener('pointercancel', onUp)
   536→    cv.addEventListener('contextmenu', onCtx)
   537→    window.addEventListener('keydown', onKeyDown)
   538→    window.addEventListener('keyup', onKeyUp)
   539→    document.addEventListener('visibilitychange', onVis)

   541→    /* ----- résolution d'une remontée ----- */

   543→      const hooked = w.fishes.filter((f) => f.hooked)
   544→      const n = hooked.length

   548→      let newSpecies = false
   549→      let deepCaught = false
   550→      let legendary = false
   551→      for (const f of hooked) {
   552→        const spec = SPECIES[f.sp]
   553→        xp += RARITY_XP[spec.rarity]
   554→        if (spec.band === 2) deepCaught = true
   555→        if (spec.rarity === 'legendary') legendary = true
   556→        if (!caughtRef.current.has(spec.id)) {
   557→          newSpecies = true
   558→          caughtRef.current.add(spec.id)
   559→          discoveryQueueRef.current.push(spec)
   560→        }
   561→        // Vol vers le compteur
   562→        w.flyers.push({ sp: f.sp, x0: fishX(f), y0: fishY(f), t: 0, delay: f.slot * 0.09 })
   563→      }
   564→      // Remplace les poissons pêchés
   565→      w.fishes = w.fishes.filter((f) => !f.hooked)
   566→      for (const f of hooked) w.fishes.push(spawnFish(SPECIES[f.sp].band, true, w.W))
   567→      w.castCount = 0
   568→      // Gamification
   569→      addXp(xp)
   570→      setBest(gameId, 'best-catch', n)
   571→      const size = caughtRef.current.size
   572→      if (newSpecies) {
   573→        setBest(gameId, 'species', size)
   574→        setCaught(new Set(caughtRef.current))
   575→        if (size >= 10) unlockBadge('fish-10')
   576→        if (hooked.some((f) => SPECIES[f.sp].rarity !== 'common')) unlockBadge('fish-rare')
   577→        processDiscovery()
   578→      }
   579→      if (hooked.some((f) => SPECIES[f.sp].rarity !== 'common')) unlockBadge('fish-rare')
   580→      if (n >= 5) addWin(gameId)
   581→      if (legendary) setConfetti((k) => k + 1)
   582→      // Défis du jour
   583→      const day = dayRef.current
   584→      day.dayCount += n
   585→      day.bestCastToday = Math.max(day.bestCastToday, n)
   586→      if (deepCaught) day.deepToday = true
   587→      setDayStats({ dayCount: day.dayCount, bestCast: day.bestCastToday, deep: day.deepToday })
   588→      setHud((h) => ({ ...h, today: day.dayCount, species: caughtRef.current.size }))
   589→      persist()

   591→        dailyGoal === 0 ? day.bestCastToday >= 5
   592→        : dailyGoal === 1 ? day.deepToday
   593→        : day.dayCount >= 10

   595→        completeDaily(gameId)
   596→        sfx.play('win')
   597→        setConfetti((k) => k + 1)
   598→      }

   601→    /* ----- simulation ----- */
   602→    let lastDepth = -1
   603→    const update = (dt: number) => {
   604→      const playing = phaseRef.current === 'playing'
   605→      // Hameçon
   606→      if (playing) {
   607→        if (w.hookState === 'top' && w.holding) {
   608→          w.hookState = 'down'
   609→          w.rippleT = 0.5
   610→          w.rippleY = surfaceY()
   611→          sfx.play('splash', { volume: 0.6 })
   612→        } else if (w.hookState === 'down') {
   613→          w.hookY += 235 * dt
   614→          if (!w.holding) w.hookState = 'up'
   615→          if (w.hookY >= bedY()) {
   616→            w.hookY = bedY()
   617→            if (!w.holding) w.hookState = 'up'
   618→          }
   619→        } else if (w.hookState === 'up') {
   620→          w.hookY -= 270 * dt
   621→          // Capture à la remontée
   622→          for (const f of w.fishes) {
   623→            if (f.hooked) continue
   624→            const spec = SPECIES[f.sp]
   625→            const r = spec.size / 2
   626→            const fx = fishX(f)
   627→            const fy = fishY(f)
   628→            if (Math.abs(fx - w.W / 2) < r + 14 && Math.abs(fy - w.hookY) < r + 16) {
   629→              f.hooked = true
   630→              f.slot = w.castCount
   631→              w.castCount++
   632→              sfx.play('splash', { volume: 0.5 })
   633→              pushHud()
   634→            }
   635→          }
   636→          if (w.hookY <= surfaceY() + 8) {
   637→            w.hookY = surfaceY() + 8
   638→            w.hookState = 'top'
   639→            resolveCast()
   640→            pushHud()
   641→          }
   642→        }
   643→      }
   644→
   645→      // Poissons
   646→      for (const f of w.fishes) {
   647→        if (f.hooked) continue
   648→        f.x += f.dir * f.speed * dt
   649→        if (f.x < -70 && f.dir === -1) {
   650→          const nf = spawnFish(SPECIES[f.sp].band, false, w.W)
   651→          f.sp = nf.sp
   652→          f.yFrac = nf.yFrac
   653→          f.dir = -1
   654→          f.speed = SPECIES[f.sp].speed * (0.85 + Math.random() * 0.3)
   655→          f.bobA = nf.bobA
   656→          f.bobF = nf.bobF
   657→          f.phase = nf.phase
   658→          f.x = w.W + 60
   659→        } else if (f.x > w.W + 70 && f.dir === 1) {
   660→          const nf = spawnFish(SPECIES[f.sp].band, false, w.W)
   661→          f.sp = nf.sp
   662→          f.yFrac = nf.yFrac
   663→          f.dir = 1
   664→          f.speed = SPECIES[f.sp].speed * (0.85 + Math.random() * 0.3)
   665→          f.bobA = nf.bobA
   666→          f.bobF = nf.bobF
   667→          f.phase = nf.phase
   668→          f.x = -60
   669→        }
   670→      }
   671→
   672→      // Bulles
   673→      w.bubbleT -= dt
   674→      if (w.bubbleT <= 0 && !reduceRef.current) {
   675→        w.bubbleT = 0.35 + Math.random() * 0.4
   676→        w.bubbles.push({
   677→          x: 14 + Math.random() * (w.W - 28),
   678→          y: bedY() - 4,
   679→          r: 1.5 + Math.random() * 3,
   680→          vy: 26 + Math.random() * 30,
   681→          phase: Math.random() * Math.PI * 2,
   682→        })
   683→      }
   684→      for (let i = w.bubbles.length - 1; i >= 0; i--) {
   685→        const b = w.bubbles[i]
   686→        b.y -= b.vy * dt
   687→        b.x += Math.sin(w.t * 2 + b.phase) * 8 * dt
   688→        if (b.y < surfaceY() + 4) w.bubbles.splice(i, 1)
   689→      }
   690→
   691→      // Vols vers le compteur
   692→      for (let i = w.flyers.length - 1; i >= 0; i--) {
   693→        const fl = w.flyers[i]
   694→        if (fl.delay > 0) {
   695→          fl.delay -= dt
   696→          continue
   697→        }
   698→        fl.t += dt / 0.55
   699→        if (fl.t >= 1) w.flyers.splice(i, 1)
   700→      }
   701→
   702→      if (w.rippleT > 0) w.rippleT -= dt
   703→
   704→      // Profondeur (HUD)
   705→      const depth = Math.max(0, Math.round((w.hookY - surfaceY()) / 10))
   706→      if (playing && depth !== lastDepth) {
   707→        lastDepth = depth
   708→        pushHud()
   709→      }
   710→    }
   711→
   712→    /* ----- rendu ----- */
   713→    const draw = () => {
   714→      const W = w.W, H = w.H
   715→      const sY = surfaceY()
   716→      ctx.setTransform(dpr, 0, 0, dpr, 0, 0)
   717→      ctx.clearRect(0, 0, W, H)
   718→      drawWater(ctx, W, H, sY)
   719→
   720→      // Rayons de lumière
   721→      ctx.save()
   722→      for (let i = 0; i < 3; i++) {
   723→        const sway = reduceRef.current ? 0 : Math.sin(w.t * 0.35 + i * 2) * 26
   724→        const rx = W * (0.22 + i * 0.3) + sway
   725→        ctx.fillStyle = 'rgba(255,246,220,0.07)'
   726→        ctx.beginPath()
   727→        ctx.moveTo(rx - 14, sY)
   728→        ctx.lineTo(rx + 14, sY)
   729→        ctx.lineTo(rx + 76, H)
   730→        ctx.lineTo(rx + 20, H)
   731→        ctx.closePath()
   732→        ctx.fill()
   733→      }
   734→      ctx.restore()
   735→
   736→      // Algues
   737→      for (let i = 0; i < 4; i++) {
   738→        const sx = W * (0.1 + i * 0.26)
   739→        const sway = reduceRef.current ? 0 : Math.sin(w.t * 1.1 + i) * 8
   740→        ctx.strokeStyle = 'rgba(46,124,92,0.75)'
   741→        ctx.lineWidth = 3
   742→        ctx.lineCap = 'round'
   743→        ctx.beginPath()
   744→        ctx.moveTo(sx, H - 8)
   745→        ctx.quadraticCurveTo(sx + sway, H - 26, sx - sway * 0.6, H - 44 - (i % 2) * 10)
   746→        ctx.stroke()
   747→      }
   748→
   749→      // Bulles
   750→      for (const b of w.bubbles) {
   751→        ctx.strokeStyle = 'rgba(240,255,246,0.5)'
   752→        ctx.lineWidth = 1
   753→        ctx.beginPath()
   754→        ctx.arc(b.x, b.y, b.r, 0, Math.PI * 2)
   755→        ctx.stroke()
   756→      }
   757→
   758→      // Ligne + hameçon
   759→      const hookX = W / 2
   760→      ctx.strokeStyle = 'rgba(244,233,208,0.75)'
   761→      ctx.lineWidth = 1.2
   762→      ctx.beginPath()
   763→      ctx.moveTo(hookX, sY - 35)
   764→      ctx.lineTo(hookX, w.hookState === 'top' ? sY + 8 : w.hookY)
   765→      ctx.stroke()
   766→      if (w.hookState !== 'top' || phaseRef.current === 'playing') {
   767→        const hy = w.hookState === 'top' ? sY + 8 : w.hookY
   768→        ctx.strokeStyle = '#e8e0c8'
   769→        ctx.lineWidth = 2.5
   770→        ctx.beginPath()
   771→        ctx.arc(hookX + 2, hy + 4, 6, -Math.PI * 0.4, Math.PI * 0.75)
   772→        ctx.stroke()
   773→        ctx.fillStyle = '#e2725b'
   774→        ctx.beginPath()
   775→        ctx.arc(hookX, hy - 1, 3, 0, Math.PI * 2)
   776→        ctx.fill()

   779→      // Poissons
   780→      ctx.textAlign = 'center'
   781→      ctx.textBaseline = 'middle'
   782→      for (const f of w.fishes) {
   783→        const spec = SPECIES[f.sp]
   784→        const fx = fishX(f)
   785→        const fy = fishY(f)
   786→        if (fx < -60 || fx > W + 60) continue
   787→        ctx.save()
   788→        ctx.translate(fx, fy)
   789→        if (f.hooked) ctx.rotate(Math.sin(w.t * 6 + f.slot) * 0.25)
   790→        else ctx.scale(f.dir === 1 ? -1 : 1, 1)
   791→        if (spec.rarity === 'legendary') {
   792→          const g = ctx.createRadialGradient(0, 0, 2, 0, 0, spec.size)
   793→          g.addColorStop(0, 'rgba(247,200,115,0.65)')
   794→          g.addColorStop(1, 'rgba(247,200,115,0)')
   795→          ctx.fillStyle = g
   796→          ctx.beginPath()
   797→          ctx.arc(0, 0, spec.size, 0, Math.PI * 2)
   798→          ctx.fill()
   799→        }
   800→        ctx.font = emojiFont(spec.size)
   801→        ctx.fillText(spec.emoji, 0, 0)
   802→        if (spec.rarity === 'legendary') {
   803→          ctx.font = emojiFont(12)
   804→          ctx.fillText('✨', -spec.size * 0.55, -spec.size * 0.45)

   806→        ctx.restore()

   808→

   810→      for (const fl of w.flyers) {
   811→        if (fl.delay > 0) continue
   812→        const spec = SPECIES[fl.sp]
   813→        const t = clamp(fl.t, 0, 1)
   814→        const ex = W - 52, ey = 24
   815→        const cxp = (fl.x0 + ex) / 2
   816→        const cyp = Math.min(fl.y0, ey) - 110
   817→        const x = (1 - t) * (1 - t) * fl.x0 + 2 * (1 - t) * t * cxp + t * t * ex
   818→        const y = (1 - t) * (1 - t) * fl.y0 + 2 * (1 - t) * t * cyp + t * t * ey
   819→        ctx.save()
   820→        ctx.globalAlpha = 1 - t * 0.25
   821→        ctx.translate(x, y)
   822→        ctx.scale(1 - t * 0.4, 1 - t * 0.4)
   823→        ctx.font = emojiFont(spec.size)
   824→        ctx.fillText(spec.emoji, 0, 0)
   825→        ctx.restore()
   826→      }
   827→
   828→      // Bateau par-dessus la ligne
   829→      drawBoat(ctx, W, sY, w.t, reduceRef.current)
   830→
   831→      // Onde de plongée
   832→      if (w.rippleT > 0) {
   833→        ctx.strokeStyle = `rgba(255,255,255,${w.rippleT})`
   834→        ctx.lineWidth = 2
   835→        ctx.beginPath()
   836→        ctx.ellipse(hookX, w.rippleY, 26 * (1 - w.rippleT) + 6, 7, 0, 0, Math.PI * 2)
   837→        ctx.stroke()
   838→      }
   839→    }
   840→
   841→    let last = performance.now()
   842→    const step = (now: number) => {
   843→      rafRef.current = requestAnimationFrame(step)
   844→      const dt = Math.min((now - last) / 1000, 0.05)
   845→      last = now
   846→      if (pausedRef.current) return
   847→      w.t += dt
   848→      update(dt)
   849→      draw()
   850→    }
   851→    rafRef.current = requestAnimationFrame(step)
   852→
   853→    const ro = new ResizeObserver(resize)
   854→    ro.observe(wrap)
   855→
   856→    return () => {
   857→      cancelAnimationFrame(rafRef.current)
   858→      ro.disconnect()
   859→      cv.removeEventListener('pointerdown', onDown)
   860→      cv.removeEventListener('pointerup', onUp)
   861→      cv.removeEventListener('pointercancel', onUp)
   862→      cv.removeEventListener('contextmenu', onCtx)
   863→      window.removeEventListener('keydown', onKeyDown)
   864→      window.removeEventListener('keyup', onKeyUp)
   865→      document.removeEventListener('visibilitychange', onVis)
   866→    }
   867→  }, [addWin, addXp, completeDaily, dailyGoal, persist, processDiscovery, pushHud, setBest, unlockBadge])
   868→
   869→  /* ================= Rendu React ================= */
   870→  const rarityChip = (r: Rarity) =>
   871→    r === 'common'
   872→      ? 'bg-emerald-100 text-emerald-700 dark:bg-emerald-500/15 dark:text-emerald-300'
   873→      : r === 'rare'
   874→        ? 'bg-violet-100 text-violet-700 dark:bg-violet-500/15 dark:text-violet-300'
   875→        : 'bg-amber-100 text-amber-700 dark:bg-amber-500/20 dark:text-amber-300'
   876→
   877→  const goalProgress = (i: number) => {
   878→    if (i === 0) return dayStats.bestCast >= 5
   879→    if (i === 1) return dayStats.deep
   880→    return dayStats.dayCount >= 10
   881→  }
   882→
   883→  return (
   884→    <div className="relative select-none">
   885→      <Confetti fire={confetti} count={reduce ? 0 : 80} />
   886→
   887→      {/* Barre d'état */}
   888→      <div className="mb-3 flex flex-wrap items-center gap-2">
   889→        {phase === 'playing' ? (
   890→          <>
   891→            <StatChip icon="🎣" label={T.cast} value={`×${hud.cast}`} />
   892→            <StatChip icon="📅" label={T.today} value={hud.today} />
