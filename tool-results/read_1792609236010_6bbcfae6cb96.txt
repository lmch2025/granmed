   118→   118→type Diff = 'easy' | 'medium' | 'hard'
   119→   119→type Op = '+' | '-' | '×' | '÷'
   120→   120→
   121→   121→interface Q {
   122→   122→  a: number
   123→   123→  op: Op
   124→   124→  b: number
   125→   125→  answer: number
   126→   126→  options: number[]
   127→   127→}

   131→   131→function makeQuestion(diff: Diff, boost: number, rand: () => number): Q {
   132→   132→  const ri = (lo: number, hi: number) => lo + Math.floor(rand() * (hi - lo + 1))
   133→   133→  let a = 0
   134→   134→  let b = 0
   135→   135→  let op: Op = '+'
   136→   136→  if (diff === 'easy') {
   137→   137→    op = rand() < 0.5 ? '+' : '-'
   138→   138→    a = ri(2, 20 + boost * 3)
   139→   139→    b = ri(2, 20 + boost * 3)
   140→   140→    if (op === '-' && b > a) [a, b] = [b, a]
   141→   141→  } else if (diff === 'medium') {
   142→   142→    const r = rand()
   143→   143→    if (r < 0.35) {
   144→   144→      op = '+'
   145→   145→      a = ri(2, 30 + boost * 4)
   146→   146→      b = ri(2, 30 + boost * 4)
   147→   147→    } else if (r < 0.7) {
   148→   148→      op = '-'
   149→   149→      a = ri(2, 30 + boost * 4)
   150→   150→      b = ri(2, 30 + boost * 4)
   151→   151→      if (b > a) [a, b] = [b, a]
   152→   152→    } else {
   153→   153→      op = '×'
   154→   154→      a = ri(2, 9)
   155→   155→      b = ri(2, 9 + boost)
   156→   156→    }
   157→   157→  } else {
   158→   158→    const r = rand()
   159→   159→    if (r < 0.25) {
   160→   160→      op = '+'
   161→   161→      a = ri(5, 60 + boost * 5)
   162→   162→      b = ri(5, 60 + boost * 5)
   163→   163→    } else if (r < 0.5) {
   164→   164→      op = '-'
   165→   165→      a = ri(5, 60 + boost * 5)
   166→   166→      b = ri(5, 60 + boost * 5)
   167→   167→      if (b > a) [a, b] = [b, a]
   168→   168→    } else if (r < 0.75) {
   169→   169→      op = '×'
   170→   170→      a = ri(3, 12)
   171→   171→      b = ri(3, 9)
   172→   172→    } else {
   173→   173→      op = '÷'
   174→   174→      b = ri(2, 9)
   175→   175→      const quotient = ri(2, 12 + boost * 2)
   176→   176→      a = b * quotient
   177→   177→    }
   178→   178→  }
   179→   179→  const answer = op === '+' ? a + b : op === '-' ? a - b : op === '×' ? a * b : a / b
   180→   180→  // Distracteurs plausibles, tous différents et positifs ou nuls.
   181→   181→  const candidates = shuffle(
   182→   182→    [answer + 1, answer - 1, answer + 2, answer - 2, answer + 10, answer - 10, answer + 5, answer - 5, a, b, answer + 4, answer - 4],
   183→   183→    rand,
   184→   184→  )
   185→   185→  const options = [answer]
   186→   186→  for (const c of candidates) {
   187→   187→    if (options.length >= 4) break
   188→   188→    if (c >= 0 && !options.includes(c)) options.push(c)
   189→   189→  }
   190→   190→  let extra = 3
   191→   191→  while (options.length < 4) {
   192→   192→    extra += 3
   193→   193→    if (!options.includes(answer + extra)) options.push(answer + extra)
   194→   194→  }
   195→   195→  return { a, op, b, answer, options: shuffle(options, rand) }
   196→   196→}
   197→   197→
   198→   198→/** Série déterministe du défi du jour (difficulté tirée de la graine). */
   199→   199→function dailySeries(): { diff: Diff; questions: Q[] } {
   200→   200→  const rand = mulberry32(seedFromString(`${gameId}:${todayKey()}`))
   201→   201→  const diff: Diff = (['easy', 'medium', 'hard'] as Diff[])[Math.floor(rand() * 3)]
   202→   202→  const questions = Array.from({ length: 240 }, (_, i) => makeQuestion(diff, Math.floor(i / 12), rand))
   203→   203→  return { diff, questions }
   204→   204→}
