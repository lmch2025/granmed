   320→   320→function emptyBoard(n: number): Cell[] {
   321→   321→  return Array.from({ length: n }, () => ({ mine: false, adj: 0, revealed: false, flagged: false }))
   322→   322→}

   324→   324→function neighborsOf(idx: number, w: number, h: number): number[] {
   325→   325→  const x = idx % w
   326→   326→  const y = Math.floor(idx / w)
   327→   327→  const out: number[] = []
   328→   328→  for (let dy = -1; dy <= 1; dy++) {
   329→   329→    for (let dx = -1; dx <= 1; dx++) {
   330→   330→      if (dx === 0 && dy === 0) continue
   331→   331→      const nx = x + dx
   332→   332→      const ny = y + dy
   333→   333→      if (nx >= 0 && nx < w && ny >= 0 && ny < h) out.push(ny * w + nx)
   334→   334→    }
   335→   335→  }
   336→   336→  return out
   337→   337→}

   339→   339→/** Place les mines en excluant la case sûre et ses voisines, puis calcule les compteurs. */
   340→   340→function placeMines(cells: Cell[], w: number, h: number, mines: number, safeIdx: number, rand: () => number): void {
   341→   341→  const forbidden = new Set<number>([safeIdx, ...neighborsOf(safeIdx, w, h)])
   342→   342→  const candidates: number[] = []
   343→   343→  for (let i = 0; i < cells.length; i++) {
   344→   344→    if (!forbidden.has(i)) candidates.push(i)
   345→   345→  }
   346→   346→  const count = Math.min(mines, candidates.length)
   347→   347→  for (let i = 0; i < count; i++) {
   348→   348→    const j = i + Math.floor(rand() * (candidates.length - i))
   349→   349→    const tmp = candidates[i]
   350→   350→    candidates[i] = candidates[j]
   351→   351→    candidates[j] = tmp
   352→   352→    cells[candidates[i]].mine = true
   353→   353→  }
   354→   354→  for (let i = 0; i < cells.length; i++) {
   355→   355→    if (cells[i].mine) continue
   356→   356→    let n = 0
   357→   357→    for (const k of neighborsOf(i, w, h)) {
   358→   358→      if (cells[k].mine) n++
   359→   359→    }
   360→   360→    cells[i].adj = n
   361→   361→  }
   362→   362→}

   364→   364→/** Révélation en cascade (pile) ; retourne l’index de la mine touchée, ou -1 si rien n’explose. */
   365→   365→function floodReveal(cells: Cell[], w: number, h: number, start: number): number {
   366→   366→  if (cells[start].revealed || cells[start].flagged) return -1
   367→   367→  if (cells[start].mine) {
   368→   368→    cells[start].revealed = true
   369→   369→    return start
   370→   370→  }
   371→   371→  const stack = [start]
   372→   372→  while (stack.length > 0) {
   373→   373→    const i = stack.pop() as number
   374→   374→    if (cells[i].revealed || cells[i].flagged) continue
   375→   375→    cells[i].revealed = true
   376→   376→    if (cells[i].adj === 0) {
   377→   377→      for (const n of neighborsOf(i, w, h)) {
   378→   378→        if (!cells[n].revealed && !cells[n].mine) stack.push(n)
   379→   379→      }
   380→   380→    }
   381→   381→  }
   382→   382→  return -1
   383→   383→}

   386→   386→  let n = 0
   387→   387→  for (const c of cells) if (c.revealed) n++
   388→   388→  return n
   389→   389→}

   593→   593→      finishLose(hit, next)
   594→   594→      return
   595→   595→    }
   596→   596→    if (opened > 0) sfx.play(opened >= 4 ? 'sweep' : 'pop')
   597→   597→    if (countRevealed(next) === next.length - c.mines) finishWin(next)
   598→   598→  }
