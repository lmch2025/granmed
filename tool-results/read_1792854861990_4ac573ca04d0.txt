   170→/** é→e, è→e, à→a, ç→c… et ligatures œ→oe, æ→ae (décomposition NFD + retrait des diacritiques). */
   171→function foldAccents(s: string): string {
   172→  return s
   173→    .replace(/œ/g, 'oe')
   174→    .replace(/Œ/g, 'OE')
   175→    .replace(/æ/g, 'ae')
   176→    .replace(/Æ/g, 'AE')
   177→    .normalize('NFD')
   178→    .replace(/[\u0300-\u036f]/g, '')
   179→}
   180→
   181→type Slot = { kind: 'sep'; ch: string } | { kind: 'letter'; ch: string; li: number }
   182→
   183→interface ParsedExpression {
   184→  slots: Slot[]
   185→  letters: string
   186→}
   187→
   188→/** Découpe l'expression : une case par lettre (index li), séparateurs toujours visibles. */
   189→function parseExpression(expr: string): ParsedExpression {
   190→  const slots: Slot[] = []
   191→  let letters = ''
   192→  const folded = foldAccents(expr).replace(/’/g, "'")
   193→  for (const ch of folded) {
   194→    if (/^[a-z]$/i.test(ch)) {
   195→      const up = ch.toUpperCase()
   196→      slots.push({ kind: 'letter', ch: up, li: letters.length })
   197→      letters += up
   198→    } else {
   199→      slots.push({ kind: 'sep', ch })
   200→    }
   201→  }
   202→  return { slots, letters }
   203→}

   208→function pickDaily(lang: 'fr' | 'en', count: number): IdiomEntry[] {
   209→  const pool = lang === 'fr' ? FR_IDIOMS : EN_IDIOMS
   210→  const rand = mulberry32(seedFromString(`${gameId}:${todayKey()}`))
   211→  const idxs = new Set<number>()
   212→  while (idxs.size < count) idxs.add(Math.floor(rand() * pool.length))
   213→  return [...idxs].map((i) => pool[i])
   214→}
   215→
   216→function pickFree(lang: 'fr' | 'en', count: number): IdiomEntry[] {
   217→  const pool = lang === 'fr' ? FR_IDIOMS : EN_IDIOMS
   218→  return shuffle(
   219→    pool.map((_, i) => i),
   220→    Math.random,
   221→  )
   222→    .slice(0, count)
   223→    .map((i) => pool[i])
   224→}
