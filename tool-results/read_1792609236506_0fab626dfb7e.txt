   133→function popcount9(v: number): number {
   134→  let n = 0
   135→  while (v) {
   136→    v &= v - 1
   137→    n++
   138→  }
   139→  return n
   140→}

   142→/** Étape 1 : remplit une grille complète valide par backtracking randomisé. */
   143→function generateSolution(rand: () => number): number[] {
   144→  const grid = new Array<number>(81).fill(0)
   145→  const rows = new Array<number>(9).fill(0)
   146→  const cols = new Array<number>(9).fill(0)
   147→  const boxes = new Array<number>(9).fill(0)
   148→  const digits = [1, 2, 3, 4, 5, 6, 7, 8, 9]
   149→
   150→  function fill(pos: number): boolean {
   151→    if (pos === 81) return true
   152→    const r = (pos / 9) | 0
   153→    const c = pos % 9
   154→    const b = ((r / 3) | 0) * 3 + ((c / 3) | 0)
   155→    for (const d of shuffle(digits, rand)) {
   156→      const bit = 1 << d
   157→      if ((rows[r] | cols[c] | boxes[b]) & bit) continue
   158→      grid[pos] = d
   159→      rows[r] |= bit
   160→      cols[c] |= bit
   161→      boxes[b] |= bit
   162→      if (fill(pos + 1)) return true
   163→      grid[pos] = 0
   164→      rows[r] ^= bit
   165→      cols[c] ^= bit
   166→      boxes[b] ^= bit
   167→    }
   168→    return false
   169→  }
   170→  fill(0)
   171→  return grid
   172→}

   174→/** Étape 2 : compte les solutions d'une grille (arrêt dès `limit` atteint). Ne modifie pas l'entrée. */
   175→function countSolutions(input: number[], limit: number): number {
   176→  const g = input.slice()
   177→  const rows = new Array<number>(9).fill(0)
   178→  const cols = new Array<number>(9).fill(0)
   179→  const boxes = new Array<number>(9).fill(0)
   180→  for (let i = 0; i < 81; i++) {
   181→    if (g[i] === 0) continue
   182→    const r = (i / 9) | 0
   183→    const c = i % 9
   184→    const b = ((r / 3) | 0) * 3 + ((c / 3) | 0)
   185→    const bit = 1 << g[i]
   186→    rows[r] |= bit
   187→    cols[c] |= bit
   188→    boxes[b] |= bit
   189→  }
   190→  let count = 0
   191→
   192→  function backtrack(): boolean {
   193→    // MRV : chercher la case vide avec le moins de candidats
   194→    let best = -1
   195→    let bestMask = 0
   196→    let bestN = 10
   197→    for (let i = 0; i < 81; i++) {
   198→      if (g[i] !== 0) continue
   199→      const r = (i / 9) | 0
   200→      const c = i % 9
   201→      const b = ((r / 3) | 0) * 3 + ((c / 3) | 0)
   202→      const mask = ~(rows[r] | cols[c] | boxes[b]) & 0x3fe
   203→      const n = popcount9(mask)
   204→      if (n === 0) return false // impasse
   205→      if (n < bestN) {
   206→        bestN = n
   207→        best = i
   208→        bestMask = mask
   209→        if (n === 1) break
   210→      }
   211→    }
   212→    if (best === -1) {
   213→      count++
   214→      return count >= limit
   215→    }
   216→    const r = (best / 9) | 0
   217→    const c = best % 9
   218→    const b = ((r / 3) | 0) * 3 + ((c / 3) | 0)
   219→    for (let d = 1; d <= 9; d++) {
   220→      const bit = 1 << d
   221→      if (!(bestMask & bit)) continue
   222→      g[best] = d
   223→      rows[r] |= bit
   224→      cols[c] |= bit
   225→      boxes[b] |= bit
   226→      const stop = backtrack()
   227→      g[best] = 0
   228→      rows[r] ^= bit
   229→      cols[c] ^= bit
   230→      boxes[b] ^= bit
   231→      if (stop) return true
   232→    }
   233→    return false
   234→  }
   235→
   236→  backtrack()
   237→  return count
   238→}

   240→/** Étape 3 : retire des chiffres un à un en vérifiant l'unicité de la solution À CHAQUE tentative. */
   241→function makePuzzle(targetGivens: number, rand: () => number): { givens: number[]; solution: number[] } {
   242→  const solution = generateSolution(rand)
   243→  const givens = solution.slice()
   244→  const positions = shuffle(Array.from({ length: 81 }, (_, i) => i), rand)
   245→  let toRemove = 81 - targetGivens
   246→  for (const pos of positions) {
   247→    if (toRemove <= 0) break
   248→    const backup = givens[pos]
   249→    if (backup === 0) continue
   250→    givens[pos] = 0
   251→    if (countSolutions(givens, 2) === 1) {
   252→      toRemove--
   253→    } else {
   254→      givens[pos] = backup // l'unicité serait perdue : on remet le chiffre
   255→    }
   256→  }
   257→  return { givens, solution }
   258→}

   260→/** Cases voisines (ligne + colonne + bloc) précalculées. */
   261→const PEERS: number[][] = (() => {
   262→  const peers: number[][] = []
   263→  for (let i = 0; i < 81; i++) {
   264→    const r = (i / 9) | 0
   265→    const c = i % 9
   266→    const set = new Set<number>()
   267→    for (let k = 0; k < 9; k++) {
   268→      set.add(r * 9 + k)
   269→      set.add(k * 9 + c)
   270→    }
   271→    const br = ((r / 3) | 0) * 3
   272→    const bc = ((c / 3) | 0) * 3
   273→    for (let dr = 0; dr < 3; dr++) {
   274→      for (let dc = 0; dc < 3; dc++) set.add((br + dr) * 9 + (bc + dc))
   275→    }
   276→    set.delete(i)
   277→    peers.push([...set])
   278→  }
   279→  return peers
   280→})()

   287→function findConflicts(grid: number[]): Set<number> {
   288→  const out = new Set<number>()
   289→  for (let i = 0; i < 81; i++) {
   290→    const v = grid[i]
   291→    if (!v) continue
   292→    for (const j of PEERS[i]) {
   293→      if (grid[j] === v) {
   294→        out.add(i)
   295→        out.add(j)
   296→      }
   297→    }
   298→  }
   299→  return out
   300→}

   302→function conflictsWith(grid: number[], i: number, v: number): boolean {
   303→  for (const j of PEERS[i]) if (grid[j] === v) return true
   304→  return false

   563→  function inputDigit(d: number) {
   564→    if (phase !== 'playing' || !running || sel < 0) return
   565→    if (givens[sel] !== 0 || hinted.includes(sel)) {
   566→      sfx.play('wrong', { volume: 0.25 })
   567→      return
   568→    }
   569→    if (notesMode) {
   570→      if (grid[sel] !== 0) {
   571→        sfx.play('click', { volume: 0.3 })
   572→        return
   573→      }
   574→      const nn = [...notes]
   575→      nn[sel] ^= 1 << d
   576→      setNotes(nn)
   577→      sfx.play('click', { volume: 0.4 })
   578→      return
   579→    }
   580→    const ng = [...grid]
   581→    ng[sel] = ng[sel] === d ? 0 : d
   582→    const nn = [...notes]
   583→    nn[sel] = 0
   584→    setGrid(ng)
   585→    setNotes(nn)
   586→    if (ng[sel] === 0) sfx.play('click', { volume: 0.4 })
   587→    else sfx.play('pop', { volume: 0.6 })
   588→    if (ng[sel] !== 0 && autoCheck && conflictsWith(ng, sel, ng[sel])) sfx.play('wrong', { volume: 0.3 })
   589→    checkWin(ng)
   590→  }

   592→  function eraseCell() {
   593→    if (phase !== 'playing' || !running || sel < 0) return
   594→    if (givens[sel] !== 0 || hinted.includes(sel)) return
   595→    if (grid[sel] === 0 && notes[sel] === 0) return

   597→    ng[sel] = 0
   598→    const nn = [...notes]
   599→    nn[sel] = 0
   600→    setGrid(ng)
   601→    setNotes(nn)
   602→    sfx.play('sweep', { volume: 0.4 })

   605→  function giveHint() {
   606→    if (phase !== 'playing' || !running || hintsUsed >= 3) return
   607→    const fixable = (i: number) => givens[i] === 0 && !hinted.includes(i) && grid[i] !== solution[i]
   608→    let target = sel
   609→    if (target < 0 || !fixable(target)) {
   610→      const candidates: number[] = []
   611→      for (let i = 0; i < 81; i++) if (fixable(i)) candidates.push(i)
   612→      if (candidates.length === 0) return
   613→      target = candidates[Math.floor(Math.random() * candidates.length)]
   614→    }
   615→    const ng = [...grid]
   616→    ng[target] = solution[target]
   617→    const nn = [...notes]
   618→    nn[target] = 0
   619→    setGrid(ng)
   620→    setNotes(nn)
   621→    setHinted([...hinted, target])
   622→    setHintsUsed((h) => h + 1)
   623→    setSel(target)
   624→    sfx.play('correct')
   625→    checkWin(ng, hintsUsed + 1)
   626→  }

   628→  function verify() {
   629→    if (phase !== 'playing') return
   630→    let wrong = 0
   631→    for (let i = 0; i < 81; i++) if (grid[i] !== 0 && grid[i] !== solution[i]) wrong++

   633→      sfx.play('correct')
   634→      toast('✅', { description: L.noErrors, duration: 3500 })
   635→    } else {
   636→      sfx.play('wrong')
   637→      toast('⚠️', { description: L.errors(wrong), duration: 3500 })
   638→    }
   639→  }

   642→    for (let i = 0; i < 81; i++) if (g[i] !== solution[i]) return

   647→  function finishWin(hintsOverride?: number) {
   648→    const ms = currentElapsed()

   656→    sfx.play('win')
   657→    addWin(gameId)
   658→    setBest(gameId, `best-time-${diff}`, Math.round(ms / 100) / 10, true)
   659→    addXp(xp)
   660→    if (diff === 'hard') unlockBadge('sudoku-hard')
   661→    if (diff === 'expert' && hu === 0) unlockBadge('sudoku-expert-nohints')
   662→    if (daily) completeDaily(gameId)
   663→    clearSave(gameId)
   664→  }

   666→  function handleKey(e: KeyboardEvent) {
   667→    if (phase !== 'playing' || !running) return

   676→      e.preventDefault()
