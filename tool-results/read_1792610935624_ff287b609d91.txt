   122→const ROUND_MS = 60000

   124→const MIN_WORD_LEN = 3
   125→const VOWELS = new Set(['a', 'e', 'i', 'o', 'u'])

   127→function wordsFor(lang: 'fr' | 'en'): ReadonlySet<string> {
   128→  return lang === 'fr' ? FR_WORDS : EN_WORDS
   129→}
   130→
   131→/** é→e, è→e, à→a, ç→c… et ligatures œ→oe, æ→ae. */
   132→function foldAccents(s: string): string {
   133→  return s
   134→    .replace(/œ/g, 'oe')
   135→    .replace(/Œ/g, 'OE')
   136→    .replace(/æ/g, 'ae')
   137→    .replace(/Æ/g, 'AE')
   138→    .normalize('NFD')
   139→    .replace(/[\u0300-\u036f]/g, '')
   140→}
   141→
   142→/** Mot candidat : majuscules, sans accents, lettres A-Z uniquement. */
   143→function normalizeWord(s: string): string {
   144→  return foldAccents(s).toUpperCase().replace(/[^A-Z]/g, '')
   145→}

   147→/** Points d'un mot : longueur ; 8 lettres ×2 ; 9 lettres (tout le tirage) ×5. */
   148→function wordPoints(len: number): number {
   149→  if (len === 9) return 45
   150→  if (len === 8) return 16
   151→  return len
   152→}
   153→
   154→/** Tirage de 9 lettres : 3 à 6 voyelles (donc 3 à 6 consonnes), pondéré, ordre mélangé. */
   155→function drawLetters(rand: () => number, weights: Record<string, number>): string[] {
   156→  const vowelTarget = 3 + Math.floor(rand() * 4)
   157→  const pick = (vowel: boolean): string => {
   158→    const pool = Object.keys(weights).filter((k) => VOWELS.has(k) === vowel && weights[k] > 0)
   159→    const total = pool.reduce((s, k) => s + weights[k], 0)
   160→    let r = rand() * total
   161→    for (const k of pool) {
   162→      r -= weights[k]
   163→      if (r <= 0) return k
   164→    }
   165→    return pool[pool.length - 1] ?? 'e'
   166→  }
   167→  const letters: string[] = []
   168→  for (let i = 0; i < vowelTarget; i++) letters.push(pick(true))
   169→  for (let i = 0; i < LETTER_COUNT - vowelTarget; i++) letters.push(pick(false))
   170→  return shuffle(letters, rand).map((l) => l.toUpperCase())
   171→}
