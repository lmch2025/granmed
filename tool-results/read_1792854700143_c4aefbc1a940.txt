   320→function emptyBoard(n: number): Cell[] {
   321→  return Array.from({ length: n }, () => ({ mine: false, adj: 0, revealed: false, flagged: false }))
   322→}
   323→
   324→function neighborsOf(idx: number, w: number, h: number): number[] {
   325→  const x = idx % w
   326→  const y = Math.floor(idx / w)
   327→  const out: number[] = []
   328→  for (let dy = -1; dy <= 1; dy++) {
   329→    for (let dx = -1; dx <= 1; dx++) {
   330→      if (dx === 0 && dy === 0) continue
   331→      const nx = x + dx
   332→      const ny = y + dy
   333→      if (nx >= 0 && nx < w && ny >= 0 && ny < h) out.push(ny * w + nx)
   334→    }
   335→  }
   336→  return out
   337→}

   339→/** Place les mines en excluant la case sûre et ses voisines, puis calcule les compteurs. */
   340→function placeMines(cells: Cell[], w: number, h: number, mines: number, safeIdx: number, rand: () => number): void {
   341→  const forbidden = new Set<number>([safeIdx, ...neighborsOf(safeIdx, w, h)])
   342→  const candidates: number[] = []
   343→  for (let i = 0; i < cells.length; i++) {
   344→    if (!forbidden.has(i)) candidates.push(i)
   345→  }
   346→  const count = Math.min(mines, candidates.length)
   347→  for (let i = 0; i < count; i++) {
   348→    const j = i + Math.floor(rand() * (candidates.length - i))
   349→    const tmp = candidates[i]
   350→    candidates[i] = candidates[j]
   351→    candidates[j] = tmp
   352→    cells[candidates[i]].mine = true
   353→  }
   354→  for (let i = 0; i < cells.length; i++) {
   355→    if (cells[i].mine) continue
   356→    let n = 0
   357→    for (const k of neighborsOf(i, w, h)) {
   358→      if (cells[k].mine) n++
   359→    }
   360→    cells[i].adj = n
   361→  }
   362→}

   364→/** Révélation en cascade (pile) ; retourne l’index de la mine touchée, ou -1 si rien n’explose. */
   365→function floodReveal(cells: Cell[], w: number, h: number, start: number): number {
   366→  if (cells[start].revealed || cells[start].flagged) return -1
   367→  if (cells[start].mine) {
   368→    cells[start].revealed = true
   369→    return start
   370→  }
   371→  const stack = [start]
   372→  while (stack.length > 0) {
   373→    const i = stack.pop() as number
   374→    if (cells[i].revealed || cells[i].flagged) continue
   375→    cells[i].revealed = true
   376→    if (cells[i].adj === 0) {
   377→      for (const n of neighborsOf(i, w, h)) {
   378→        if (!cells[n].revealed && !cells[n].mine) stack.push(n)
   379→      }
   380→    }
   381→  }
   382→  return -1
   383→}

   385→function countRevealed(cells: Cell[]): number {
   386→  let n = 0
   387→  for (const c of cells) if (c.revealed) n++
   388→  return n
   389→}

   540→  /** Révèle une case cachée — ou joue le chording si la case est déjà révélée et chiffrée. */
   541→  function revealAt(idx: number) {
   542→    if (phaseRef.current !== 'playing') return
   543→    const cur = boardRef.current
   544→    if (cur.length === 0) return
   545→    if (cur[idx].revealed) {
   546→      chordAt(idx)
   547→      return
   548→    }
   549→    if (cur[idx].flagged) return
   550→    const c = DIFFS[diffRef.current]
   551→    const next = cur.map((x) => ({ ...x }))
   552→    if (!minesPlacedRef.current) {
   553→      placeMines(next, c.w, c.h, c.mines, idx, Math.random)
   554→      minesPlacedRef.current = true
   555→      startRef.current = Date.now()
   556→    }
   557→    const hit = floodReveal(next, c.w, c.h, idx)
   558→    const opened = countRevealed(next) - countRevealed(cur)
   559→    setBoard(next)
   560→    if (hit >= 0) {
   561→      finishLose(hit, next)
   562→      return
   563→    }
   564→    if (opened > 0) sfx.play(opened >= 4 ? 'sweep' : 'pop')
   565→    if (countRevealed(next) === next.length - c.mines) finishWin(next)
   566→  }

   568→  /** Chording : clic sur un chiffre dont les drapeaux voisins sont exactement posés. */
   569→  function chordAt(idx: number) {
   570→    if (phaseRef.current !== 'playing') return
   571→    const cur = boardRef.current
   572→    if (cur.length === 0) return
   573→    const cell = cur[idx]
   574→    if (!cell.revealed || cell.mine || cell.adj === 0) return
   575→    const c = DIFFS[diffRef.current]
   576→    const nbrs = neighborsOf(idx, c.w, c.h)
   577→    let flagged = 0
   578→    for (const n of nbrs) {
   579→      if (cur[n].flagged) flagged++
   580→    }
   581→    if (flagged !== cell.adj) return
   582→    const next = cur.map((x) => ({ ...x }))
   583→    let hit = -1
   584→    for (const n of nbrs) {
   585→      if (!next[n].revealed && !next[n].flagged) {
   586→        const r = floodReveal(next, c.w, c.h, n)
   587→        if (r >= 0) hit = r
   588→      }
   589→    }
   590→    const opened = countRevealed(next) - countRevealed(cur)
   591→    setBoard(next)
   592→    if (hit >= 0) {
   593→      finishLose(hit, next)
   594→      return
   595→    }
   596→    if (opened > 0) sfx.play(opened >= 4 ? 'sweep' : 'pop')
   597→    if (countRevealed(next) === next.length - c.mines) finishWin(next)
   598→  }

   600→  function toggleFlagAt(idx: number) {
   601→    if (phaseRef.current !== 'playing') return
   602→    const cur = boardRef.current
   603→    if (cur.length === 0 || cur[idx].revealed) return

   606→    setBoard(next)
   607→    sfx.play(next[idx].flagged ? 'drop' : 'flip')
   608→    buzz()
   609→  }

   611→  function finishWin(cells: Cell[]) {
   612→    const d = diffRef.current
   613→    const c = DIFFS[d]
   614→    const ms = startRef.current > 0 ? Date.now() - startRef.current : 0
   615→    const seconds = Math.max(1, Math.round(ms / 1000))
   616→    const next = cells.map((x) => ({ ...x }))
   617→    for (const cell of next) {
   618→      if (cell.mine) cell.flagged = true
   619→    }
   620→    setBoard(next)
   621→    setPhase('won')
   622→    setFinalTime(seconds)
   623→    setConfettiKey((k) => k + 1)
   624→    sfx.play('win')
   625→    setTipIdx((i) => (i + 1) % HEALTH_TIPS.length)
   626→    setTipOpen(true)
   627→    // Rejouer un défi déjà réussi ne recompte pas la récompense du jour.
   628→    if (d === 'daily' && dailyDone) return
   629→    addWin(gameId)
   630→    addXp(c.xp)
   631→    setFinalXp(c.xp)
   632→    setNewRecord(setBest(gameId, c.metric, seconds, true))
   633→    if (d === 'medium') unlockBadge('mine-medium')
   634→    if (d === 'expert') unlockBadge('mine-expert')
   635→    if (d === 'daily') completeDaily(gameId)

   708→      case 'ArrowLeft':
   709→        target = idx % c.w > 0 ? idx - 1 : idx
   710→        break
   711→      case 'ArrowDown':
   712→        target = idx + c.w < c.w * c.h ? idx + c.w : idx
   713→        break
   714→      case 'ArrowUp':
   715→        target = idx - c.w >= 0 ? idx - c.w : idx
   716→        break
   717→      case 'Enter':
   718→      case ' ':
   719→        e.preventDefault()
   720→        revealAt(idx)
   721→        return
   722→      case 'f':
   723→      case 'F':
   724→        toggleFlagAt(idx)
   725→        return
   726→      default:
   727→        return
   728→    }
   729→    e.preventDefault()
   730→    if (target >= 0 && target !== idx) cellRefs.current[target]?.focus()
