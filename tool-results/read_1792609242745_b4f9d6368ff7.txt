   133→   133→function popcount9(v: number): number {
   134→   134→  let n = 0
   135→   135→  while (v) {
   136→   136→    v &= v - 1
   137→   137→    n++
   138→   138→  }
   139→   139→  return n
   140→   140→}

   142→   142→/** Étape 1 : remplit une grille complète valide par backtracking randomisé. */
   143→   143→function generateSolution(rand: () => number): number[] {
   144→   144→  const grid = new Array<number>(81).fill(0)
   145→   145→  const rows = new Array<number>(9).fill(0)
   146→   146→  const cols = new Array<number>(9).fill(0)
   147→   147→  const boxes = new Array<number>(9).fill(0)
   148→   148→  const digits = [1, 2, 3, 4, 5, 6, 7, 8, 9]
   149→   149→
   150→   150→  function fill(pos: number): boolean {
   151→   151→    if (pos === 81) return true
   152→   152→    const r = (pos / 9) | 0
   153→   153→    const c = pos % 9
   154→   154→    const b = ((r / 3) | 0) * 3 + ((c / 3) | 0)
   155→   155→    for (const d of shuffle(digits, rand)) {
   156→   156→      const bit = 1 << d
   157→   157→      if ((rows[r] | cols[c] | boxes[b]) & bit) continue
   158→   158→      grid[pos] = d
   159→   159→      rows[r] |= bit
   160→   160→      cols[c] |= bit
   161→   161→      boxes[b] |= bit
   162→   162→      if (fill(pos + 1)) return true
   163→   163→      grid[pos] = 0
   164→   164→      rows[r] ^= bit
   165→   165→      cols[c] ^= bit
   166→   166→      boxes[b] ^= bit
   167→   167→    }
   168→   168→    return false
   169→   169→  }
   170→   170→  fill(0)
   171→   171→  return grid
   172→   172→}

   174→   174→/** Étape 2 : compte les solutions d'une grille (arrêt dès `limit` atteint). Ne modifie pas l'entrée. */
   175→   175→function countSolutions(input: number[], limit: number): number {
   176→   176→  const g = input.slice()
   177→   177→  const rows = new Array<number>(9).fill(0)
   178→   178→  const cols = new Array<number>(9).fill(0)
   179→   179→  const boxes = new Array<number>(9).fill(0)
   180→   180→  for (let i = 0; i < 81; i++) {
   181→   181→    if (g[i] === 0) continue
   182→   182→    const r = (i / 9) | 0
   183→   183→    const c = i % 9
   184→   184→    const b = ((r / 3) | 0) * 3 + ((c / 3) | 0)
   185→   185→    const bit = 1 << g[i]
   186→   186→    rows[r] |= bit
   187→   187→    cols[c] |= bit
   188→   188→    boxes[b] |= bit
   189→   189→  }
   190→   190→  let count = 0
   191→   191→
   192→   192→  function backtrack(): boolean {
   193→   193→    // MRV : chercher la case vide avec le moins de candidats
   194→   194→    let best = -1
   195→   195→    let bestMask = 0
   196→   196→    let bestN = 10
   197→   197→    for (let i = 0; i < 81; i++) {
   198→   198→      if (g[i] !== 0) continue
   199→   199→      const r = (i / 9) | 0
   200→   200→      const c = i % 9
   201→   201→      const b = ((r / 3) | 0) * 3 + ((c / 3) | 0)
   202→   202→      const mask = ~(rows[r] | cols[c] | boxes[b]) & 0x3fe
   203→   203→      const n = popcount9(mask)
   204→   204→      if (n === 0) return false // impasse
   205→   205→      if (n < bestN) {
   206→   206→        bestN = n
   207→   207→        best = i
   208→   208→        bestMask = mask
   209→   209→        if (n === 1) break
   210→   210→      }
   211→   211→    }
   212→   212→    if (best === -1) {
   213→   213→      count++
   214→   214→      return count >= limit
   215→   215→    }
   216→   216→    const r = (best / 9) | 0
   217→   217→    const c = best % 9
   218→   218→    const b = ((r / 3) | 0) * 3 + ((c / 3) | 0)
   219→   219→    for (let d = 1; d <= 9; d++) {
   220→   220→      const bit = 1 << d
   221→   221→      if (!(bestMask & bit)) continue
   222→   222→      g[best] = d
   223→   223→      rows[r] |= bit
   224→   224→      cols[c] |= bit
   225→   225→      boxes[b] |= bit
   226→   226→      const stop = backtrack()
   227→   227→      g[best] = 0
   228→   228→      rows[r] ^= bit
   229→   229→      cols[c] ^= bit
   230→   230→      boxes[b] ^= bit
   231→   231→      if (stop) return true
   232→   232→    }
   233→   233→    return false
   234→   234→  }
   235→   235→
   236→   236→  backtrack()
   237→   237→  return count
   238→   238→}

   240→   240→/** Étape 3 : retire des chiffres un à un en vérifiant l'unicité de la solution À CHAQUE tentative. */
   241→   241→function makePuzzle(targetGivens: number, rand: () => number): { givens: number[]; solution: number[] } {
   242→   242→  const solution = generateSolution(rand)
   243→   243→  const givens = solution.slice()
   244→   244→  const positions = shuffle(Array.from({ length: 81 }, (_, i) => i), rand)
   245→   245→  let toRemove = 81 - targetGivens
   246→   246→  for (const pos of positions) {
   247→   247→    if (toRemove <= 0) break
   248→   248→    const backup = givens[pos]
   249→   249→    if (backup === 0) continue
   250→   250→    givens[pos] = 0
   251→   251→    if (countSolutions(givens, 2) === 1) {
   252→   252→      toRemove--
   253→   253→    } else {
   254→   254→      givens[pos] = backup // l'unicité serait perdue : on remet le chiffre
   255→   255→    }
   256→   256→  }
   257→   257→  return { givens, solution }
   258→   258→}

   260→   260→/** Cases voisines (ligne + colonne + bloc) précalculées. */
   261→   261→const PEERS: number[][] = (() => {
   262→   262→  const peers: number[][] = []
   263→   263→  for (let i = 0; i < 81; i++) {
   264→   264→    const r = (i / 9) | 0
   265→   265→    const c = i % 9
   266→   266→    const set = new Set<number>()
   267→   267→    for (let k = 0; k < 9; k++) {
   268→   268→      set.add(r * 9 + k)
   269→   269→      set.add(k * 9 + c)
   270→   270→    }
   271→   271→    const br = ((r / 3) | 0) * 3
   272→   272→    const bc = ((c / 3) | 0) * 3
   273→   273→    for (let dr = 0; dr < 3; dr++) {
   274→   274→      for (let dc = 0; dc < 3; dc++) set.add((br + dr) * 9 + (bc + dc))
   275→   275→    }
   276→   276→    set.delete(i)
   277→   277→    peers.push([...set])
   278→   278→  }
   279→   279→  return peers
   280→   280→})()

   287→   287→function findConflicts(grid: number[]): Set<number> {
   288→   288→  const out = new Set<number>()
   289→   289→  for (let i = 0; i < 81; i++) {
   290→   290→    const v = grid[i]
   291→   291→    if (!v) continue
   292→   292→    for (const j of PEERS[i]) {
   293→   293→      if (grid[j] === v) {
   294→   294→        out.add(i)
   295→   295→        out.add(j)
   296→   296→      }
   297→   297→    }
   298→   298→  }
   299→   299→  return out
   300→   300→}

   563→   563→  function inputDigit(d: number) {
   564→   564→    if (phase !== 'playing' || !running || sel < 0) return
   565→   565→    if (givens[sel] !== 0 || hinted.includes(sel)) {
   566→   566→      sfx.play('wrong', { volume: 0.25 })

   570→   570→      if (grid[sel] !== 0) {
   571→   571→        sfx.play('click', { volume: 0.3 })
   572→   572→        return
   573→   573→      }
   574→   574→      const nn = [...notes]
   575→   575→      nn[sel] ^= 1 << d
   576→   576→      setNotes(nn)
   577→   577→      sfx.play('click', { volume: 0.4 })
   578→   578→      return
   579→   579→    }
   580→   580→    const ng = [...grid]
   581→   581→    ng[sel] = ng[sel] === d ? 0 : d
   582→   582→    const nn = [...notes]
   583→   583→    nn[sel] = 0
   584→   584→    setGrid(ng)
   585→   585→    setNotes(nn)
   586→   586→    if (ng[sel] === 0) sfx.play('click', { volume: 0.4 })
   587→   587→    else sfx.play('pop', { volume: 0.6 })
   588→   588→    if (ng[sel] !== 0 && autoCheck && conflictsWith(ng, sel, ng[sel])) sfx.play('wrong', { volume: 0.3 })
   589→   589→    checkWin(ng)
   590→   590→  }

   605→   605→  function giveHint() {
   606→   606→    if (phase !== 'playing' || !running || hintsUsed >= 3) return
   607→   607→    const fixable = (i: number) => givens[i] === 0 && !hinted.includes(i) && grid[i] !== solution[i]
   608→   608→    let target = sel
   609→   609→    if (target < 0 || !fixable(target)) {
   610→   610→      const candidates: number[] = []
   611→   611→      for (let i = 0; i < 81; i++) if (fixable(i)) candidates.push(i)
   612→   612→      if (candidates.length === 0) return
   613→   613→      target = candidates[Math.floor(Math.random() * candidates.length)]
   614→   614→    }
   615→   615→    const ng = [...grid]
   616→   616→    ng[target] = solution[target]
   617→   617→    const nn = [...notes]
   618→   618→    nn[target] = 0
   619→   619→    setGrid(ng)
   620→   620→    setNotes(nn)
   621→   621→    setHinted([...hinted, target])
   622→   622→    setHintsUsed((h) => h + 1)
   623→   623→    setSel(target)
   624→   624→    sfx.play('correct')
   625→   625→    checkWin(ng, hintsUsed + 1)

   666→   666→  function handleKey(e: KeyboardEvent) {
   667→   667→    if (phase !== 'playing' || !running) return
   668→   668→    if (e.metaKey || e.ctrlKey || e.altKey) return

   688→   688→      ArrowLeft: [0, -1],
   689→   689→      ArrowRight: [0, 1],

   700→   700→      setSel(r * 9 + c)
