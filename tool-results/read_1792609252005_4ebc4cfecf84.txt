   119→type Phase = 'picker' | 'playing' | 'solved'

   121→interface SaveData {
   122→  letters: Record<string, string>
   123→  puzzleId: string
   124→  hintsUsed: number
   125→  solvedIds: string[]
   126→}
   127→
   128→const KEYS = ['AZERTYUIOP', 'QSDFGHJKLM', 'WXCVBN']

   130→function puzzlesFor(lang: 'fr' | 'en'): CrosswordPuzzle[] {
   131→  return lang === 'fr' ? FR_PUZZLES : EN_PUZZLES

   134→export default function MiniCrosswordGame() {
   135→  const t = useT()
   136→  const lang = useLang()
   137→  const dailyDone = useIsDailyDone(gameId)
   138→  const addPlay = useProfileStore((s) => s.addPlay)
   139→  const addWin = useProfileStore((s) => s.addWin)
   140→  const addXp = useProfileStore((s) => s.addXp)
   141→  const setBest = useProfileStore((s) => s.setBest)
   142→  const unlockBadge = useProfileStore((s) => s.unlockBadge)
   143→  const completeDaily = useProfileStore((s) => s.completeDaily)
   144→
   145→  const [phase, setPhase] = useState<Phase>('picker')
   146→  const [puzzle, setPuzzle] = useState<CrosswordPuzzle | null>(null)
   147→  const [isDaily, setIsDaily] = useState(false)
   148→  const [letters, setLetters] = useState<Record<string, string>>({})
   149→  const [active, setActive] = useState<{ r: number; c: number } | null>(null)
   150→  const [dir, setDir] = useState<'across' | 'down'>('across')
   151→  const [hintsUsed, setHintsUsed] = useState(0)
   152→  const [wrongCells, setWrongCells] = useState<Set<string>>(new Set())
   153→  const [solvedIds, setSolvedIds] = useState<string[]>([])
   154→  const [justSolved, setJustSolved] = useState(false)
   155→  const [confettiKey, setConfettiKey] = useState(0)
   156→  const [restored, setRestored] = useState(false)
   157→  const boardRef = useRef<HTMLDivElement>(null)
   158→
   159→  const dailyPuzzle = useMemo(() => {
   160→    const list = puzzlesFor(lang)
   161→    const rand = mulberry32(seedFromString('mini-crossword:' + todayKey()))
   162→    return list[Math.floor(rand() * list.length)]
   163→  }, [lang])
   164→
   165→  const grid = useMemo(() => (puzzle ? buildGrid(puzzle) : []), [puzzle])
   166→  const nums = useMemo(() => (puzzle ? numberCells(puzzle) : new Map<string, number>()), [puzzle])
   167→
   168→  /* Restauration de la sauvegarde (callback, jamais dans le corps d'un effet) */
   169→  useEffect(() => {
   170→    let alive = true
   171→    Promise.resolve().then(() => {
   172→      if (!alive) return
   173→      const save = loadSave<SaveData>(gameId)
   174→      if (save) {
   175→        setSolvedIds(save.solvedIds ?? [])
   176→        if (save.puzzleId && save.letters && Object.keys(save.letters).length > 0) {
   177→          const p = [...FR_PUZZLES, ...EN_PUZZLES].find((x) => x.id === save.puzzleId)
   178→          if (p) {
   179→            setPuzzle(p)
   180→            setLetters(save.letters)
   181→            setHintsUsed(save.hintsUsed ?? 0)
   182→            setPhase('playing')
   183→            setRestored(true)
   184→          }
   185→        }
   186→      }
   187→    })
   188→    return () => {
   189→      alive = false
   190→    }
   191→  }, [])
   192→
   193→  const persist = useCallback(
   194→    (l: Record<string, string>, p: CrosswordPuzzle | null, h: number, solved: string[]) => {
   195→      if (!p) return
   196→      saveSave<SaveData>(gameId, { letters: l, puzzleId: p.id, hintsUsed: h, solvedIds: solved })
   197→    },
   198→    [],
   199→  )
   200→
   201→  const startPuzzle = useCallback(
   202→    (p: CrosswordPuzzle, daily: boolean) => {
   203→      setPuzzle(p)
   204→      setIsDaily(daily)
   205→      setLetters({})
   206→      setHintsUsed(0)
   207→      setWrongCells(new Set())
   208→      setActive(p.words[0] ? { r: p.words[0].row, c: p.words[0].col } : null)
   209→      setDir(p.words[0]?.dir ?? 'across')
   210→      setPhase('playing')
   211→      setJustSolved(false)
   212→      setRestored(false)
   213→      addPlay(gameId)
   214→      persist({}, p, 0, solvedIds)
   215→    },
   216→    [addPlay, persist, solvedIds],
   217→  )
   218→
   219→  /* ---- Mots ---- */
   220→  const wordAtCell = useCallback(
   221→    (r: number, c: number, d: 'across' | 'down'): CrosswordWord | undefined =>
   222→      puzzle?.words.find(
   223→        (w) =>
   224→          w.dir === d &&
   225→          r >= w.row && r <= w.row + (w.answer.length - 1) * (d === 'down' ? 1 : 0) &&
   226→          c >= w.col && c <= w.col + (w.answer.length - 1) * (d === 'across' ? 1 : 0) &&
   227→          (d === 'down' ? c === w.col : r === w.row),
   228→      ),
   229→    [puzzle],
   230→  )
   231→
   232→  const finishSolve = useCallback(
   233→    (l: Record<string, string>) => {
   234→      if (!puzzle) return
   235→      setPhase('solved')
   236→      setJustSolved(true)
   237→      setConfettiKey((k) => k + 1)
   238→      sfx.play('win')
   239→      const newSolved = solvedIds.includes(puzzle.id) ? solvedIds : [...solvedIds, puzzle.id]
   240→      setSolvedIds(newSolved)
   241→      persist({}, puzzle, hintsUsed, newSolved)
   242→      addWin(gameId)
   243→      addXp(30)
   244→      const wasSolved = solvedIds.includes(puzzle.id)
   245→      if (!wasSolved) {
   246→        const total = newSolved.length
   247→        setBest(gameId, 'solved', total, false)
   248→        unlockBadge('crossword-first')
   249→        if (total >= 3) unlockBadge('crossword-3')
   250→      }
   251→      if (isDaily) completeDaily(gameId)
   252→    },
   253→    [puzzle, solvedIds, persist, addWin, addXp, setBest, unlockBadge, completeDaily, isDaily, hintsUsed],
   254→  )
   255→
   256→  const typeLetter = useCallback(
   257→    (ch: string) => {
   258→      if (!puzzle || !active || phase !== 'playing') return
   259→      const key = `${active.r},${active.c}`
   260→      if (grid[active.r][active.c] === null) return
   261→      sfx.play('click', { volume: 0.4 })
   262→      const next = { ...letters, [key]: ch }
   263→      setLetters(next)
   264→      setWrongCells((prev) => {
   265→        const s = new Set(prev)
   266→        s.delete(key)
   267→        return s
   268→      })
   269→      // avancer dans le mot
   270→      const w = wordAtCell(active.r, active.c, dir)
   271→      if (w) {
   272→        const idx = dir === 'across' ? active.c - w.col : active.r - w.row
   273→        if (idx < w.answer.length - 1) {
   274→          setActive(
   275→            dir === 'across' ? { r: active.r, c: active.c + 1 } : { r: active.r + 1, c: active.c },
   276→          )
   277→        }
   278→      }
   279→      // détection de résolution (dans le handler, jamais dans un effet)
   280→      const allFilled = grid.every((row, r) => row.every((cell, c) => cell === null || next[`${r},${c}`]))
   281→      if (allFilled) {
   282→        const allCorrect = grid.every((row, r) =>
   283→          row.every((cell, c) => cell === null || next[`${r},${c}`] === cell),
   284→        )
   285→        if (allCorrect) finishSolve(next)
   286→        else sfx.play('wrong', { volume: 0.3 })
   287→      }
   288→      persist(next, puzzle, hintsUsed, solvedIds)
   289→    },
   290→    [puzzle, active, phase, grid, letters, dir, wordAtCell, finishSolve, persist, hintsUsed, solvedIds],
   291→  )
   292→
   293→  const eraseLetter = useCallback(() => {
   294→    if (!puzzle || !active || phase !== 'playing') return
   295→    const key = `${active.r},${active.c}`
   296→    const next = { ...letters }
   297→    if (next[key]) {
   298→      delete next[key]
   299→      setLetters(next)
   300→      sfx.play('click', { volume: 0.3 })
   301→      persist(next, puzzle, hintsUsed, solvedIds)
   302→    } else {
   303→      // reculer
   304→      const w = wordAtCell(active.r, active.c, dir)
   305→      if (w) {
   306→        const idx = dir === 'across' ? active.c - w.col : active.r - w.row
   307→        if (idx > 0) {
   308→          const prevPos =
   309→            dir === 'across' ? { r: active.r, c: active.c - 1 } : { r: active.r - 1, c: active.c }
   310→          setActive(prevPos)
   311→          const pk = `${prevPos.r},${prevPos.c}`
   312→          const n2 = { ...next }
   313→          delete n2[pk]
   314→          setLetters(n2)
   315→          persist(n2, puzzle, hintsUsed, solvedIds)
   316→        }
   317→      }
   318→    }
   319→  }, [puzzle, active, phase, letters, dir, wordAtCell, persist, hintsUsed, solvedIds])
   320→
   321→  const moveActive = useCallback(
   322→    (dr: number, dc: number) => {
   323→      if (!puzzle || !active) return
   324→      let r = active.r + dr
   325→      let c = active.c + dc
   326→      while (r >= 0 && r < puzzle.size && c >= 0 && c < puzzle.size) {
   327→        if (grid[r][c] !== null) {
   328→          setActive({ r, c })
   329→          return
   330→        }
   331→        r += dr
   332→        c += dc
   333→      }
   334→    },
   335→    [puzzle, active, grid],
   336→  )
   337→
   338→  const selectCell = useCallback(
   339→    (r: number, c: number) => {
   340→      if (!puzzle || grid[r][c] === null) return
   341→      sfx.play('click', { volume: 0.3 })
   342→      if (active && active.r === r && active.c === c) {
   343→        const hasA = Boolean(wordAtCell(r, c, 'across'))
   344→        const hasD = Boolean(wordAtCell(r, c, 'down'))
   345→        if (hasA && hasD) setDir((d) => (d === 'across' ? 'down' : 'across'))
   346→      } else {
   347→        setActive({ r, c })
   348→        const hasA = Boolean(wordAtCell(r, c, 'across'))
   349→        setDir(hasA ? 'across' : 'down')
   350→      }
   351→    },
   352→    [puzzle, grid, active, wordAtCell],
   353→  )
   354→
   355→  const checkGrid = useCallback(() => {
   356→    if (!puzzle) return
   357→    sfx.play('click')
   358→    const wrong = new Set<string>()
   359→    let filled = 0
   360→    let total = 0
   361→    grid.forEach((row, r) =>
   362→      row.forEach((cell, c) => {
   363→        if (cell === null) return
   364→        total++
   365→        const l = letters[`${r},${c}`]
   366→        if (l) filled++
   367→        if (l && l !== cell) wrong.add(`${r},${c}`)
   368→      }),
   369→    )
   370→    setWrongCells(wrong)
   371→    if (wrong.size === 0 && filled === total) {
   372→      finishSolve(letters)
   373→    } else if (wrong.size > 0) {
   374→      sfx.play('wrong', { volume: 0.4 })
   375→    }
   376→  }, [puzzle, grid, letters, finishSolve])
   377→
   378→  const useHint = useCallback(() => {
   379→    if (!puzzle || hintsUsed >= 3 || !active || phase !== 'playing') return
   380→    // révèle une lettre vide de la grille (case active en priorité, sinon la première vide)
   381→    let target: { r: number; c: number } | null = null
   382→    if (!letters[`${active.r},${active.c}`] && grid[active.r][active.c] !== null) {
   383→      target = active
   384→    } else {
   385→      outer: for (let r = 0; r < puzzle.size; r++) {
   386→        for (let c = 0; c < puzzle.size; c++) {
   387→          if (grid[r][c] !== null && !letters[`${r},${c}`]) {
   388→            target = { r, c }
   389→            break outer
   390→          }
   391→        }
   392→      }
   393→    }
   394→    if (!target) return
   395→    sfx.play('reveal')
   396→    const next = { ...letters, [`${target.r},${target.c}`]: grid[target.r][target.c]! }
   397→    setLetters(next)
   398→    const h = hintsUsed + 1
   399→    setHintsUsed(h)
   400→    persist(next, puzzle, h, solvedIds)
   401→    const allFilled = grid.every((row, r) => row.every((cell, c) => cell === null || next[`${r},${c}`]))
   402→    const allCorrect = grid.every((row, r) => row.every((cell, c) => cell === null || next[`${r},${c}`] === cell))
   403→    if (allFilled && allCorrect) finishSolve(next)
   404→  }, [puzzle, hintsUsed, active, phase, letters, grid, persist, solvedIds, finishSolve])
   405→
   406→  /* Clavier physique via le plateau focusable */
   407→  const onKey = useCallback(
   408→    (e: React.KeyboardEvent) => {
   409→      if (phase !== 'playing') return
   410→      const k = e.key
   411→      if (/^[a-zA-Z]$/.test(k)) {
   412→        e.preventDefault()
   413→        typeLetter(k.toUpperCase())
   414→      } else if (k === 'Backspace') {
   415→        e.preventDefault()
   416→        eraseLetter()
   417→      } else if (k === 'ArrowLeft') moveActive(0, -1)
   418→      else if (k === 'ArrowRight') moveActive(0, 1)
   419→      else if (k === 'ArrowUp') moveActive(-1, 0)
   420→      else if (k === 'ArrowDown') moveActive(1, 0)
   421→      else if (k === ' ') {
   422→        e.preventDefault()
   423→        setDir((d) => (d === 'across' ? 'down' : 'across'))
   424→      }
   425→    },
   426→    [phase, typeLetter, eraseLetter, moveActive],
   427→  )

   430→    () => ({

   432→        choose: 'Choisissez une grille',
   433→        daily: 'Grille du jour',
   434→        across: 'Horizontalement',
   435→        down: 'Verticalement',
   436→        check: 'Vérifier',
   437→        hint: 'Indice',

   440→        restart: 'Recommencer',

   443→        resume: 'Reprendre la grille en cours',
   444→        newOne: 'Nouvelle grille',
   445→        restored: 'Reprise de votre grille sauvegardée.',
   446→        dailyDone: 'Défi du jour déjà réussi !',
   447→        fillAll: 'Remplissez toutes les cases pour vérifier.',
   448→      },
   449→      en: {
   450→        choose: 'Choose a grid',
   451→        daily: 'Grid of the day',
   452→        across: 'Across',
   453→        down: 'Down',
   454→        check: 'Check',
   455→        hint: 'Hint',
   456→        erase: 'Erase',
   457→        solvedTag: 'Solved',
   458→        restart: 'Restart',
   459→        back: 'Other grids',
   460→        solvedMsg: 'Grid solved! Well done!',
