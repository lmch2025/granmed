   144→/** é→e, è→e, à→a, ç→c… et ligatures œ→oe, æ→ae (décomposition NFD + retrait des diacritiques). */
   145→function foldAccents(s: string): string {
   146→  return s
   147→    .replace(/œ/g, 'oe')
   148→    .replace(/Œ/g, 'OE')
   149→    .replace(/æ/g, 'ae')
   150→    .replace(/Æ/g, 'AE')
   151→    .normalize('NFD')
   152→    .replace(/[\u0300-\u036f]/g, '')
   153→}
   154→
   155→/** Valeur du champ : majuscules sans accents, lettres + espace/apostrophe/tiret conservés. */
   156→function normalizeInput(v: string): string {
   157→  return foldAccents(v)
   158→    .replace(/’/g, "'")
   159→    .toUpperCase()
   160→    .replace(/[^A-Z' -]/g, '')
   161→}
   162→
   163→type Slot = { kind: 'sep'; ch: string } | { kind: 'letter'; ch: string; li: number }
   164→
   165→interface ParsedExpression {
   166→  slots: Slot[]
   167→  letters: string
   168→}
   169→
   170→/** Découpe l'expression : une case par lettre (index li), séparateurs toujours visibles. */
   171→function parseExpression(expr: string): ParsedExpression {
   172→  const slots: Slot[] = []
   173→  let letters = ''
   174→  const folded = foldAccents(expr).replace(/’/g, "'")
   175→  for (const ch of folded) {
   176→    if (/^[a-z]$/i.test(ch)) {
   177→      const up = ch.toUpperCase()
   178→      slots.push({ kind: 'letter', ch: up, li: letters.length })
   179→      letters += up
   180→    } else if (ch === "'") {
   181→      slots.push({ kind: 'sep', ch: "'" })
   182→    } else {
   183→      slots.push({ kind: 'sep', ch })
   184→    }
   185→  }
   186→  return { slots, letters }
   187→}

   189→function pickDaily(lang: 'fr' | 'en', count: number): IdiomEntry[] {
   190→  const pool = lang === 'fr' ? FR_IDIOMS : EN_IDIOMS
   191→  const rand = mulberry32(seedFromString(`${gameId}:${todayKey()}`))
   192→  const idxs = new Set<number>()
   193→  while (idxs.size < count) idxs.add(Math.floor(rand() * pool.length))
   194→  return [...idxs].map((i) => pool[i])
   195→}
   196→
   197→function pickFree(lang: 'fr' | 'en', count: number): IdiomEntry[] {
   198→  const pool = lang === 'fr' ? FR_IDIOMS : EN_IDIOMS
   199→  return shuffle(
   200→    pool.map((_, i) => i),
   201→    Math.random,
   202→  )
   203→    .slice(0, count)
   204→    .map((i) => pool[i])
   205→}
