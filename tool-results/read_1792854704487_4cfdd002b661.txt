   164→function puzzlesFor(lang: Lang): CrosswordPuzzle[] {
   165→  return lang === 'fr' ? FR_PUZZLES : EN_PUZZLES
   166→}
   167→
   168→function wordKey(w: CrosswordWord): string {
   169→  return `${w.dir}-${w.row}-${w.col}`
   170→}
   171→
   172→function cellKey(r: number, c: number): string {
   173→  return `${r},${c}`
   174→}
   175→
   176→/** Mot entièrement rempli avec les bonnes lettres ? */
   177→function isWordComplete(w: CrosswordWord, l: Record<string, string>): boolean {
   178→  for (let i = 0; i < w.answer.length; i++) {
   179→    const r = w.row + (w.dir === 'down' ? i : 0)
   180→    const c = w.col + (w.dir === 'across' ? i : 0)
   181→    if (l[cellKey(r, c)] !== w.answer[i]) return false
   182→  }
   183→  return true
   184→}
   185→
   186→/** A-Z après suppression des diacritiques (É→E, à→A…) ; null sinon. */
   187→function normalizeLetter(k: string): string | null {
   188→  // NFD puis retrait des marques diacritiques combinantes (U+0300-U+036F)
   189→  const base = k
   190→    .normalize('NFD')
   191→    .split('')
   192→    .filter((ch) => {
   193→      const code = ch.charCodeAt(0)
   194→      return code < 0x300 || code > 0x36f
   195→    })
   196→    .join('')
   197→  return /^[a-zA-Z]$/.test(base) ? base.toUpperCase() : null
   198→}

   705→  const giveHint = useCallback(() => {
   706→    if (!puzzle || phase !== 'playing' || hintsUsed >= HINTS_MAX) return

   769→    setHintsUsed(HINTS_MAX)

   773→      for (const k of prev) {
   774→        const [r, c] = k.split(',').map(Number)
   775→        if (next[k] !== grid[r][c]) s.add(k)
   776→      }
   777→      return s

   780→    const allCorrect = grid.every((row, r) => row.every((cell, c) => cell === null || next[cellKey(r, c)] === cell))
