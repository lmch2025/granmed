   268→interface DWStats {
   269→  gamesPlayed: number
   270→  wins: number
   271→  currentStreak: number
   272→  maxStreak: number
   273→  distribution: number[]
   274→  lastPlayedDate: string

   277→}

   279→interface DWSave {
   280→  v: number
   281→  stats: DWStats

   283→}
   284→
   285→function defaultStats(): DWStats {
   286→  return {
   287→    gamesPlayed: 0,
   288→    wins: 0,
   289→    currentStreak: 0,
   290→    maxStreak: 0,
   291→    distribution: [0, 0, 0, 0, 0, 0],
   292→    lastPlayedDate: '',
   293→    lastWonDate: null,
   294→    history: {},
   295→  }
   296→}
   297→
   298→function parseStats(raw: unknown): DWStats {
   299→  const d = defaultStats()
   300→  if (!raw || typeof raw !== 'object') return d
   301→  const s = raw as Partial<DWStats>
   302→  const num = (v: unknown): number => (typeof v === 'number' && v >= 0 ? v : 0)
   303→  const history: DWStats['history'] = {}
   304→  if (s.history && typeof s.history === 'object') {
   305→    for (const [k, v] of Object.entries(s.history)) {
   306→      if (v && typeof v === 'object') {
   307→        const e = v as { word?: unknown; tries?: unknown; won?: unknown }
   308→        if (typeof e.word === 'string' && typeof e.tries === 'number' && typeof e.won === 'boolean') {
   309→          history[k] = { word: e.word, tries: e.tries, won: e.won }
   310→        }
   311→      }
   312→    }
   313→  }
   314→  return {
   315→    gamesPlayed: num(s.gamesPlayed),
   316→    wins: num(s.wins),
   317→    currentStreak: num(s.currentStreak),
   318→    maxStreak: num(s.maxStreak),
   319→    distribution:
   320→      Array.isArray(s.distribution) && s.distribution.length === 6
   321→        ? s.distribution.map((n) => num(n))
   322→        : [0, 0, 0, 0, 0, 0],
   323→    lastPlayedDate: typeof s.lastPlayedDate === 'string' ? s.lastPlayedDate : '',
   324→    lastWonDate: typeof s.lastWonDate === 'string' ? s.lastWonDate : null,
   325→    history,
   326→  }
   327→}

   329→function fmtDate(dk: string, lang: Lang): string {
   330→  const [y, m, d] = dk.split('-').map((v) => Number.parseInt(v, 10))
   331→  if (!y || !m || !d) return dk
   332→  return new Date(y, m - 1, d).toLocaleDateString(lang === 'fr' ? 'fr-FR' : 'en-US', {
   333→    weekday: 'short',
   334→    day: 'numeric',
   335→    month: 'short',
   336→  })
   337→}

   345→  )
   346→}

   370→  const [shakeRow, setShakeRow] = useState(false)
   371→  const [message, setMessage] = useState<string | null>(null)
   372→  const [confettiKey, setConfettiKey] = useState(0)

   376→  const lockRef = useRef(false)
   377→  const genRef = useRef(0)
   378→  const playCountedRef = useRef(false)
   379→  const statsRef = useRef<DWStats>(defaultStats())
   380→  const dailyRef = useRef<DWDaily | null>(null)
   381→  const lastWordRef = useRef('')
   382→
   383→  const REVEAL_STEP = reduce ? 60 : 200
   384→
   385→  function applyStats(s: DWStats) {
   386→    statsRef.current = s
   387→    setStats(s)
   388→  }
   389→
   390→  function persist(s: DWStats, d: DWDaily | null) {
   391→    saveSave<DWSave>(gameId, { v: 1, stats: s, daily: d })
   392→  }
   393→
   394→  /** Timer protégé par une génération : les callbacks meurent si une nouvelle partie démarre. */
   395→  function schedule(fn: () => void, ms: number) {
   396→    const gen = genRef.current
   397→    window.setTimeout(() => {
   398→      if (genRef.current === gen) fn()
   399→    }, ms)
   400→  }
   401→
   402→  /* --- Restauration (callback asynchrone post-mount, conforme react-hooks/set-state-in-effect) --- */
   403→  useEffect(() => {
   404→    let alive = true
   405→    Promise.resolve().then(() => {
   406→      if (!alive) return
   407→      const today = todayKey()
   408→      const save = loadSave<DWSave>(gameId)
   409→      const s = parseStats(save?.stats)
   410→      const d = save?.daily
   411→      genRef.current++
   412→      lockRef.current = false
   413→      setAnswer(wordForDate(lang, today))
   414→      setCurrent('')
   415→      setMessage(null)
   416→      setShakeRow(false)
   417→      setRevealIdx(5)
   418→      setMode('daily')
   419→      applyStats(s)
   420→      if (d && d.date === today && d.lang === lang && Array.isArray(d.guesses)) {
   421→        dailyRef.current = d
   422→        playCountedRef.current = d.counted
   423→        const gs = d.guesses.filter((g) => typeof g === 'string' && g.length === 5)
   424→        setGuesses(gs)
   425→        const ans = wordForDate(lang, today)
   426→        setPhase(gs.includes(ans) ? 'won' : gs.length >= 6 ? 'lost' : 'playing')
   427→      } else {
   428→        dailyRef.current = null
   429→        playCountedRef.current = false
   430→        setGuesses([])
   431→        setPhase('playing')
   432→      }
   433→      setReady(true)
   434→    })
   435→    return () => {
   436→      alive = false
   437→    }
   438→  }, [lang])
   439→
   440→  /* --- Saisie --- */
   441→  function addLetter(ch: string) {
   442→    if (phase !== 'playing' || lockRef.current) return
   443→    if (current.length >= 5) return
   444→    sfx.play('click', { volume: 0.3 })
   445→    setCurrent(current + ch)
   446→  }
   447→
   448→  function removeLetter() {
   449→    if (phase !== 'playing' || lockRef.current) return
   450→    if (!current.length) return
   451→    setCurrent(current.slice(0, -1))
   452→  }
   453→
   454→  function flashInvalid(msg: string) {
   455→    setMessage(msg)
   456→    setShakeRow(true)
   457→    sfx.play('wrong', { volume: 0.35 })
   458→    schedule(() => setShakeRow(false), 500)
   459→    schedule(() => setMessage(null), 2200)
   460→  }
   461→
   462→  function submitGuess() {
   463→    if (phase !== 'playing' || lockRef.current) return
   464→    if (current.length !== 5) {
   465→      flashInvalid(L.incomplete)
   466→      return
   467→    }
   468→    if (!dictionary.has(current)) {
   469→      flashInvalid(L.unknown)
   470→      return
   471→    }
   472→    if (!playCountedRef.current) {
   473→      playCountedRef.current = true
   474→      addPlay(gameId)
   475→    }
   476→    const guess = current
   477→    const nextGuesses = [...guesses, guess]
   478→    const won = guess === answer
   479→    const lost = !won && nextGuesses.length >= 6
   480→    setGuesses(nextGuesses)
   481→    setCurrent('')
   482→    setMessage(null)
   483→    lockRef.current = true
   484→    setRevealIdx(0)
   485→    if (mode === 'daily') {
   486→      const d: DWDaily = { date: todayKey(), lang, guesses: nextGuesses, counted: playCountedRef.current }
   487→      dailyRef.current = d
   488→      persist(statsRef.current, d)
   489→    }
   490→    for (let i = 0; i < 5; i++) {
   491→      schedule(() => {
   492→        sfx.play('reveal', { volume: 0.45 })
   493→        setRevealIdx(i + 1)
   494→      }, REVEAL_STEP * (i + 1))
   495→    }
   496→    schedule(
   497→      () => {
   498→        lockRef.current = false
   499→        if (won || lost) finishGame(won, nextGuesses.length, nextGuesses)
   500→      },
   501→      REVEAL_STEP * 5 + 350,
   502→    )
   503→  }

   505→  function finishGame(won: boolean, tries: number, gs: string[]) {
   506→    setPhase(won ? 'won' : 'lost')
   507→    if (won) {
   508→      sfx.play('win')
   509→      setConfettiKey((k) => k + 1)
   510→    } else {
   511→      sfx.play('lose')
   512→    }
   513→    if (mode === 'daily') finishDailyStats(won, tries, gs)
   514→  }

   516→  function finishDailyStats(won: boolean, tries: number, gs: string[]) {
   517→    const today = todayKey()
   518→    const prev = statsRef.current
   519→    if (prev.history[today]) return // déjà comptabilisé aujourd'hui (robustesse)
   520→    const s: DWStats = {
   521→      ...prev,
   522→      distribution: [...prev.distribution],
   523→      history: { ...prev.history, [today]: { word: answer, tries, won } },
   524→    }
   525→    s.gamesPlayed += 1
   526→    s.lastPlayedDate = today
   527→    if (won) {
   528→      s.wins += 1
   529→      s.distribution[tries - 1] += 1
   530→      s.currentStreak = prev.lastWonDate === dateKeyOffset(-1) ? prev.currentStreak + 1 : 1
   531→      s.maxStreak = Math.max(s.maxStreak, s.currentStreak)
   532→      s.lastWonDate = today
   533→    } else {
   534→      s.currentStreak = 0
   535→    }
   536→    applyStats(s)
   537→    const d: DWDaily = { date: today, lang, guesses: gs, counted: playCountedRef.current }
   538→    dailyRef.current = d
   539→    persist(s, d)
   540→    if (won) {
   541→      addWin(gameId)
   542→      addXp(25 + (6 - tries) * 5)
   543→      setBest(gameId, 'wins', s.wins)
   544→      setBest(gameId, 'max-streak', s.maxStreak)
   545→      if (tries <= 3) unlockBadge('word-fast3')
   546→      if (s.currentStreak >= 5) unlockBadge('word-streak5')
   547→      completeDaily(gameId)
   548→    }
   549→  }
