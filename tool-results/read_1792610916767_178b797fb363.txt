   246→
   247→  const playing = game != null && won == null
   248→
   249→  function updateGame(g: PlayState | null) {

   253→
   254→  function getElapsed(): number {
   255→    return elapsedBaseRef.current + (runningSinceRef.current != null ? Date.now() - runningSinceRef.current : 0)
   256→  }
   257→
   258→  function toProgress(g: PlayState): Progress {
   259→    return {
   260→      imageId: g.imageId,
   261→      count: g.count,
   262→      locked: g.pieces.map((p) => p.locked),
   263→      pos: g.pieces.map((p) => ({ x: p.x, y: p.y })),
   264→      elapsed: 0,

   266→  }
   267→
   268→  /* --- Restauration au montage (callback asynchrone) --- */
   269→  useEffect(() => {

   272→      if (!alive) return

   276→        galleryRef.current = s.gallery

   286→        p.locked.length === p.count &&
   287→        p.pos.length === p.count &&

   321→          elapsedBaseRef.current += Date.now() - runningSinceRef.current

   325→        runningSinceRef.current = Date.now()

   330→      window.clearInterval(id)
   331→      document.removeEventListener('visibilitychange', onVis)
   332→      if (runningSinceRef.current != null) {
   333→        elapsedBaseRef.current += Date.now() - runningSinceRef.current
   334→        runningSinceRef.current = null
   335→      }
   336→    }

   338→
   339→  /* --- Redimensionnement : mesure + mise à l'échelle des pièces libres --- */
   340→  function applyAreaW(w: number) {
   341→    const prev = areaWRef.current
   342→    if (w <= 0 || w === prev) return
   343→    areaWRef.current = w
   344→    setAreaW(w)
   345→    const g = gameRef.current
   346→    if (g && prev > 0) {
   347→      const f = w / prev
   348→      updateGame({ ...g, pieces: g.pieces.map((p) => (p.locked ? p : { ...p, x: p.x * f, y: p.y * f })) })
   349→    }
   350→  }
   351→
   352→  useEffect(() => {
   353→    const onResize = () => {
   354→      const el = areaRef.current
   355→      if (el) applyAreaW(el.clientWidth)
   356→    }
   357→    window.addEventListener('resize', onResize)
   358→    return () => window.removeEventListener('resize', onResize)
   359→  }, [])

   361→  function provisionalWidth(): number {
   362→    return Math.min(Math.round(window.innerWidth * 0.92), 420)
   363→  }
   364→
   365→  function startGame(imageId: string, count: number) {
   366→    const { cols, rows } = dimsFor(count)
   367→    const w = provisionalWidth()
   368→    const pw = w / cols
   369→    const ph = w / rows
   370→    const rand = mulberry32(seedFromString(`jigsaw:${imageId}:${count}`))
   371→    const pad = 8
   372→    const trayTop = w + BOARD_TRAY_GAP + pad
   373→    const pieces: Piece[] = []
   374→    for (let r = 0; r < rows; r++) {
   375→      for (let c = 0; c < cols; c++) {
   376→        pieces.push({
   377→          id: r * cols + c,
   378→          row: r,
   379→          col: c,
   380→          locked: false,
   381→          x: pad + rand() * Math.max(1, w - pw - pad * 2),
   382→          y: trayTop + rand() * Math.max(1, TRAY_H - ph - pad * 2),
   383→        })
   384→      }
   385→    }
   386→    elapsedBaseRef.current = 0
   387→    runningSinceRef.current = Date.now()
   388→    setElapsedMs(0)
   389→    setWon(null)
   390→    setDraggingId(null)
   391→    setSnapFx(null)
   392→    setShuffleFx(false)
   393→    setRestored(null)
   394→    areaWRef.current = w
   395→    setAreaW(w)
   396→    updateGame({ imageId, count, pieces })
   397→    addPlay(gameId)
   398→    sfx.play('pop')
   399→    window.requestAnimationFrame(() => {
   400→      const el = areaRef.current
   401→      if (el) applyAreaW(el.clientWidth)
   402→    })
   403→  }
   404→
   405→  function resumeGame() {
   406→    if (!restored) return
   407→    const { cols, rows } = dimsFor(restored.count)
   408→    const w = provisionalWidth()
   409→    const pw = w / cols
   410→    const ph = w / rows
   411→    const pieces: Piece[] = restored.locked.map((lk, i) => {
   412→      const r = Math.floor(i / cols)
   413→      const c = i % cols
   414→      const pos = restored.pos[i] ?? { x: c * pw, y: w + TRAY_H / 2 }
   415→      const locked = !!lk
   416→      return { id: i, row: r, col: c, locked, x: locked ? c * pw : pos.x, y: locked ? r * ph : pos.y }
   417→    })
   418→    elapsedBaseRef.current = restored.elapsed || 0
   419→    runningSinceRef.current = Date.now()
   420→    setElapsedMs(restored.elapsed || 0)
   421→    setWon(null)
   422→    setDraggingId(null)
   423→    setSnapFx(null)
   424→    setShuffleFx(false)
   425→    setRestored(null)
   426→    areaWRef.current = w
   427→    setAreaW(w)
   428→    updateGame({ imageId: restored.imageId, count: restored.count, pieces })
   429→    addPlay(gameId)
   430→    sfx.play('pop')
   431→    window.requestAnimationFrame(() => {
   432→      const el = areaRef.current
   433→      if (el) applyAreaW(el.clientWidth)
   434→    })
   435→  }
   436→
   437→  function discardRestored() {
   438→    sfx.play('click')
   439→    setRestored(null)
   440→    saveSave(gameId, { gallery: galleryRef.current, progress: null })
   441→  }
   442→
   443→  function backToSelection() {
   444→    sfx.play('click', { volume: 0.5 })
   445→    dragRef.current = null
   446→    setDraggingId(null)
   447→    setSnapFx(null)
   448→    setShuffleFx(false)
   449→    setWon(null)
   450→    setShowPreview(false)
   451→    updateGame(null)
   452→  }

   454→  /* --- Victoire (appelée depuis le callback de dépôt, jamais depuis un effet) --- */
   455→  function finishWin(g: PlayState) {
   456→    const ms = getElapsed()
   457→    elapsedBaseRef.current = ms
   458→    runningSinceRef.current = null
   459→    setElapsedMs(ms)
   460→    const seconds = Math.max(1, Math.round(ms / 1000))
   461→    const isRecord = setBest(gameId, `best-time-${g.count}`, seconds, true)
   462→    const key = `${g.imageId}:${g.count}`
   463→    sfx.play('win')
   464→    setConfettiKey((k) => k + 1)
   465→    addWin(gameId)
   466→    addXp(g.count === 12 ? 30 : g.count === 24 ? 60 : 120)
   467→    if (g.count === 48) unlockBadge('jigsaw-48')
   468→    const prev = galleryRef.current[key]
   469→    const nextGallery = { ...galleryRef.current }
   470→    if (prev == null || seconds < prev) nextGallery[key] = seconds
   471→    galleryRef.current = nextGallery
   472→    setGallery(nextGallery)
   473→    saveSave(gameId, { gallery: nextGallery, progress: null })
   474→    setWon({ seconds, isRecord })
   475→  }

   477→  /* --- Glisser-déposer (Pointer Events + capture) --- */
   478→  function onPieceDown(e: React.PointerEvent<HTMLDivElement>, p: Piece) {
   479→    if (!game || won != null || p.locked || dragRef.current != null) return
   480→    e.preventDefault()
   481→    const el = e.currentTarget
   482→    try {
   483→      el.setPointerCapture(e.pointerId)
   484→    } catch {
   485→      /* ignoré */
   486→    }
   487→    dragRef.current = { id: p.id, pointerId: e.pointerId, startX: e.clientX, startY: e.clientY, origX: p.x, origY: p.y }
   488→    setDraggingId(p.id)
   489→    sfx.play('click', { volume: 0.2 })
   490→  }

   492→  function onPieceMove(e: React.PointerEvent<HTMLDivElement>) {
   493→    const d = dragRef.current
   494→    if (!d || d.pointerId !== e.pointerId) return
   495→    const el = e.currentTarget
   496→    el.style.left = `${d.origX + (e.clientX - d.startX)}px`
   497→    el.style.top = `${d.origY + (e.clientY - d.startY)}px`
   498→  }

   500→  function onPieceUp(e: React.PointerEvent<HTMLDivElement>) {
   501→    const d = dragRef.current
   502→    if (!d || d.pointerId !== e.pointerId) return
   503→    dragRef.current = null
   504→    setDraggingId(null)
   505→    const g = gameRef.current
   506→    const el = e.currentTarget
   507→    const piece = g?.pieces.find((x) => x.id === d.id)
   508→    if (!g || !piece || piece.locked) return
   509→    const { cols, rows } = dimsFor(g.count)
   510→    const w = areaWRef.current
   511→    const pw = w / cols
   512→    const ph = w / rows
   513→    const nx = d.origX + (e.clientX - d.startX)
   514→    const ny = d.origY + (e.clientY - d.startY)
   515→
   516→    // Aimantation : centre de la pièce à moins de 26 px du centre de sa case
   517→    const tx = piece.col * pw
   518→    const ty = piece.row * ph
   519→    const dist = Math.hypot(nx + pw / 2 - (tx + pw / 2), ny + ph / 2 - (ty + ph / 2))
   520→    if (dist <= 26) {
   521→      sfx.play('drop')
   522→      // Aimantation animée : la transition est posée directement sur le DOM avant la valeur cible,
   523→      // l'état est commité après la transition (150 ms) — React n'écrase jamais left/top pendant le geste.
   524→      el.style.transition = 'left 150ms ease-out, top 150ms ease-out'
   525→      el.style.left = `${tx}px`
   526→      el.style.top = `${ty}px`
   527→      window.setTimeout(() => {
   528→        el.style.transition = ''
   529→        const g2 = gameRef.current
   530→        if (!g2) return
   531→        const pieces = g2.pieces.map((x) => (x.id === piece.id ? { ...x, x: tx, y: ty, locked: true } : x))
   532→        updateGame({ ...g2, pieces })
   533→        setSnapFx(piece.id)
   534→        window.setTimeout(() => setSnapFx(null), 500)
   535→        if (pieces.every((x) => x.locked)) finishWin(g2)
   536→      }, 170)
   537→    } else {
   538→      const areaH = w + BOARD_TRAY_GAP + TRAY_H
   539→      const cx = clamp(nx, -pw * 0.25, w - pw * 0.75)
   540→      const cy = clamp(ny, -ph * 0.25, areaH - ph * 0.75)
   541→      const pieces = g.pieces.map((x) => (x.id === piece.id ? { ...x, x: cx, y: cy } : x))
   542→      updateGame({ ...g, pieces })
   543→    }
   544→  }

   546→  /* --- Mélanger : redisperse les pièces non verrouillées dans le bac --- */
   547→  function reshuffle() {
   548→    const g = gameRef.current
   549→    if (!g || won != null) return
   550→    sfx.play('swish')
   551→    const { cols, rows } = dimsFor(g.count)
   552→    const w = areaWRef.current
   553→    const pw = w / cols
   554→    const ph = w / rows
   555→    const pad = 8

   557→    if (g.pieces.every((p) => p.locked)) return
   558→    const pieces = g.pieces.map((p) =>
   559→      p.locked
   560→        ? p
   561→        : {
   562→            ...p,
   563→            x: pad + Math.random() * Math.max(1, w - pw - pad * 2),
   564→            y: trayTop + Math.random() * Math.max(1, TRAY_H - ph - pad * 2),
   565→          },
   566→    )
   567→    setShuffleFx(true)
   568→    updateGame({ ...g, pieces })
   569→    window.setTimeout(() => setShuffleFx(false), 350)
   570→  }

   589→          <span className="text-2xl" aria-hidden="true">🧩</span>

   591→            <span className="block font-bold text-amber-700 dark:text-amber-300">{L.savedGame}</span>
   592→            <span className="block text-xs text-muted-foreground">
   593→              {IMAGES.find((im) => im.id === restored.imageId)?.title[lang]} · {restored.count} {L.pieces} ·{' '}
   594→              {restored.locked.filter(Boolean).length}/{restored.count} {L.placed}

   840→              <div className="mt-4 flex flex-wrap justify-center gap-2.5">
   841→                <Button size="lg" className="rounded-full" onClick={() => startGame(game.imageId, game.count)}>
   842→                  <RotateCcw className="h-4 w-4" aria-hidden="true" /> {t('common.playAgain')}
   843→                </Button>
   844→                <Button size="lg" variant="secondary" className="rounded-full" onClick={backToSelection}>
   845→                  <Puzzle className="h-4 w-4" aria-hidden="true" /> {L.otherImages}
   846→                </Button>
   847→              </div>
   848→            </div>
   849→          ) : null}
   850→        </div>
