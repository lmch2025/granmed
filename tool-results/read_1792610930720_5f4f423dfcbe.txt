   179→const STORY_START: Record<string, string> = { garden: 'g-start', letter: 'l-start' }

   181→/* ================= Sauvegarde ================= */
   182→interface ChoiceRef {
   183→  node: string
   184→  idx: number
   185→}
   186→
   187→interface StorySave {
   188→  /** Ids des nœuds de fin découverts (les noms se résolvent selon la langue). */
   189→  discovered: string[]
   190→  /** Nœud courant ; null = aucune lecture en cours. */
   191→  node: string | null
   192→  /** Décisions prises depuis le début de la lecture. */
   193→  history: ChoiceRef[]
   194→}
   195→
   196→type SaveMap = Record<string, StorySave>
   197→
   198→function sanitizeSaveMap(raw: unknown, endingsByStory: Record<string, string[]>): SaveMap {
   199→  const out: SaveMap = {}
   200→  if (!raw || typeof raw !== 'object') return out
   201→  const rec = raw as Record<string, unknown>
   202→  for (const st of STORIES) {
   203→    const v = rec[st.id]
   204→    if (!v || typeof v !== 'object') continue
   205→    const s = v as Partial<StorySave>
   206→    const validEndings = new Set(endingsByStory[st.id] ?? [])
   207→    const discovered = Array.isArray(s.discovered)
   208→      ? s.discovered.filter((d): d is string => typeof d === 'string' && validEndings.has(d))
   209→      : []
   210→    const node =
   211→      typeof s.node === 'string' && st.nodes.fr[s.node] && !st.nodes.fr[s.node]?.ending ? s.node : null
   212→    const history = Array.isArray(s.history)
   213→      ? s.history.filter((h): h is ChoiceRef => {
   214→          if (!h || typeof h !== 'object') return false
   215→          const r = h as Partial<ChoiceRef>
   216→          return typeof r.node === 'string' && typeof r.idx === 'number' && st.nodes.fr[r.node] != null
   217→        })
   218→      : []
   219→    out[st.id] = { discovered, node, history }
   220→  }
   221→  return out
   222→}

   224→/* ================= Composant principal ================= */
   225→export default function InteractiveStoryGame() {
   226→  const lang = useLang()
   227→  const reduce = useReducedMotion()
   228→  const L = DICT[lang]
   229→
   230→  const addPlay = useProfileStore((s) => s.addPlay)
   231→  const addWin = useProfileStore((s) => s.addWin)
   232→  const addXp = useProfileStore((s) => s.addXp)
   233→  const setBest = useProfileStore((s) => s.setBest)
   234→  const unlockBadge = useProfileStore((s) => s.unlockBadge)
   235→
   236→  const [saveMap, setSaveMap] = useState<SaveMap | null>(null)
   237→  const [screen, setScreen] = useState<'picker' | 'read'>('picker')
   238→  const [storyId, setStoryId] = useState<string | null>(null)
   239→  const [nodeId, setNodeId] = useState('')
   240→  const [history, setHistory] = useState<ChoiceRef[]>([])
   241→  const [runKey, setRunKey] = useState(0)
   242→  const [confetti, setConfetti] = useState(0)
   243→  const [pendingResume, setPendingResume] = useState<string | null>(null)
   244→  const [historyOpen, setHistoryOpen] = useState(false)
   245→
   246→  const saveRef = useRef<SaveMap>({})
   247→
   248→  /* --- Fins de chaque histoire (graphe identique FR/EN → version fr) --- */
   249→  const endingIds = useMemo(() => {
   250→    const map: Record<string, string[]> = {}
   251→    for (const st of STORIES) {
   252→      map[st.id] = Object.entries(st.nodes.fr)
   253→        .filter(([, n]) => n.ending)
   254→        .map(([id]) => id)
   255→    }
   256→    return map
   257→  }, [])
   258→
   259→  /* --- Restauration asynchrone post-mount (conforme React 19 : jamais de
   260→   *     setState synchrone dans le corps d'un useEffect) --- */
   261→  useEffect(() => {
   262→    let alive = true
   263→    Promise.resolve().then(() => {
   264→      if (!alive) return
   265→      const clean = sanitizeSaveMap(loadSave<SaveMap>(gameId), endingIds)
   266→      saveRef.current = clean
   267→      setSaveMap(clean)
   268→    })
   269→    return () => {
   270→      alive = false
   271→    }
   272→  }, [endingIds])
   273→
   274→  const commit = useCallback((map: SaveMap) => {
   275→    saveRef.current = map
   276→    setSaveMap(map)
   277→    saveSave(gameId, map)
   278→  }, [])
   279→
   280→  const story: Story | undefined = useMemo(
   281→    () => STORIES.find((s) => s.id === storyId),
   282→    [storyId],
   283→  )
   284→  const node = story && nodeId ? story.nodes[lang][nodeId] : undefined
   285→  const chapter = history.length + 1
   286→
   287→  const getStory = (id: string): Story | undefined => STORIES.find((s) => s.id === id)
   288→
   289→  /* Lecture en cours pour une histoire ? (basé sur l'état, jamais sur un ref
   290→   * pendant le rendu — le saveRef n'est lu que dans les handlers) */
   291→  const isSavedInProgress = (id: string): boolean => {
   292→    const st = getStory(id)
   293→    const sv = saveMap?.[id]
   294→    if (!st || !sv?.node) return false
   295→    const n = st.nodes[lang][sv.node]
   296→    return n != null && !n.ending
   297→  }
   298→
   299→  /* --- Lancer une nouvelle lecture --- */
   300→  const startStory = useCallback(
   301→    (id: string) => {
   302→      const st = getStory(id)
   303→      if (!st) return
   304→      const start = STORY_START[id]
   305→      if (!start || !st.nodes[lang][start]) return
   306→      sfx.play('reveal')
   307→      setStoryId(id)
   308→      setNodeId(start)
   309→      setHistory([])
   310→      setRunKey((k) => k + 1)
   311→      setScreen('read')
   312→      setHistoryOpen(false)
   313→      addPlay(gameId)
   314→      const map: SaveMap = { ...saveRef.current }
   315→      const prev = map[id]
   316→      map[id] = { discovered: prev?.discovered ?? [], node: start, history: [] }
   317→      commit(map)
   318→    },
   319→    [lang, addPlay, commit],
   320→  )
   321→
   322→  /* --- Reprendre une lecture sauvegardée --- */
   323→  const resumeStory = useCallback(
   324→    (id: string) => {
   325→      const st = getStory(id)
   326→      const sv = saveMap?.[id]
   327→      if (!st || !sv?.node || !st.nodes[lang][sv.node]) {
   328→        startStory(id)
   329→        return
   330→      }
   331→      sfx.play('reveal')
   332→      setStoryId(id)
   333→      setNodeId(sv.node)
   334→      setHistory(sv.history)
   335→      setRunKey((k) => k + 1)
   336→      setScreen('read')
   337→      setHistoryOpen(false)
   338→      addPlay(gameId)
   339→    },
   340→    [lang, saveMap, addPlay, startStory],
   341→  )
   342→
   343→  /* --- Choisir une décision (toutes les transitions et récompenses
   344→   *     se font dans le handler, jamais dans un effet) --- */
   345→  const choose = useCallback(
   346→    (idx: number) => {
   347→      if (!story || !node || node.ending) return
   348→      const choice = node.choices[idx]
   349→      if (!choice) return
   350→      const nextNode = story.nodes[lang][choice.next]
   351→      if (!nextNode) return
   352→      sfx.play('click')
   353→
   354→      const newHistory = [...history, { node: nodeId, idx }]
   355→      setHistory(newHistory)
   356→      setNodeId(choice.next)
   357→
   358→      const map: SaveMap = { ...saveRef.current }
   359→      const sv: StorySave = { ...(map[story.id] ?? { discovered: [], node: null, history: [] }) }
   360→
   361→      if (nextNode.ending) {
   362→        /* ----- Fin atteinte ----- */
   363→        const isNew = !sv.discovered.includes(choice.next)
   364→        if (isNew) sv.discovered = [...sv.discovered, choice.next]
   365→        sv.node = null
   366→        sv.history = []
   367→        map[story.id] = sv
   368→        commit(map)
   369→
   370→        setConfetti((k) => k + 1)
   371→        sfx.play('win')
   372→        if (isNew) {
   373→          addWin(gameId)
   374→          // 20 XP (nouvelle fin) + 5 XP (lecture terminée) — un rejoue de la
   375→          // même fin ne rapporte rien.
   376→          addXp(25)
   377→          const total = STORIES.reduce((n, s) => n + (map[s.id]?.discovered.length ?? 0), 0)
   378→          setBest(gameId, 'endings-found', total, false)
   379→          unlockBadge('story-first')
   380→          if (sv.discovered.length >= endingIds[story.id].length) unlockBadge('story-explorer')
   381→        }
   382→      } else {
   383→        /* ----- Chapitre suivant ----- */
   384→        sfx.play('reveal', { volume: 0.8 })
   385→        sv.node = choice.next
   386→        sv.history = newHistory
   387→        map[story.id] = sv
   388→        commit(map)
   389→      }
   390→    },
   391→    [story, node, lang, nodeId, history, endingIds, addWin, addXp, setBest, unlockBadge, commit],
   392→  )
   393→
   394→  const backToPicker = () => {
   395→    sfx.play('click')
   396→    setScreen('picker')
   397→    setStoryId(null)
   398→    setNodeId('')
   399→    setHistory([])
   400→    setHistoryOpen(false)
   401→  }
   402→
   403→  const onCardClick = (id: string) => {
   404→    if (isSavedInProgress(id)) {
   405→      sfx.play('click')
   406→      setPendingResume(id)
   407→    } else {
   408→      startStory(id)
   409→    }
   410→  }
   411→
   412→  /* --- Libellé d'une décision passée, résolu dans la langue courante --- */
   413→  const choiceLabel = (ref: ChoiceRef): string => {
   414→    if (!story) return ''
   415→    return story.nodes[lang][ref.node]?.choices[ref.idx]?.label ?? ''
   416→  }
   417→
   418→  const totalEndings = STORIES.reduce((n, s) => n + endingIds[s.id].length, 0)
   419→  const totalFound = saveMap
   420→    ? STORIES.reduce((n, s) => n + (saveMap[s.id]?.discovered.length ?? 0), 0)
   421→    : 0
   422→  const currentEnding = node?.ending
   423→  const isNewEnding = !!(story && currentEnding && saveMap && !(saveMap[story.id]?.discovered.includes(nodeId)))
   424→  const storyFound = story ? saveMap?.[story.id]?.discovered.length ?? 0 : 0
   425→  const storyTotal = story ? endingIds[story.id].length : 0
   426→
   427→  /* ================= Rendu ================= */
   428→  return (
   429→    <div className="relative select-none">
   430→      <Confetti fire={confetti} count={reduce ? 0 : 120} />
   431→
   432→      {screen === 'picker' || !story || !node ? (
   433→        /* ---------- Bibliothèque ---------- */
   434→        <div className="anim-rise">
   435→          <div className="mb-4 flex flex-wrap items-end justify-between gap-3">
   436→            <div className="min-w-0">
   437→              <h2 className="flex items-center gap-2.5 font-display text-2xl font-bold tracking-tight sm:text-3xl">
   438→                <Library className="h-6 w-6 text-orange-500" aria-hidden="true" />
   439→                {L.pickerTitle}
   440→              </h2>
   441→              <p className="mt-1 max-w-2xl text-sm text-muted-foreground sm:text-base">{L.pickerSub}</p>
   442→            </div>
   443→            <span className="flex h-11 shrink-0 items-center gap-1.5 rounded-full bg-amber-100 px-3.5 text-sm font-bold text-amber-700 dark:bg-amber-500/20 dark:text-amber-300">
   444→              <Sparkles className="h-4 w-4" aria-hidden="true" />
   445→              {totalFound}/{totalEndings}
   446→            </span>
   447→          </div>
   448→
   449→          <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
   450→            {STORIES.map((st) => {
   451→              const found = saveMap?.[st.id]?.discovered.length ?? 0
   452→              const total = endingIds[st.id].length
   453→              const resume = isSavedInProgress(st.id)
   454→              return (
   455→                <button
   456→                  key={st.id}
