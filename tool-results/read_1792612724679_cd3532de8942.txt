   238→/** Éparpillement « meilleur candidat » : essaie plusieurs positions et garde
   239→ *  la plus éloignée des pièces déjà posées — bac aéré sans chevauchement massif. */
   240→function scatterOne(
   241→  rand: () => number,
   242→  placed: { x: number; y: number }[],
   243→  pw: number,
   244→  ph: number,
   245→  w: number,
   246→  trayH: number,
   247→): { x: number; y: number } {
   248→  const top = w + GAP + PAD
   249→  const maxX = Math.max(1, w - pw - PAD * 2)
   250→  const maxY = Math.max(1, trayH - ph - PAD * 2)
   251→  let bestX = PAD
   252→  let bestY = top + PAD
   253→  let bestD = -1
   254→  for (let k = 0; k < 8; k++) {
   255→    const x = PAD + rand() * maxX
   256→    const y = top + PAD + rand() * maxY
   257→    let d = Infinity
   258→    for (const q of placed) {
   259→      const dd = Math.hypot(q.x - x, q.y - y)
   260→      if (dd < d) d = dd
   261→    }
   262→    if (d > bestD) {
   263→      bestD = d
   264→      bestX = x
   265→      bestY = y
   266→    }
   267→  }
   268→  return { x: bestX, y: bestY }
   269→}

   455→  /* --- Dimensionnement de la zone de jeu (plateau + bac) :
   456→         la largeur est bornée par le conteneur ET par le budget vertical
   457→         (plateau + bac tiennent exactement dans l'écran, sans scroll),
   458→         avec un bac jamais plus bas qu'une pièce du pire cas (12 pièces).
   459→         ResizeObserver (conteneur + aire, déclenche à l'attachement) et
   460→         resize fenêtre (barre d'URL mobile) rappellent le calcul ; les
   461→         pièces libres sont mises à l'échelle proportionnellement. --- */
   462→  function availableSpace(): number {
   463→    const area = areaRef.current
   464→    if (!area) return -1

   469→  function applyAreaSize(measuredW: number, measuredTh: number) {
   470→    const area = areaRef.current
   471→    if (!area) return
   472→    const containerW = area.parentElement?.clientWidth ?? 0
   473→    const baseW = containerW > 0 ? containerW : measuredW
   474→    if (baseW <= 0) return
   475→    const avail = availableSpace()
   476→    const cssCap = Math.min(420, Math.max(180, window.innerHeight - 412))
   477→    let w: number
   478→    let th: number
   479→    if (avail <= 0) {
   480→      w = Math.min(baseW, cssCap)
   481→      th = measuredTh > 0 ? measuredTh : TRAY_FALLBACK
   482→    } else {
   483→      // Pire cas vertical : 12 pièces → 3 lignes → pièce = plateau/3.
   484→      let trayMin = 132
   485→      w = Math.min(baseW, cssCap, avail - GAP - trayMin)
   486→      trayMin = Math.max(132, w / 3 + 20)
   487→      w = Math.min(baseW, cssCap, avail - GAP - trayMin)
   488→      th = clamp(avail - w - GAP, trayMin, 280)
   489→    }
   490→    w = Math.round(clamp(w, 180, 420))
   491→    th = Math.min(Math.round(th), Math.max(132, Math.floor(avail > 0 ? avail - w - GAP : th)))
   492→    const h = w + GAP + th
   493→    if (w === areaWRef.current && th === trayHRef.current) return
   494→    const prevW = areaWRef.current
   495→    const prevH = areaHRef.current
   496→    const g = gameRef.current
   497→    if (g && prevW > 0 && prevH > 0 && (w !== prevW || h !== prevH)) {
   498→      const fx = w / prevW
   499→      const fy = h / prevH
   500→      const dragged = dragRef.current?.id
   501→      updateGame({
   502→        ...g,
   503→        pieces: g.pieces.map((p) =>
   504→          p.locked || p.id === dragged ? p : { ...p, x: p.x * fx, y: p.y * fy },
   505→        ),
   506→      })
   507→    }
   508→    areaWRef.current = w
   509→    trayHRef.current = th
   510→    areaHRef.current = h
   511→    setAreaW(w)
   512→    setTrayH(th)
   513→  }

   515→  function measureNow() {
   516→    window.requestAnimationFrame(() => {
   517→      const el = areaRef.current
   518→      if (el) applyAreaSize(el.clientWidth, trayRef.current?.clientHeight ?? 0)

   528→    const ro = new ResizeObserver(measure)
   529→    ro.observe(area)
   530→    if (area.parentElement) ro.observe(area.parentElement)
   531→    window.addEventListener('resize', measure)
   532→    return () => {
   533→      ro.disconnect()
   534→      window.removeEventListener('resize', measure)
   535→    }

   545→  /* --- Démarrage d'une partie (image + taille) --- */
   546→  function startGame(imageId: string, count: number) {
   547→    const now = Date.now()
   548→    if (now - lastStartRef.current < 350) return
   549→    lastStartRef.current = now
   550→    clearTimers()
   551→    genRef.current++

   554→    const { cols, rows } = dimsFor(count)
   555→    const w = provisionalSize()
   556→    const pw = w / cols
   557→    const ph = w / rows
   558→    const rand = mulberry32(seedFromString(`jigsaw:${imageId}:${count}`))
   559→    const placed: { x: number; y: number }[] = []
   560→    const pieces: Piece[] = []
   561→    for (let r = 0; r < rows; r++) {
   562→      for (let c = 0; c < cols; c++) {
   563→        const pos = scatterOne(rand, placed, pw, ph, w, TRAY_FALLBACK)
   564→        placed.push(pos)
   565→        pieces.push({ id: r * cols + c, row: r, col: c, locked: false, x: pos.x, y: pos.y })
   566→      }
   567→    }
   568→    elapsedBaseRef.current = 0
   569→    runningSinceRef.current = Date.now()
   570→    areaWRef.current = w
   571→    trayHRef.current = TRAY_FALLBACK
   572→    areaHRef.current = w + GAP + TRAY_FALLBACK
   573→    setElapsedMs(0)
   574→    setWon(null)
   575→    setDraggingId(null)
   576→    setSnapFx(null)
   577→    setBumpId(null)
   578→    setRestored(null)
   579→    setAreaW(w)
   580→    setDealKey((k) => k + 1)
   581→    updateGame({ imageId, count, pieces })
   582→    addPlay(gameId)
   583→    sfx.play('pop')
   584→    measureNow()
   585→  }

   587→  /* --- Reprise de la partie sauvegardée (positions normalisées, remises à l'échelle) --- */
   588→  function resumeGame() {
   589→    if (!restored) return
   590→    const now = Date.now()
   591→    if (now - lastStartRef.current < 350) return
   592→    lastStartRef.current = now
   593→    clearTimers()
   594→    genRef.current++
   595→    snapLockRef.current.clear()
   596→    dragRef.current = null
   597→    const { cols, rows } = dimsFor(restored.count)
   598→    const w = provisionalSize()
   599→    const pw = w / cols
   600→    const ph = w / rows
   601→    const areaH = w + GAP + TRAY_FALLBACK
   602→    const pieces: Piece[] = restored.locked.map((lk, i) => {
   603→      const r = Math.floor(i / cols)
   604→      const c = i % cols
   605→      const sp = restored.pos[i] ?? { x: 0.5, y: 0.8 }
   606→      let px: number
   607→      let py: number
   608→      if (restored.v === 2) {
   609→        px = sp.x * w
   610→        py = sp.y * areaH
   611→      } else {
   612→        // Sauvegarde héritée (v1, pixels bruts) : simple rabattement dans les bornes.
   613→        px = sp.x
   614→        py = sp.y
   615→      }
   616→      const locked = !!lk
   617→      if (locked) {
   618→        px = c * pw
   619→        py = r * ph
   620→      } else {
   621→        px = clamp(px, -pw * 0.25, w - pw * 0.75)
   622→        py = clamp(py, -ph * 0.25, areaH - ph * 0.75)
   623→      }
   624→      return { id: i, row: r, col: c, locked, x: px, y: py }
   625→    })
   626→    elapsedBaseRef.current = restored.elapsed || 0
   627→    runningSinceRef.current = Date.now()
   628→    areaWRef.current = w
   629→    trayHRef.current = TRAY_FALLBACK
   630→    areaHRef.current = areaH
   631→    setElapsedMs(restored.elapsed || 0)
   632→    setWon(null)
   633→    setDraggingId(null)
   634→    setSnapFx(null)
   635→    setBumpId(null)
   636→    setRestored(null)
   637→    setAreaW(w)
   638→    setDealKey((k) => k + 1)
   639→    const nextGame: PlayState = { imageId: restored.imageId, count: restored.count, pieces }
   640→    updateGame(nextGame)
   641→    addPlay(gameId)
   642→    sfx.play('pop')
   643→    measureNow()
   644→    // Filet de sécurité : une sauvegarde intégralement verrouillée (victoire interrompue
   645→    // pendant le délai de célébration) déclenche la victoire à la reprise.
   646→    if (pieces.every((p) => p.locked)) {
   647→      setTimer(() => finishWin(nextGame), 300)
   648→    }
   649→  }

   729→  function onPieceUp(e: React.PointerEvent<HTMLDivElement>) {
   730→    const d = dragRef.current
   731→    if (!d || d.pointerId !== e.pointerId) return
   732→    dragRef.current = null
   733→    setDraggingId(null)
   734→    const g = gameRef.current
   735→    const el = e.currentTarget
   736→    const piece = g?.pieces.find((x) => x.id === d.id)
   737→    if (!g || !piece || piece.locked) {
   738→      el.style.transition = ''
   739→      return
   740→    }
   741→    const { cols, rows } = dimsFor(g.count)
   742→    const w = areaWRef.current
   743→    const pw = w / cols
   744→    const ph = w / rows
   745→    // Position réellement affichée (style impératif inclus) — jamais de saut.
   746→    const nx = el.offsetLeft
   747→    const ny = el.offsetTop
   748→
   749→    // Aimantation : distance du coin au coin (équivalente centre-centre, la pièce
   750→    // et sa case ayant exactement les mêmes dimensions).
   751→    const tx = piece.col * pw
   752→    const ty = piece.row * ph
   753→    if (Math.hypot(nx - tx, ny - ty) <= snapRadius(pw, ph)) {
   754→      snapLockRef.current.add(piece.id)
   755→      sfx.play('drop')
   756→      buzz(10)
   757→      // Transition posée directement sur le DOM, commit d'état après l'atterrissage.
   758→      el.style.transition = 'left 160ms ease-out, top 160ms ease-out'
   759→      el.style.left = `${tx}px`
   760→      el.style.top = `${ty}px`
   761→      const gen = genRef.current
   762→      setTimer(() => {
   763→        el.style.transition = ''
   764→        snapLockRef.current.delete(piece.id)
   765→        if (genRef.current !== gen) return
   766→        const g2 = gameRef.current
   767→        if (!g2) return
   768→        const pieces = g2.pieces.map((x) => (x.id === piece.id ? { ...x, x: tx, y: ty, locked: true } : x))
   769→        updateGame({ ...g2, pieces })
   770→        setBumpId(piece.id)
   771→        setSnapFx(piece.id)
   772→        setTimer(() => setSnapFx(null), 500)
   773→        if (pieces.every((x) => x.locked)) {
   774→          // Petit délai pour laisser voir la dernière pièce se verrouiller.
   775→          setTimer(() => {
   776→            if (genRef.current === gen) finishWin(g2)
   777→          }, 420)
   778→        }
   779→      }, 170)
   780→    } else {
   781→      const areaH = totalAreaH()
   782→      const cx = clamp(nx, -pw * 0.25, w - pw * 0.75)
   783→      const cy = clamp(ny, -ph * 0.25, areaH - ph * 0.75)
   784→      if (cx !== nx || cy !== ny) {
   785→        el.style.transition = 'left 120ms ease-out, top 120ms ease-out'
   786→        setTimer(() => {
   787→          el.style.transition = ''
   788→        }, 140)
   789→      }
   790→      sfx.play('flip', { volume: 0.35 })
   791→      const pieces = g.pieces.map((x) => (x.id === piece.id ? { ...x, x: cx, y: cy } : x))
   792→      updateGame({ ...g, pieces })
   793→      setBumpId(piece.id)
   794→    }
   795→  }
