   199→function rr(ctx: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number) {
   200→  const rad = Math.min(r, w / 2, h / 2)
   201→  ctx.beginPath()
   202→  ctx.moveTo(x + rad, y)
   203→  ctx.arcTo(x + w, y, x + w, y + h, rad)
   204→  ctx.arcTo(x + w, y + h, x, y + h, rad)
   205→  ctx.arcTo(x, y + h, x, y, rad)
   206→  ctx.arcTo(x, y, x + w, y, rad)
   207→  ctx.closePath()
   208→}

   214→function segCircle(
   215→  x0: number, y0: number, x1: number, y1: number,
   216→  cx: number, cy: number, r: number,
   217→): boolean {
   218→  const dx = x1 - x0
   219→  const dy = y1 - y0
   220→  const l2 = dx * dx + dy * dy
   221→  let t = 0
   222→  if (l2 > 0) t = clamp(((cx - x0) * dx + (cy - y0) * dy) / l2, 0, 1)
   223→  const px = x0 + t * dx
   224→  const py = y0 + t * dy
   225→  return (px - cx) * (px - cx) + (py - cy) * (py - cy) <= r * r
   226→}

   228→function drawBackdrop(ctx: CanvasRenderingContext2D, W: number, H: number) {
   229→  // Ciel chaud
   230→  const sky = ctx.createLinearGradient(0, 0, 0, H)
   231→  sky.addColorStop(0, '#ffedd0')
   232→  sky.addColorStop(0.55, '#f9d9a6')
   233→  sky.addColorStop(1, '#f2c48c')
   234→  ctx.fillStyle = sky
   235→  ctx.fillRect(0, 0, W, H)
   236→  // Halo de soleil
   237→  const sun = ctx.createRadialGradient(W * 0.78, H * 0.16, 8, W * 0.78, H * 0.16, H * 0.5)
   238→  sun.addColorStop(0, 'rgba(255,244,214,0.9)')
   239→  sun.addColorStop(1, 'rgba(255,244,214,0)')
   240→  ctx.fillStyle = sun
   241→  ctx.fillRect(0, 0, W, H)
   242→  // Collines
   243→  ctx.fillStyle = 'rgba(143,191,127,0.55)'
   244→  ctx.beginPath()
   245→  ctx.moveTo(0, H)
   246→  ctx.lineTo(0, H - 34)
   247→  ctx.quadraticCurveTo(W * 0.3, H - 72, W * 0.62, H - 30)
   248→  ctx.quadraticCurveTo(W * 0.85, H - 2, W, H - 26)
   249→  ctx.lineTo(W, H)
   250→  ctx.closePath()
   251→  ctx.fill()
   252→  // Silhouettes d'arbres
   253→  const tree = (x: number, s: number) => {
   254→    ctx.fillStyle = 'rgba(109,124,82,0.75)'
   255→    ctx.fillRect(x - 3 * s, H - 34 * s, 6 * s, 34 * s)
   256→    ctx.beginPath()
   257→    ctx.arc(x, H - 44 * s, 16 * s, 0, Math.PI * 2)
   258→    ctx.arc(x - 12 * s, H - 36 * s, 11 * s, 0, Math.PI * 2)
   259→    ctx.arc(x + 12 * s, H - 36 * s, 11 * s, 0, Math.PI * 2)
   260→    ctx.fill()
   261→  }
   262→  tree(W * 0.14, 1)
   263→  tree(W * 0.52, 0.8)
   264→  tree(W * 0.88, 1.1)
   265→  // Sol
   266→  ctx.fillStyle = 'rgba(122,96,58,0.5)'
   267→  ctx.fillRect(0, H - 12, W, 12)
   268→}

   414→    const sliceFruit = (f: Fruit, ang: number, now: number) => {
   415→      if (w.dead) return
   416→      w.fruits = w.fruits.filter((o) => o.id !== f.id)
   417→      const px = -Math.sin(ang)
   418→      const py = Math.cos(ang)
   419→      for (const side of [-1, 1]) {
   420→        w.halves.push({
   421→          emoji: f.emoji,
   422→          x: f.x + px * side * f.r * 0.4,
   423→          y: f.y + py * side * f.r * 0.4,
   424→          vx: f.vx * 0.55 + px * side * 190,
   425→          vy: f.vy * 0.55 + py * side * 190 - 60,
   426→          r: f.r,
   427→          rot: ang,
   428→          vr: side * 3.2,
   429→          side,
   430→          life: 0.9,
   431→        })
   432→      }
   433→      if (!reduceRef.current) {
   434→        for (let i = 0; i < 9; i++) {
   435→          const a = Math.random() * Math.PI * 2
   436→          const sp = 50 + Math.random() * 200
   437→          w.juices.push({
   438→            x: f.x, y: f.y,
   439→            vx: Math.cos(a) * sp, vy: Math.sin(a) * sp - 60,
   440→            r: 2 + Math.random() * 4,
   441→            life: 0.5 + Math.random() * 0.3, max: 0.8,
   442→            color: f.juice,
   443→          })
   444→        }
   445→      }
   446→
   447→      if (f.soap) {
   448→        w.lives--
   449→        w.flash = 1
   450→        w.chain = 0
   451→        sfx.play('wrong')
   452→        w.pops.push({ x: f.x, y: f.y - 10, text: L[langRef.current].lostLife, life: 1, max: 1, color: '#d9553f', size: 20 })
   453→        if (w.lives <= 0) {
   454→          finishDeath()
   455→        }
   456→      } else {
   457→        w.fruitsSliced++
   458→        w.chain = now - w.lastSlice <= 350 ? w.chain + 1 : 1
   459→        w.lastSlice = now
   460→        w.bestChain = Math.max(w.bestChain, w.chain)
   461→        let gained = f.points
   462→        if (w.chain >= 3) {
   463→          const bonus = w.chain * 10
   464→          gained += bonus
   465→          sfx.play('coin')
   466→          w.pops.push({
   467→            x: f.x, y: f.y - 34,
   468→            text: `COMBO ×${w.chain} +${bonus}`,
   469→            life: 1.1, max: 1.1, color: '#b45309', size: 19,
   470→          })
   471→        } else {
   472→          sfx.play('slice')
   473→        }
   474→        w.score += gained
   475→        w.pops.push({ x: f.x, y: f.y - 6, text: `+${gained}`, life: 0.85, max: 0.85, color: '#7c4a1e', size: 15 })
   476→        if (w.chain >= 4) unlockBadge('fruit-combo-4')
   477→        // Défi du jour
   478→        if (!dailyDoneRef.current && w.score >= dailyTarget) {
   479→          completeDaily(gameId)
   480→          sfx.play('win')
   481→          setConfetti((k) => k + 1)
   482→        }
   483→      }
   484→      pushHud()
   485→    }
   486→
   487→    /* ----- entrées ----- */
   488→    const rectPos = (e: PointerEvent) => {
   489→      const r = cv.getBoundingClientRect()
   490→      return { x: e.clientX - r.left, y: e.clientY - r.top }
   491→    }
   492→    const onDown = (e: PointerEvent) => {
   493→      if (phaseRef.current === 'over') {
   494→        handleStart()
   495→        return
   496→      }
   497→      if (phaseRef.current !== 'playing') return
   498→      const p = rectPos(e)
   499→      w.swipe = { active: true, x: p.x, y: p.y, t: performance.now() }
   500→      w.trail.push({ x: p.x, y: p.y, t: performance.now() })
   501→      cv.setPointerCapture(e.pointerId)
   502→    }
   503→    const onMove = (e: PointerEvent) => {
   504→      if (!w.swipe.active || phaseRef.current !== 'playing') return
   505→      const now = performance.now()
   506→      const p = rectPos(e)
   507→      const dt = now - w.swipe.t
   508→      const dist = Math.hypot(p.x - w.swipe.x, p.y - w.swipe.y)
   509→      // Seuil : ~6 px par frame (0,36 px/ms) pour que la lame coupe
   510→      if (dt > 0 && dist / dt > 0.36 && dist >= 3) {
   511→        const ang = Math.atan2(p.y - w.swipe.y, p.x - w.swipe.x)
   512→        for (const f of [...w.fruits]) {
   513→          if (segCircle(w.swipe.x, w.swipe.y, p.x, p.y, f.x, f.y, f.r * 0.95)) {
   514→            sliceFruit(f, ang, now)
   515→            if (phaseRef.current !== 'playing') break
   516→          }
   517→        }
   518→      }
   519→      w.swipe.x = p.x
   520→      w.swipe.y = p.y
   521→      w.swipe.t = now
   522→      w.trail.push({ x: p.x, y: p.y, t: now })
   523→    }
   524→    const onUp = () => { w.swipe.active = false }
   525→    const onCtx = (e: Event) => e.preventDefault()
   526→    const onVis = () => { pausedRef.current = document.hidden }
   527→
   528→    cv.addEventListener('pointerdown', onDown)
   529→    cv.addEventListener('pointermove', onMove)
   530→    cv.addEventListener('pointerup', onUp)
   531→    cv.addEventListener('pointercancel', onUp)
   532→    cv.addEventListener('contextmenu', onCtx)
   533→    document.addEventListener('visibilitychange', onVis)
   534→
   535→    /* ----- simulation ----- */
   536→    const update = (dt: number, now: number) => {
   537→      if (phaseRef.current !== 'playing') {
   538→        // Les restes continuent de tomber derrière l'écran de fin
   539→        for (const f of w.fruits) {
   540→          f.vy += gravity() * dt
   541→          f.x += f.vx * dt
   542→          f.y += f.vy * dt
   543→          f.rot += f.vr * dt
   544→        }
   545→        w.fruits = w.fruits.filter((f) => f.y < w.H + f.r + 60)
   546→        updateDebris(dt)
   547→        return
   548→      }
   549→      const elapsed = w.t - w.startT
   550→      w.spawnT -= dt
   551→      if (w.spawnT <= 0) {
   552→        w.spawnT = Math.max(0.55, 1.35 - elapsed * 0.007) * (0.75 + Math.random() * 0.5)
   553→        const count = 1 + (Math.random() < 0.35 ? 1 : 0) + (Math.random() < 0.2 ? 1 : 0) + (Math.random() < 0.1 ? 1 : 0)
   554→        for (let i = 0; i < count; i++) spawnItem()
   555→      }
   556→      for (const f of w.fruits) {
   557→        f.vy += gravity() * dt
   558→        f.x += f.vx * dt
   559→        f.y += f.vy * dt
   560→        f.rot += f.vr * dt
   561→      }
   562→      // Fruits manqués : casse le combo, rien de plus
   563→      const before = w.fruits.length
   564→      w.fruits = w.fruits.filter((f) => f.y < w.H + f.r + 40)
   565→      if (w.fruits.length < before && w.chain > 0) {
   566→        w.chain = 0
   567→        pushHud()
   568→      }
   569→      // Fin de fenêtre combo
   570→      if (w.chain > 0 && now - w.lastSlice > 350) {
   571→        w.chain = 0
   572→        pushHud()
   573→      }
   574→      updateDebris(dt)
   575→    }
   576→    const updateDebris = (dt: number) => {
   577→      for (let i = w.halves.length - 1; i >= 0; i--) {
   578→        const h = w.halves[i]
   579→        h.vy += gravity() * dt
   580→        h.x += h.vx * dt
   581→        h.y += h.vy * dt
   582→        h.rot += h.vr * dt
   583→        h.life -= dt
   584→        if (h.life <= 0 || h.y > w.H + 90) w.halves.splice(i, 1)
   585→      }
   586→      for (let i = w.juices.length - 1; i >= 0; i--) {
   587→        const j = w.juices[i]
   588→        j.vy += 700 * dt
   589→        j.x += j.vx * dt
   590→        j.y += j.vy * dt
   591→        j.life -= dt
   592→        if (j.life <= 0) w.juices.splice(i, 1)
   593→      }
   594→      for (let i = w.pops.length - 1; i >= 0; i--) {
   595→        const p = w.pops[i]
   596→        p.y -= 32 * dt
   597→        p.life -= dt
   598→        if (p.life <= 0) w.pops.splice(i, 1)
   599→      }
   600→      const cutoff = performance.now() - 150
   601→      while (w.trail.length > 0 && w.trail[0].t < cutoff) w.trail.shift()
   602→      if (w.flash > 0) w.flash = Math.max(0, w.flash - dt * 2.2)
   603→    }
   604→
   605→    /* ----- rendu ----- */
   606→    const draw = () => {
   607→      const W = w.W, H = w.H
   608→      ctx.setTransform(dpr, 0, 0, dpr, 0, 0)
   609→      ctx.clearRect(0, 0, W, H)
   610→      // Secousse à l'erreur (désactivée si mouvement réduit)
   611→      if (w.flash > 0.55 && !reduceRef.current) {
   612→        ctx.translate((Math.random() - 0.5) * 6 * w.flash, (Math.random() - 0.5) * 6 * w.flash)
   613→      }
   614→      if (bg) ctx.drawImage(bg, 0, 0, W, H)
   615→
   616→      // Trainée lumineuse du geste
   617→      if (!reduceRef.current && w.trail.length > 1) {
   618→        ctx.save()
   619→        ctx.lineCap = 'round'
   620→        ctx.lineJoin = 'round'
   621→        for (let i = 1; i < w.trail.length; i++) {
   622→          const a = i / w.trail.length
   623→          ctx.strokeStyle = `rgba(255,246,220,${0.7 * a})`
   624→          ctx.lineWidth = 2 + 7 * a
   625→          ctx.shadowColor = 'rgba(247,200,115,0.9)'
   626→          ctx.shadowBlur = 12
   627→          ctx.beginPath()
   628→          ctx.moveTo(w.trail[i - 1].x, w.trail[i - 1].y)
   629→          ctx.lineTo(w.trail[i].x, w.trail[i].y)
   630→          ctx.stroke()
   631→        }
   632→        ctx.restore()
   633→      }
   634→
   635→      // Jus
   636→      for (const j of w.juices) {
   637→        ctx.globalAlpha = clamp(j.life / j.max, 0, 1)
   638→        ctx.fillStyle = j.color
   639→        ctx.beginPath()
   640→        ctx.arc(j.x, j.y, j.r, 0, Math.PI * 2)
   641→        ctx.fill()
   642→      }
   643→      ctx.globalAlpha = 1
   644→
   645→      ctx.textAlign = 'center'
   646→      ctx.textBaseline = 'middle'
   647→
   648→      // Moitiés
   649→      for (const h of w.halves) {
   650→        ctx.save()
   651→        ctx.globalAlpha = clamp(h.life / 0.9, 0, 1)
   652→        ctx.translate(h.x, h.y)
   653→        ctx.rotate(h.rot)
   654→        const cr = h.r * 1.6
   655→        ctx.beginPath()
   656→        ctx.rect(-cr, h.side < 0 ? -cr : 0, cr * 2, cr)
   657→        ctx.clip()
   658→        ctx.font = emojiFont(h.r * 2)
   659→        ctx.fillText(h.emoji, 0, 0)
   660→        ctx.restore()
   661→      }
   662→
   663→      // Fruits et savons
   664→      for (const f of w.fruits) {
   665→        ctx.save()
   666→        ctx.translate(f.x, f.y)
   667→        ctx.rotate(f.rot)
   668→        ctx.font = emojiFont(f.r * 2)
   669→        ctx.fillText(f.emoji, 0, 0)
   670→        if (f.soap) {
   671→          ctx.fillStyle = 'rgba(217,140,179,0.3)'
   672→          ctx.beginPath()
   673→          ctx.arc(0, 0, f.r + 7, 0, Math.PI * 2)
   674→          ctx.fill()
   675→        }
   676→        ctx.restore()
   677→      }
   678→
   679→      // Textes flottants
   680→      for (const p of w.pops) {
   681→        ctx.globalAlpha = clamp(p.life / p.max, 0, 1)
   682→        ctx.fillStyle = p.color
   683→        ctx.font = `800 ${p.size}px "Nunito", system-ui, sans-serif`
   684→        ctx.fillText(p.text, p.x, p.y)
   685→      }
   686→      ctx.globalAlpha = 1
   687→
   688→      // Flash rouge (savon tranché)
   689→      if (w.flash > 0) {
   690→        const g = ctx.createRadialGradient(W / 2, H / 2, Math.min(W, H) * 0.3, W / 2, H / 2, Math.max(W, H) * 0.75)
   691→        g.addColorStop(0, 'rgba(200,50,40,0)')
   692→        g.addColorStop(1, `rgba(200,50,40,${0.55 * w.flash})`)
   693→        ctx.fillStyle = g
   694→        ctx.fillRect(0, 0, W, H)
   695→      }
   696→    }
   697→
   698→    let last = performance.now()
   699→    const step = (now: number) => {
   700→      rafRef.current = requestAnimationFrame(step)
   701→      const dt = Math.min((now - last) / 1000, 0.05)
   702→      last = now
   703→      if (pausedRef.current) return
   704→      w.t += dt
   705→      update(dt, now)
   706→      draw()
   707→    }
   708→    rafRef.current = requestAnimationFrame(step)
   709→
   710→    const ro = new ResizeObserver(resize)
   711→    ro.observe(wrap)
   712→
   713→    return () => {
   714→      cancelAnimationFrame(rafRef.current)
   715→      ro.disconnect()
   716→      cv.removeEventListener('pointerdown', onDown)
   717→      cv.removeEventListener('pointermove', onMove)
   718→      cv.removeEventListener('pointerup', onUp)
   719→      cv.removeEventListener('pointercancel', onUp)
   720→      cv.removeEventListener('contextmenu', onCtx)
   721→      document.removeEventListener('visibilitychange', onVis)
   722→    }
   723→  }, [completeDaily, dailyTarget, finishDeath, handleStart, pushHud, unlockBadge])
   724→
   725→  /* ================= Rendu React ================= */
   726→  return (
   727→    <div className="relative select-none">
   728→      <Confetti fire={confetti} count={reduce ? 0 : 90} />
   729→
   730→      {/* Barre d'état */}
   731→      <div className="mb-3 flex flex-wrap items-center gap-2">
   732→        {phase === 'playing' ? (
   733→          <>
   734→            <StatChip icon="⭐" label={T.score} value={hud.score} />
   735→            <div className="flex h-11 items-center gap-1.5 rounded-full bg-secondary/70 px-3.5 text-sm font-semibold" aria-label={T.lives}>
   736→              <span className="text-muted-foreground">{T.lives}</span>
   737→              <span className="flex gap-0.5" aria-hidden="true">
   738→                {[0, 1, 2].map((i) => (
   739→                  <Heart key={i} className={cn('h-4 w-4', i < hud.lives ? 'fill-rose-500 text-rose-500' : 'text-foreground/20')} />
   740→                ))}
   741→              </span>
   742→            </div>
   743→            {hud.chain >= 2 ? <StatChip icon="🔥" label={T.combo} value={`×${hud.chain}`} /> : null}
   744→            {!dailyDone ? (
   745→              <span className="flex h-11 items-center gap-1.5 rounded-full bg-amber-100 px-3.5 text-sm font-bold text-amber-700 dark:bg-amber-500/20 dark:text-amber-300">
   746→                🎯 {T.daily.replace('{n}', String(dailyTarget))}
   747→              </span>
   748→            ) : null}
   749→          </>
   750→        ) : (
   751→          <>
   752→            {hydrated && bestScore > 0 ? <StatChip icon="⭐" label={`${T.score} — ${T.best}`} value={bestScore} /> : null}
   753→            {!dailyDone ? (
   754→              <span className="flex h-11 items-center gap-1.5 rounded-full bg-amber-100 px-3.5 text-sm font-bold text-amber-700 dark:bg-amber-500/20 dark:text-amber-300">
   755→                🎯 {T.daily.replace('{n}', String(dailyTarget))}
   756→              </span>
   757→            ) : null}
   758→          </>
   759→        )}
   760→      </div>
   761→
   762→      {/* Canvas + habillage */}
   763→      <div ref={wrapRef} className="relative w-full overflow-hidden rounded-2xl border shadow-inner">
   764→        <canvas
   765→          ref={canvasRef}
   766→          className="block w-full touch-none"
   767→          style={{ touchAction: 'none' }}
   768→          aria-label="Fruit Slice"
   769→        />
   770→
   771→        {/* Écran d'accueil */}
   772→        {phase === 'idle' ? (
   773→          <div className="absolute inset-0 z-20 flex flex-col items-center justify-center gap-4 bg-stone-900/40 p-6 text-center backdrop-blur-[2px]">
   774→            <div className="flex gap-2" aria-hidden="true">
   775→              {['🍉', '🔪', '🍎'].map((e, i) => (
   776→                <span key={i} className="text-4xl anim-floaty" style={{ animationDelay: `${i * 0.4}s` }}>{e}</span>
   777→              ))}
   778→            </div>
   779→            <h3 className="font-display text-2xl font-extrabold text-white drop-shadow">🍉 {T.title}</h3>
   780→            <p className="max-w-sm text-sm text-stone-100">{content.tagline[lang]}</p>
   781→            <p className="max-w-xs rounded-xl bg-white/15 px-3 py-2 text-xs text-stone-100">{T.hint}</p>
   782→            <Button size="lg" className="h-12 rounded-full px-7 text-base anim-glow" onClick={handleStart}>
   783→              <Play className="h-5 w-5" aria-hidden="true" /> {T.play}
   784→            </Button>
   785→          </div>
   786→        ) : null}
   787→
   788→        {/* Game over */}
   789→        {phase === 'over' && result ? (
   790→          <div className="absolute inset-0 z-20 flex items-center justify-center bg-stone-900/55 p-4 backdrop-blur-[2px]">
   791→            <div className="anim-rise w-full max-w-sm rounded-2xl border bg-card p-6 text-center shadow-xl">
   792→              <p className="font-display text-xl font-extrabold">🍉 {T.over}</p>
   793→              <p className="mt-3 text-5xl font-extrabold tabular-nums text-primary">{result.score}</p>
   794→              <p className="text-xs text-muted-foreground">{T.score}</p>
   795→              <div className="mt-3 flex justify-center gap-3 text-sm">
   796→                <span className="rounded-xl bg-secondary/70 px-3 py-2">
   797→                  <Swords className="mr-1 inline h-4 w-4 text-muted-foreground" aria-hidden="true" />
   798→                  <span className="font-bold tabular-nums">{result.sliced}</span> <span className="text-muted-foreground">{T.fruitsSliced}</span>
   799→                </span>
   800→                <span className="rounded-xl bg-secondary/70 px-3 py-2">
   801→                  🔥 <span className="font-bold tabular-nums">×{result.bestChain}</span> <span className="text-muted-foreground">{T.bestCombo}</span>
   802→                </span>
   803→              </div>
   804→              {result.isRecord ? (
   805→                <p className="mt-3 text-sm font-bold text-amber-600 dark:text-amber-400">🏆 {T.newBest}</p>
   806→              ) : (
   807→                <p className="mt-3 text-sm text-muted-foreground">
   808→                  {T.best} : <span className="font-bold tabular-nums">{Math.max(bestScore, result.score)}</span>
   809→                </p>
   810→              )}
   811→              <p className="mt-2 text-xs text-muted-foreground">{T.tip}</p>
   812→              <div className="mt-4 flex justify-center">
   813→                <Button size="lg" className="h-12 rounded-full px-8" onClick={handleStart}>
   814→                  <RotateCcw className="h-5 w-5" aria-hidden="true" /> {T.replay}
   815→                </Button>
   816→              </div>
   817→            </div>
   818→          </div>
   819→        ) : null}
   820→      </div>
   821→    </div>
   822→  )
   823→}
   824→
