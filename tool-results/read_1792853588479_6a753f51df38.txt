   338→   338→  const flip = (idx: number) => {

   353→   353→
   354→   354→    if (deck[a].emoji === deck[b].emoji) {

   657→   657→function fmt(ms: number): string {
   658→   658→  const s = ms / 1000
   659→   659→  if (s < 60) return `${s.toFixed(1)}s`
   660→   660→  const m = Math.floor(s / 60)
   661→   661→  return `${m}:${String(Math.floor(s % 60)).padStart(2, '0')}`
   662→   662→}
