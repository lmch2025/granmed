   217→/* ================= Helpers dessin ================= */
   218→function rr(ctx: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number) {
   219→  const rad = Math.min(r, w / 2, h / 2)
   220→  ctx.beginPath()
   221→  ctx.moveTo(x + rad, y)
   222→  ctx.arcTo(x + w, y, x + w, y + h, rad)
   223→  ctx.arcTo(x + w, y + h, x, y + h, rad)
   224→  ctx.arcTo(x, y + h, x, y, rad)
   225→  ctx.arcTo(x, y, x + w, y, rad)
   226→  ctx.closePath()
   227→}

   229→function drawBackdrop(ctx: CanvasRenderingContext2D, W: number, H: number) {
   230→  const g = ctx.createLinearGradient(0, 0, 0, H)
   231→  g.addColorStop(0, '#5e2a1b')
   232→  g.addColorStop(0.5, '#4a2115')
   233→  g.addColorStop(1, '#33150d')
   234→  ctx.fillStyle = g
   235→  ctx.fillRect(0, 0, W, H)
   236→  // Vignette terracotta sombre sur les bords
   237→  const v = ctx.createRadialGradient(W / 2, H / 2, Math.min(W, H) * 0.25, W / 2, H / 2, Math.max(W, H) * 0.8)
   238→  v.addColorStop(0, 'rgba(0,0,0,0)')
   239→  v.addColorStop(1, 'rgba(20,8,4,0.55)')
   240→  ctx.fillStyle = v
   241→  ctx.fillRect(0, 0, W, H)
   242→}

   244→function drawBricks(ctx: CanvasRenderingContext2D, W: number, H: number, scroll: number) {
   245→  const layers = [
   246→    { f: 0.22, a: 0.045, rowH: 46, bw: 64 },
   247→    { f: 0.5, a: 0.08, rowH: 60, bw: 78 },
   248→  ]
   249→  for (const L2 of layers) {
   250→    ctx.fillStyle = `rgba(222,164,110,${L2.a})`
   251→    const off = (scroll * L2.f) % L2.rowH
   252→    let rowIdx = -Math.ceil(H / L2.rowH) - 1
   253→    for (let y = -L2.rowH + off; y < H + L2.rowH; y += L2.rowH) {
   254→      const shift = rowIdx % 2 === 0 ? 0 : L2.bw / 2
   255→      for (let x = -L2.bw + shift; x < W + L2.bw; x += L2.bw) {
   256→        rr(ctx, x, y, L2.bw - 5, L2.rowH - 5, 3)
   257→        ctx.fill()
   258→      }
   259→      rowIdx++
   260→    }
   261→  }
   262→}

   264→function flameTongue(
   265→  ctx: CanvasRenderingContext2D, baseX: number, dir: number, ty: number,
   266→  tl: number, th: number, t: number, seed: number, reduce: boolean,
   267→) {
   268→  const tipX = baseX + dir * tl
   269→  const wave = reduce ? 0 : Math.sin(t * 7 + seed) * th * 0.28
   270→  ctx.beginPath()
   271→  ctx.moveTo(baseX, ty - th)
   272→  ctx.quadraticCurveTo(baseX + dir * tl * 0.35, ty - th - wave, tipX, ty - th * 0.12)
   273→  ctx.quadraticCurveTo(baseX + dir * tl * 0.3, ty + th * 0.65, baseX, ty + th)
   274→  ctx.closePath()
   275→  ctx.fill()
   276→}

   278→function drawFlames(
   279→  ctx: CanvasRenderingContext2D, W: number, y: number,
   280→  leftLen: number, rightLen: number, h: number,
   281→  t: number, seed: number, reduce: boolean,
   282→) {
   283→  // Halo chaleureux
   284→  for (const [len, side] of [[leftLen, -1], [rightLen, 1]] as [number, number][]) {
   285→    if (len < 8) continue
   286→    const cx = side < 0 ? len * 0.55 : W - len * 0.55
   287→    const g = ctx.createRadialGradient(cx, y, 4, cx, y, Math.max(len * 0.75, 30))
   288→    g.addColorStop(0, 'rgba(246,194,106,0.30)')
   289→    g.addColorStop(1, 'rgba(246,194,106,0)')
   290→    ctx.fillStyle = g
   291→    ctx.fillRect(0, y - h * 1.6, W, h * 3.2)
   292→  }
   293→  const layers = [
   294→    { c: '#a83226', s: 1 },
   295→    { c: '#e0702c', s: 0.72 },
   296→    { c: '#f6c26a', s: 0.44 },
   297→  ]
   298→  for (const [len, side] of [[leftLen, -1], [rightLen, 1]] as [number, number][]) {
   299→    if (len < 8) continue
   300→    const dir = side < 0 ? 1 : -1
   301→    const baseX = side < 0 ? 1 : W - 1
   302→    // Socle brûlé au mur
   303→    ctx.fillStyle = '#46281a'
   304→    rr(ctx, side < 0 ? 0 : W - 9, y - h / 2 - 4, 9, h + 8, 3)
   305→    ctx.fill()
   306→    for (const layer of layers) {
   307→      ctx.fillStyle = layer.c
   308→      for (let i = 0; i < 3; i++) {
   309→        const ty = y + (i - 1) * h * 0.3
   310→        const scale = i === 1 ? 1 : 0.76
   311→        const tl = len * layer.s * scale * (reduce ? 1 : 1 + Math.sin(t * 9 + i * 2.1 + seed) * 0.08)
   312→        const th = h * 0.3 * (i === 1 ? 1.12 : 0.82)
   313→        flameTongue(ctx, baseX, dir, ty, tl, th, t, seed + i * 1.7, reduce)
   314→      }
   315→    }
   316→  }
   317→}

   319→function drawFirefighter(
   320→  ctx: CanvasRenderingContext2D, x: number, y: number,
   321→  vy: number, holding: boolean, t: number, reduce: boolean,
   322→) {
   323→  ctx.save()
   324→  ctx.translate(x, y)
   325→  ctx.rotate(clamp(vy * 0.00045, -0.25, 0.3))
   326→  // Jet de propulsion
   327→  if (holding && !reduce) {
   328→    for (let i = 0; i < 3; i++) {
   329→      const p = (t * 2.6 + i * 0.33) % 1
   330→      ctx.globalAlpha = (1 - p) * 0.55
   331→      ctx.fillStyle = '#f4f1ea'
   332→      ctx.beginPath()
   333→      ctx.arc((i - 1) * 5, 23 + p * 15, 3 + p * 4, 0, Math.PI * 2)
   334→      ctx.fill()
   335→    }
   336→    ctx.globalAlpha = 1
   337→  }
   338→  // Bonbonne (accent violet)
   339→  ctx.fillStyle = '#9b8cf2'
   340→  rr(ctx, -14, -12, 7, 20, 3)
   341→  ctx.fill()
   342→  // Jambes
   343→  ctx.fillStyle = '#6b4a2f'
   344→  rr(ctx, -6, 8, 5, 10, 2.5)
   345→  ctx.fill()
   346→  rr(ctx, 2, 8, 5, 10, 2.5)
   347→  ctx.fill()
   348→  // Veste
   349→  ctx.fillStyle = '#f2a65a'
   350→  rr(ctx, -9, -10, 19, 22, 6)
   351→  ctx.fill()
   352→  // Bande réfléchissante
   353→  ctx.fillStyle = '#f6e3c2'
   354→  rr(ctx, -9, -1, 19, 4, 2)
   355→  ctx.fill()
   356→  // Bras
   357→  ctx.fillStyle = '#e08c3c'
   358→  rr(ctx, 8, -8, 5, 12, 2.5)
   359→  ctx.fill()
   360→  // Tête
   361→  ctx.fillStyle = '#f5cba0'
   362→  ctx.beginPath()
   363→  ctx.arc(2, -17, 7.5, 0, Math.PI * 2)
   364→  ctx.fill()
   365→  // Casque
   366→  ctx.fillStyle = '#d9553f'
   367→  ctx.beginPath()
   368→  ctx.arc(2, -18, 8, Math.PI, 0)
   369→  ctx.fill()
   370→  rr(ctx, -8.5, -19, 21, 3.5, 1.75)
   371→  ctx.fill()
   372→  ctx.fillStyle = '#f6c26a'
   373→  rr(ctx, 0.5, -23, 4, 3, 1)
   374→  ctx.fill()
   375→  ctx.restore()
   376→}

   436→    w.dead = true
   437→    const dist = Math.floor(w.scroll / 10)
   438→    const prevBest = useProfileStore.getState().profile.games[gameId]?.records?.['best-distance']?.value ?? 0
   439→    const isRecord = dist > prevBest && dist > 0
   440→    if (dist > 0) setBest(gameId, 'best-distance', dist)
   441→    addXp(Math.max(1, Math.floor(dist / 20)))
   442→    if (dist >= 500) unlockBadge('fire-500')
   443→    if (dist >= 1500) unlockBadge('fire-1500')
   444→    sfx.play('lose')
   445→    setResult({ dist, water: w.water, isRecord })
   446→    setPhase('over')
   447→  }, [addXp, setBest, unlockBadge])
   448→
   449→  /* ================= Boucle principale ================= */
   450→  useEffect(() => {
   451→    const cv = canvasRef.current
   452→    const wrap = wrapRef.current
   453→    if (!cv || !wrap) return
   454→    const ctx = cv.getContext('2d')
   455→    if (!ctx) return
   456→    const w = worldRef.current
   457→
   458→    let bg: HTMLCanvasElement | null = null
   459→    let dpr = 1
   460→
   461→    const resize = () => {
   462→      const cw = wrap.clientWidth
   463→      if (cw < 40) return
   464→      const ch = Math.round(cw * 0.75)
   465→      dpr = Math.min(window.devicePixelRatio || 1, 2)
   466→      w.W = cw
   467→      w.H = ch
   468→      cv.width = Math.round(cw * dpr)
   469→      cv.height = Math.round(ch * dpr)
   470→      cv.style.height = `${ch}px`
   471→      bg = document.createElement('canvas')
   472→      bg.width = cv.width
   473→      bg.height = cv.height
   474→      const bctx = bg.getContext('2d')
   475→      if (bctx) {
   476→        bctx.setTransform(dpr, 0, 0, dpr, 0, 0)
   477→        drawBackdrop(bctx, cw, ch)
   478→      }
   479→    }
   480→    resize()
   481→    resetWorld(w)
   482→
   483→    /* ----- entrées ----- */
   484→    const rectPos = (e: PointerEvent) => {
   485→      const r = cv.getBoundingClientRect()
   486→      return { x: e.clientX - r.left, y: e.clientY - r.top }
   487→    }
   488→    const onDown = (e: PointerEvent) => {
   489→      if (phaseRef.current === 'over') {
   490→        handleStart()
   491→        return
   492→      }
   493→      if (phaseRef.current !== 'playing') return
   494→      const p = rectPos(e)
   495→      w.holding = true
   496→      w.hasPointer = true
   497→      w.targetX = p.x
   498→      cv.setPointerCapture(e.pointerId)
   499→    }
   500→    const onMove = (e: PointerEvent) => {
   501→      if (!w.holding) return
   502→      const p = rectPos(e)
   503→      w.targetX = p.x
   504→    }
   505→    const onUp = () => {
   506→      w.holding = false
   507→      w.hasPointer = false
   508→    }
   509→    const onKeyDown = (e: KeyboardEvent) => {
   510→      if (phaseRef.current !== 'playing') return
   511→      if (e.code === 'Space' || e.code === 'ArrowUp') {
   512→        e.preventDefault()
   513→        w.holding = true
   514→      }
   515→      if (e.code === 'ArrowLeft') { e.preventDefault(); w.keys.left = true }
   516→      if (e.code === 'ArrowRight') { e.preventDefault(); w.keys.right = true }
   517→    }
   518→    const onKeyUp = (e: KeyboardEvent) => {
   519→      if (e.code === 'Space' || e.code === 'ArrowUp') w.holding = false
   520→      if (e.code === 'ArrowLeft') w.keys.left = false
   521→      if (e.code === 'ArrowRight') w.keys.right = false
   522→    }
   523→    const onCtx = (e: Event) => e.preventDefault()
   524→    const onVis = () => { pausedRef.current = document.hidden }
   525→
   526→    cv.addEventListener('pointerdown', onDown)
   527→    cv.addEventListener('pointermove', onMove)
   528→    cv.addEventListener('pointerup', onUp)
   529→    cv.addEventListener('pointercancel', onUp)
   530→    cv.addEventListener('contextmenu', onCtx)
   531→    window.addEventListener('keydown', onKeyDown)
   532→    window.addEventListener('keyup', onKeyUp)
   533→    document.addEventListener('visibilitychange', onVis)
   534→
   535→    /* ----- génération des obstacles ----- */
   536→    const genObstacles = () => {
   537→      const distM = w.scroll / 10
   538→      while (w.nextBandY < w.scroll + w.H * 1.6) {
   539→        const spacing = 350 - Math.min(distM * 0.04, 60)
   540→        const gapW = w.W * clamp(0.46 - distM * 0.00013, 0.26, 0.46)
   541→        let gapX = w.prevGapX + (Math.random() - 0.5) * w.W * 0.55
   542→        gapX = clamp(gapX, w.W * 0.07, w.W * 0.93 - gapW)
   543→        w.prevGapX = gapX
   544→        const band: Band = {
   545→          worldY: w.nextBandY,
   546→          leftLen: Math.max(gapX - 6, 4),
   547→          rightLen: Math.max(w.W - (gapX + gapW) - 6, 4),
   548→          h: 46 + Math.min(distM * 0.012, 30),
   549→          seed: Math.random() * 10,
   550→        }
   551→        w.bands.push(band)
   552→        // Poutre mobile au-delà de 150 m (une fois sur trois)
   553→        if (distM > 150 && w.bands.length % 3 === 0) {
   554→          const bw = w.W * 0.4
   555→          w.beams.push({
   556→            worldY: w.nextBandY + spacing * 0.5,
   557→            w: bw,
   558→            h: 20,
   559→            amp: (w.W - bw) / 2,
   560→            speed: 0.8 + Math.random() * 0.6,
   561→            phase: Math.random() * Math.PI * 2,
   562→          })
   563→        }
   564→        // Goutte d'eau dans le couloir
   565→        if (Math.random() < 0.55) {
   566→          w.drops.push({
   567→            worldY: w.nextBandY + spacing * 0.5,
   568→            x: clamp(gapX + gapW * (0.25 + Math.random() * 0.5), 20, w.W - 20),
   569→            taken: false,
   570→          })
   571→        }
   572→        w.nextBandY += spacing
   573→      }
   574→      // Nettoyage
   575→      const cutoff = w.scroll - 120
   576→      w.bands = w.bands.filter((b) => b.worldY > cutoff)
   577→      w.beams = w.beams.filter((b) => b.worldY > cutoff)
   578→      w.drops = w.drops.filter((d) => d.worldY > cutoff)
   579→    }
   580→
   581→    const beamX = (b: Beam) => (w.W - b.w) / 2 * (1 + Math.sin(w.t * b.speed + b.phase))
   582→
   583→    /* ----- simulation ----- */
   584→    const update = (dt: number) => {
   585→      if (phaseRef.current !== 'playing' || w.dead) return
   586→      const distM = w.scroll / 10
   587→      const scrollSpeed = 150 + Math.min(distM * 0.3, 120)
   588→      w.scroll += scrollSpeed * dt
   589→
   590→      // Physique verticale
   591→      w.pvy += (w.holding ? -2500 : 2000) * dt
   592→      w.pvy = clamp(w.pvy, -470, 560)
   593→      w.py += w.pvy * dt
   594→
   595→      // Direction horizontale
   596→      let steer: number | null = null
   597→      if (w.holding && w.hasPointer) steer = w.targetX
   598→      else if (w.keys.left) steer = w.px - 260 * dt
   599→      else if (w.keys.right) steer = w.px + 260 * dt
   600→      if (steer !== null) {
   601→        const maxStep = 440 * dt
   602→        w.px += clamp(steer - w.px, -maxStep, maxStep)
   603→      }
   604→      w.px = clamp(w.px, 16, w.W - 16)
   605→
   606→      // Plafond : le pompier ne quitte pas l'écran par le haut
   607→      if (w.py < 26) {
   608→        w.py = 26
   609→        if (w.pvy < 0) w.pvy = 0
   610→      }
   611→      // Pompier poussé hors de l'écran par le brasier du bas
   612→      if (w.py > w.H - 14) {
   613→        finishDeath()
   614→        return
   615→      }
   616→
   617→      genObstacles()
   618→
   619→      // Collisions flammes
   620→      const pr = 13
   621→      for (const b of w.bands) {
   622→        const sy = w.H - (b.worldY - w.scroll)
   623→        if (sy < -80 || sy > w.H + 80) continue
   624→        if (Math.abs(w.py - sy) < b.h / 2 + pr * 0.7) {
   625→          if (w.px < b.leftLen + pr * 0.7 || w.px > w.W - b.rightLen - pr * 0.7) {
   626→            finishDeath()
   627→            return
   628→          }
   629→        }
   630→      }
   631→      // Collisions poutres
   632→      for (const b of w.beams) {
   633→        const sy = w.H - (b.worldY - w.scroll)
   634→        if (sy < -40 || sy > w.H + 40) continue
   635→        if (Math.abs(w.py - sy) < b.h / 2 + pr * 0.7) {
   636→          const bx = beamX(b)
   637→          if (w.px > bx - pr * 0.6 && w.px < bx + b.w + pr * 0.6) {
   638→            finishDeath()
   639→            return
   640→          }
   641→        }
   642→      }
   643→      // Gouttes d'eau
   644→      for (const d of w.drops) {
   645→        if (d.taken) continue
   646→        const sy = w.H - (d.worldY - w.scroll)
   647→        if (sy < -30 || sy > w.H + 30) continue
   648→        if (Math.hypot(w.px - d.x, w.py - sy) < 26) {
   649→          d.taken = true
   650→          w.water++
   651→          sfx.play('coin')
   652→          w.pops.push({ x: d.x, y: sy, text: '+50', life: 0.8, max: 0.8, color: '#eafaf1', size: 16 })
   653→          pushHud()
   654→        }
   655→      }
   656→
   657→      // Défi du jour
   658→      const dm = Math.floor(w.scroll / 10)
   659→      if (!dailyDoneRef.current && dm >= dailyTarget && dm > 0) {
   660→        completeDaily(gameId)
   661→        sfx.play('win')
   662→        setConfetti((k) => k + 1)
   663→        w.pops.push({
   664→          x: w.W / 2, y: w.H * 0.3,
   665→          text: `🏆 ${L[langRef.current].dailyDone}`,
   666→          life: 1.6, max: 1.6, color: '#f6c26a', size: 18,
   667→        })
   668→      }
   669→
   670→      // Braises
   671→      w.emberT -= dt
   672→      if (w.emberT <= 0 && !reduceRef.current) {
   673→        w.emberT = 0.09
   674→        w.embers.push({
   675→          x: Math.random() * w.W,
   676→          y: w.H + 6,
   677→          vy: -(30 + Math.random() * 55),
   678→          r: 1.5 + Math.random() * 2.5,
   679→          life: 2.2, max: 2.2,
   680→        })
   681→      }
   682→      for (let i = w.embers.length - 1; i >= 0; i--) {
   683→        const e = w.embers[i]
   684→        e.y += e.vy * dt
   685→        e.x += Math.sin(w.t * 3 + e.r * 8) * 12 * dt
   686→        e.life -= dt
   687→        if (e.life <= 0) w.embers.splice(i, 1)
   688→      }
   689→      for (let i = w.pops.length - 1; i >= 0; i--) {
   690→        const p = w.pops[i]
   691→        p.y -= 30 * dt
   692→        p.life -= dt
   693→        if (p.life <= 0) w.pops.splice(i, 1)
   694→      }
   695→
   696→      const newDist = Math.floor(w.scroll / 10)
   697→      if (newDist !== w.distM) {
   698→        w.distM = newDist
   699→        pushHud()
   700→      }
   701→    }
   702→
   703→    /* ----- rendu ----- */
   704→    const draw = () => {
   705→      const W = w.W, H = w.H
   706→      ctx.setTransform(dpr, 0, 0, dpr, 0, 0)
   707→      ctx.clearRect(0, 0, W, H)
   708→      if (bg) ctx.drawImage(bg, 0, 0, W, H)
   709→      drawBricks(ctx, W, H, w.scroll)
   710→
   711→      // Gouttes d'eau
   712→      ctx.font = '20px "Apple Color Emoji","Segoe UI Emoji","Noto Color Emoji",system-ui,sans-serif'
   713→      ctx.textAlign = 'center'
   714→      ctx.textBaseline = 'middle'
   715→      for (const d of w.drops) {
   716→        if (d.taken) continue
   717→        const sy = H - (d.worldY - w.scroll)
   718→        if (sy < -30 || sy > H + 30) continue
   719→        ctx.fillText('💧', d.x, sy)
   720→      }
   721→
   722→      // Poutres
   723→      for (const b of w.beams) {
   724→        const sy = H - (b.worldY - w.scroll)
   725→        if (sy < -60 || sy > H + 60) continue
   726→        const bx = beamX(b)
   727→        const g = ctx.createLinearGradient(bx, sy - b.h / 2, bx + b.w, sy + b.h / 2)
   728→        g.addColorStop(0, '#5c3320')
   729→        g.addColorStop(0.5, '#7a4527')
   730→        g.addColorStop(1, '#5c3320')
   731→        ctx.fillStyle = g
   732→        rr(ctx, bx, sy - b.h / 2, b.w, b.h, 5)
   733→        ctx.fill()
   734→        ctx.fillStyle = 'rgba(246,194,106,0.5)'
   735→        rr(ctx, bx + 4, sy - b.h / 2 + 2, b.w - 8, 3, 1.5)
   736→        ctx.fill()
   737→      }
   738→
   739→      // Flammes
   740→      for (const b of w.bands) {
   741→        const sy = H - (b.worldY - w.scroll)
   742→        if (sy < -H * 0.4 || sy > H + 120) continue
   743→        drawFlames(ctx, W, sy, b.leftLen, b.rightLen, b.h, w.t, b.seed, reduceRef.current)
   744→      }
   745→
   746→      // Brasier du bas
   747→      const fireY = H - 6
   748→      const fg = ctx.createLinearGradient(0, H - 46, 0, H)
   749→      fg.addColorStop(0, 'rgba(224,112,44,0)')
   750→      fg.addColorStop(0.5, 'rgba(224,112,44,0.55)')
   751→      fg.addColorStop(1, 'rgba(168,50,38,0.9)')
   752→      ctx.fillStyle = fg
   753→      ctx.fillRect(0, H - 46, W, 46)
   754→
   755→      // Braises
   756→      for (const e of w.embers) {
   757→        ctx.globalAlpha = clamp(e.life / e.max, 0, 1) * 0.8
   758→        ctx.fillStyle = '#f6c26a'
   759→        ctx.beginPath()
   760→        ctx.arc(e.x, e.y, e.r, 0, Math.PI * 2)
   761→        ctx.fill()
   762→      }
   763→      ctx.globalAlpha = 1
   764→
   765→      // Pompier
   766→      drawFirefighter(ctx, w.px, w.py, w.pvy, w.holding, w.t, reduceRef.current)
   767→
   768→      // Textes flottants
   769→      for (const p of w.pops) {
   770→        ctx.globalAlpha = clamp(p.life / p.max, 0, 1)
   771→        ctx.fillStyle = p.color
   772→        ctx.font = `800 ${p.size}px "Nunito", system-ui, sans-serif`
   773→        ctx.fillText(p.text, p.x, p.y)
   774→      }
   775→      ctx.globalAlpha = 1
   776→    }
   777→
   778→    let last = performance.now()
   779→    const step = (now: number) => {
   780→      rafRef.current = requestAnimationFrame(step)
   781→      const dt = Math.min((now - last) / 1000, 0.05)
   782→      last = now
   783→      if (pausedRef.current) return
   784→      w.t += dt
   785→      update(dt)
   786→      draw()
   787→    }
   788→    rafRef.current = requestAnimationFrame(step)
   789→
   790→    const ro = new ResizeObserver(resize)
   791→    ro.observe(wrap)
   792→
   793→    return () => {
   794→      cancelAnimationFrame(rafRef.current)
   795→      ro.disconnect()
   796→      cv.removeEventListener('pointerdown', onDown)
   797→      cv.removeEventListener('pointermove', onMove)
   798→      cv.removeEventListener('pointerup', onUp)
   799→      cv.removeEventListener('pointercancel', onUp)
   800→      cv.removeEventListener('contextmenu', onCtx)
   801→      window.removeEventListener('keydown', onKeyDown)
   802→      window.removeEventListener('keyup', onKeyUp)
   803→      document.removeEventListener('visibilitychange', onVis)
   804→    }
   805→  }, [dailyTarget, finishDeath, handleStart, pushHud, completeDaily])
   806→
   807→  /* ================= Rendu React ================= */
   808→  return (
   809→    <div className="relative select-none">
   810→      <Confetti fire={confetti} count={reduce ? 0 : 90} />
   811→
   812→      {/* Barre d'état */}
   813→      <div className="mb-3 flex flex-wrap items-center gap-2">
   814→        {phase === 'playing' ? (
   815→          <>
   816→            <StatChip icon="📏" label={T.distance} value={`${hud.dist} ${T.meters}`} />
   817→            <StatChip icon="💧" label={T.water} value={`+${hud.water * 50}`} />
   818→            {!dailyDone ? (
   819→              <span className="flex h-11 items-center gap-1.5 rounded-full bg-amber-100 px-3.5 text-sm font-bold text-amber-700 dark:bg-amber-500/20 dark:text-amber-300">
   820→                🎯 {T.daily.replace('{n}', String(dailyTarget))}
   821→              </span>
   822→            ) : null}
   823→          </>
   824→        ) : (
   825→          <>
   826→            {hydrated && bestDist > 0 ? <StatChip icon="📏" label={`${T.distance} — ${T.best}`} value={`${bestDist} ${T.meters}`} /> : null}
   827→            {!dailyDone ? (
   828→              <span className="flex h-11 items-center gap-1.5 rounded-full bg-amber-100 px-3.5 text-sm font-bold text-amber-700 dark:bg-amber-500/20 dark:text-amber-300">
   829→                🎯 {T.daily.replace('{n}', String(dailyTarget))}
   830→              </span>
   831→            ) : null}
   832→          </>
   833→        )}
   834→      </div>
   835→
   836→      {/* Canvas + habillage */}
   837→      <div ref={wrapRef} className="relative w-full overflow-hidden rounded-2xl border shadow-inner">
   838→        <canvas
   839→          ref={canvasRef}
   840→          className="block w-full touch-none"
   841→          style={{ touchAction: 'none' }}
   842→          aria-label="Fire Escape"
   843→        />
   844→
   845→        {/* Écran d'accueil */}
   846→        {phase === 'idle' ? (
   847→          <div className="absolute inset-0 z-20 flex flex-col items-center justify-center gap-4 bg-stone-900/45 p-6 text-center backdrop-blur-[2px]">
   848→            <div className="flex gap-2" aria-hidden="true">
   849→              {['🧑‍🚒', '🔥', '💧'].map((e, i) => (
   850→                <span key={i} className="text-4xl anim-floaty" style={{ animationDelay: `${i * 0.4}s` }}>{e}</span>
   851→              ))}
   852→            </div>
   853→            <h3 className="font-display text-2xl font-extrabold text-white drop-shadow">🚒 {T.title}</h3>
   854→            <p className="max-w-sm text-sm text-stone-100">{content.tagline[lang]}</p>
   855→            <p className="max-w-xs rounded-xl bg-white/15 px-3 py-2 text-xs text-stone-100">{T.hint}</p>
   856→            <Button size="lg" className="h-12 rounded-full px-7 text-base anim-glow" onClick={handleStart}>
   857→              <Play className="h-5 w-5" aria-hidden="true" /> {T.play}
   858→            </Button>
   859→          </div>
   860→        ) : null}
   861→
   862→        {/* Game over */}
   863→        {phase === 'over' && result ? (
   864→          <div className="absolute inset-0 z-20 flex items-center justify-center bg-stone-900/60 p-4 backdrop-blur-[2px]">
   865→            <div className="anim-rise w-full max-w-sm rounded-2xl border bg-card p-6 text-center shadow-xl">
   866→              <p className="font-display text-xl font-extrabold">🔥 {T.over}</p>
   867→              <div className="mt-4 flex items-end justify-center gap-6">
   868→                <div>
   869→                  <p className="text-4xl font-extrabold tabular-nums text-primary">{result.dist}</p>
   870→                  <p className="flex items-center justify-center gap-1 text-xs text-muted-foreground">
   871→                    <Ruler className="h-3.5 w-3.5" aria-hidden="true" /> {T.distance} ({T.meters})
   872→                  </p>
   873→                </div>
   874→                <div>
   875→                  <p className="text-4xl font-extrabold tabular-nums text-emerald-600 dark:text-emerald-400">{result.water * 50}</p>
   876→                  <p className="flex items-center justify-center gap-1 text-xs text-muted-foreground">
   877→                    <Droplets className="h-3.5 w-3.5" aria-hidden="true" /> {T.waterBonus}
   878→                  </p>
   879→                </div>
   880→              </div>
   881→              {result.isRecord ? (
   882→                <p className="mt-3 text-sm font-bold text-amber-600 dark:text-amber-400">🏆 {T.newBest}</p>
   883→              ) : (
   884→                <p className="mt-3 text-sm text-muted-foreground">
   885→                  {T.best} : <span className="font-bold tabular-nums">{Math.max(bestDist, result.dist)} {T.meters}</span>
   886→                </p>
   887→              )}
   888→              <p className="mt-2 text-xs text-muted-foreground">{T.tip}</p>
   889→              <div className="mt-4 flex justify-center">
   890→                <Button size="lg" className="h-12 rounded-full px-8" onClick={handleStart}>
   891→                  <RotateCcw className="h-5 w-5" aria-hidden="true" /> {T.replay}
   892→                </Button>
   893→              </div>
   894→            </div>
   895→          </div>
   896→        ) : null}
   897→      </div>
   898→    </div>
   899→  )
   900→}
   901→
