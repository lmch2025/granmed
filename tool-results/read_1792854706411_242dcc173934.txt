   168→   168→function wordKey(w: CrosswordWord): string {

   176→   176→/** Mot entièrement rempli avec les bonnes lettres ? */
   177→   177→function isWordComplete(w: CrosswordWord, l: Record<string, string>): boolean {
   178→   178→  for (let i = 0; i < w.answer.length; i++) {
   179→   179→    const r = w.row + (w.dir === 'down' ? i : 0)
   180→   180→    const c = w.col + (w.dir === 'across' ? i : 0)
   181→   181→    if (l[cellKey(r, c)] !== w.answer[i]) return false
   182→   182→  }
   183→   183→  return true
   184→   184→}

   186→   186→/** A-Z après suppression des diacritiques (É→E, à→A…) ; null sinon. */
   187→   187→function normalizeLetter(k: string): string | null {
   188→   188→  // NFD puis retrait des marques diacritiques combinantes (U+0300-U+036F)
   189→   189→  const base = k
   190→   190→    .normalize('NFD')
   191→   191→    .split('')
   192→   192→    .filter((ch) => {
   193→   193→      const code = ch.charCodeAt(0)
   194→   194→      return code < 0x300 || code > 0x36f
   195→   195→    })
   196→   196→    .join('')
   197→   197→  return /^[a-zA-Z]$/.test(base) ? base.toUpperCase() : null
   198→   198→}
