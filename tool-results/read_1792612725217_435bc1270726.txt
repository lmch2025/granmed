   193→/** Taille des fioles relative au canevas (même difficulté sur mobile et desktop). */
   194→function jarWidthFor(W: number, H: number, round: number): number {
   195→  const s = Math.min(W, H * 1.45)
   196→  return clamp(s * (0.205 - (round - 1) * 0.012), 30, 118)
   197→}
   198→function jarHeightFor(jw: number, H: number, rows: number): number {
   199→  // Une seule étagère : fioles hautes autorisées ; plusieurs : plafonné pour
   200→  // que les rangées voisines ne se touchent jamais.
   201→  const capY = rows <= 1 ? H * 0.26 : H * 0.175
   202→  return Math.max(Math.min(jw * 1.3, capY), 20)
   203→}

   219→function startRound(w: World, round: number) {
   220→  w.round = round
   221→  w.rows = Math.min(1 + Math.floor((round - 1) / 2), 3)
   222→  const perRow = Math.min(2 + Math.ceil(round / 2), 4)
   223→  // Jamais plus de 10 fioles : rester « nettoyable » avec les 10 pilules réglementaires.
   224→  const total = Math.min(w.rows * perRow, 10)
   225→  const base = Math.floor(total / w.rows)
   226→  const extra = total % w.rows
   227→  const jw = jarWidthFor(w.W, w.H, round)
   228→  const jh = jarHeightFor(jw, w.H, w.rows)
   229→  const baseSpd = 34 + round * 8
   230→  w.jars = []
   231→  for (let r = 0; r < w.rows; r++) {
   232→    const count = base + (r < extra ? 1 : 0)
   233→    const lane = Math.max(w.W - jw - 24, 10) / count
   234→    for (let i = 0; i < count; i++) {
   235→      const mult = 0.88 + 0.12 * ((r * 2 + i) % 3)
   236→      w.jars.push({
   237→        rowIdx: r,
   238→        x: 12 + jw / 2 + lane * (i + 0.5),
   239→        vx: ((r + i) % 2 === 0 ? 1 : -1) * Math.min(baseSpd * mult, 170),
   240→        w: jw, h: jh,
   241→        alive: true, fade: 0,
   242→        glass: (r + i) % 3,
   243→      })
   244→    }
   245→  }
   246→  w.shelfWob = [0, 0, 0]
   247→  w.pill = { x: 0, y: 0, vx: 0, vy: 0, rot: 0, vr: 0, state: 'ready' }
   248→  w.pillAcc = 0
   249→  w.pillsLeft = 10
   250→  w.streak = 0
   251→  w.roundScore = 0
   252→  w.roundHits = 0
   253→  w.bonusGiven = 0
   254→  w.shards = []
   255→  w.pops = []
   256→  w.endTimer = 0
   257→  w.cleared = false
   258→  w.hudDirty = true
   259→}

   261→/** Haptique légère, sans effet sur les appareils sans vibreur. */
   262→function buzz(pattern: number | number[]) {
   263→  try {
   264→    navigator.vibrate?.(pattern)
   265→  } catch {
   266→    /* no-op */
   267→  }

   271→function rr(ctx: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number) {
   272→  const rad = Math.min(r, w / 2, h / 2)
   273→  ctx.beginPath()
   274→  ctx.moveTo(x + rad, y)
   275→  ctx.arcTo(x + w, y, x + w, y + h, rad)
   276→  ctx.arcTo(x + w, y + h, x, y + h, rad)
   277→  ctx.arcTo(x, y + h, x, y, rad)
   278→  ctx.arcTo(x, y, x + w, y, rad)
   279→  ctx.closePath()
   280→}

   282→function drawBackdrop(ctx: CanvasRenderingContext2D, W: number, H: number) {
   283→  // Mur chaleureux
   284→  const wall = ctx.createLinearGradient(0, 0, 0, H)
   285→  wall.addColorStop(0, '#fdf1de')
   286→  wall.addColorStop(0.55, '#f7e3c6')
   287→  wall.addColorStop(1, '#efd0aa')
   288→  ctx.fillStyle = wall
   289→  ctx.fillRect(0, 0, W, H)
   290→  // Rayures horizontales discrètes
   291→  ctx.fillStyle = 'rgba(214,164,110,0.10)'
   292→  for (let y = 0; y < H; y += 46) ctx.fillRect(0, y, W, 20)
   293→  // Lumière chaude
   294→  const glow = ctx.createRadialGradient(W * 0.5, -H * 0.15, 10, W * 0.5, -H * 0.15, H * 0.95)
   295→  glow.addColorStop(0, 'rgba(255,246,224,0.85)')
   296→  glow.addColorStop(1, 'rgba(255,246,224,0)')
   297→  ctx.fillStyle = glow
   298→  ctx.fillRect(0, 0, W, H)
   299→  // Croix de pharmacie (émeraude)
   300→  const s = Math.max(26, W * 0.055)
   301→  const cx = W - s * 1.7, cy = s * 1.4
   302→  ctx.fillStyle = '#2f9c74'
   303→  rr(ctx, cx - s / 2, cy - s / 2, s, s, s * 0.24)
   304→  ctx.fill()
   305→  ctx.fillStyle = '#f8fffa'
   306→  const t = s * 0.2
   307→  ctx.fillRect(cx - t / 2, cy - s * 0.32, t, s * 0.64)
   308→  ctx.fillRect(cx - s * 0.32, cy - t / 2, s * 0.64, t)
   309→  // Plinthe + sol
   310→  ctx.fillStyle = '#d9b183'
   311→  ctx.fillRect(0, H - 14, W, 14)
   312→  ctx.fillStyle = 'rgba(120,80,44,0.25)'
   313→  ctx.fillRect(0, H - 15, W, 2)
   314→}

   316→function drawShelf(ctx: CanvasRenderingContext2D, y: number, W: number, H: number) {
   317→  const th = clamp(H * 0.028, 8, 14)
   318→  const grad = ctx.createLinearGradient(0, y, 0, y + th)
   319→  grad.addColorStop(0, '#b07a48')
   320→  grad.addColorStop(1, '#84552e')
   321→  ctx.fillStyle = grad
   322→  rr(ctx, 6, y, W - 12, th, 3)
   323→  ctx.fill()
   324→  ctx.fillStyle = 'rgba(255,240,214,0.35)'
   325→  ctx.fillRect(10, y + 1.5, W - 20, 1.5)
   326→  // Équerres
   327→  ctx.fillStyle = 'rgba(90,58,30,0.55)'
   328→  for (let i = 0; i < 3; i++) {
   329→    const bx = 26 + (i * (W - 52)) / 2
   330→    ctx.beginPath()
   331→    ctx.moveTo(bx, y + th)
   332→    ctx.lineTo(bx + 12, y + th)
   333→    ctx.lineTo(bx, y + th + 10)
   334→    ctx.closePath()
   335→    ctx.fill()
   336→  }
   337→}

   345→function drawJar(ctx: CanvasRenderingContext2D, x: number, yCenter: number, j: Jar, alpha: number) {
   346→  const { w, h } = j
   347→  const [c1, c2] = JAR_GLASS[j.glass]
   348→  ctx.save()
   349→  ctx.globalAlpha = alpha
   350→  // Corps
   351→  const grad = ctx.createLinearGradient(x - w / 2, yCenter - h / 2, x + w / 2, yCenter + h / 2)
   352→  grad.addColorStop(0, c1)
   353→  grad.addColorStop(1, c2)
   354→  ctx.fillStyle = grad
   355→  rr(ctx, x - w / 2, yCenter - h / 2, w, h, 7)
   356→  ctx.fill()
   357→  ctx.strokeStyle = 'rgba(122,74,34,0.45)'
   358→  ctx.lineWidth = 1.5
   359→  rr(ctx, x - w / 2, yCenter - h / 2, w, h, 7)
   360→  ctx.stroke()
   361→  // Col + bouchon
   362→  ctx.fillStyle = 'rgba(122,74,34,0.28)'
   363→  rr(ctx, x - w * 0.2, yCenter - h / 2 - h * 0.1, w * 0.4, h * 0.14, 3)
   364→  ctx.fill()
   365→  ctx.fillStyle = '#c05b3f'
   366→  rr(ctx, x - w * 0.24, yCenter - h / 2 - h * 0.19, w * 0.48, h * 0.1, 4)
   367→  ctx.fill()
   368→  // Reflet
   369→  ctx.fillStyle = 'rgba(255,248,232,0.4)'
   370→  rr(ctx, x - w * 0.36, yCenter - h * 0.32, w * 0.14, h * 0.6, 4)
   371→  ctx.fill()
   372→  // Étiquette + signe plus
   373→  ctx.fillStyle = '#fdf6e9'
   374→  rr(ctx, x - w * 0.26, yCenter - h * 0.12, w * 0.52, h * 0.34, 4)
   375→  ctx.fill()
   376→  ctx.fillStyle = '#c05b3f'
   377→  const lt = Math.max(2.5, w * 0.07)
   378→  ctx.fillRect(x - lt / 2, yCenter - h * 0.04 - lt * 0.9, lt, lt * 1.8)
   379→  ctx.fillRect(x - lt * 0.9, yCenter - h * 0.04 - lt / 2, lt * 1.8, lt)
   380→  ctx.restore()
   381→}

   383→function drawPill(ctx: CanvasRenderingContext2D, x: number, y: number, rot: number, hl: number) {
   384→  const hh = hl * 0.54
   385→  ctx.save()
   386→  ctx.translate(x, y)
   387→  ctx.rotate(rot)
   388→  rr(ctx, -hl, -hh, hl * 2, hh * 2, hh)
   389→  ctx.fillStyle = '#f8f1e4'
   390→  ctx.fill()
   391→  ctx.save()
   392→  rr(ctx, -hl, -hh, hl * 2, hh * 2, hh)
   393→  ctx.clip()
   394→  ctx.fillStyle = '#e2725b'
   395→  ctx.fillRect(-hl, -hh, hl, hh * 2)
   396→  ctx.restore()
   397→  ctx.strokeStyle = '#b34d3a'
   398→  ctx.lineWidth = Math.max(1.2, hl * 0.11)
   399→  rr(ctx, -hl, -hh, hl * 2, hh * 2, hh)
   400→  ctx.stroke()
   401→  ctx.restore()
   402→}

   498→    w.maxStreakSession = 0

   500→    setSummary(null)
   501→    setPhase('playing')
   502→    pushHud()
   503→  }, [pushHud])
   504→
   505→  const handleEnd = useCallback(() => {
   506→    sfx.play('click')
   507→    setPhase('done')
   508→  }, [])
   509→
   510→  /* ---------- boucle principale ---------- */
   511→  useEffect(() => {
   512→    const cv = canvasRef.current
   513→    const wrap = wrapRef.current
   514→    if (!cv || !wrap) return
   515→    const ctx = cv.getContext('2d')
   516→    if (!ctx) return
   517→    const w = worldRef.current
   518→
   519→    let bg: HTMLCanvasElement | null = null
   520→    let dpr = 1
   521→    let aimId = -1
   522→
   523→    const pillHL = () => clamp(Math.min(w.W, w.H * 1.45) * 0.037, 10, 15)
   524→    // Lance-pierre centré en bas : toute la puissance est atteignable EN
   525→    // RESTANT dans le canevas (traction vers le bas / bas-gauche / bas-droite).
   526→    const anchor = () => ({ x: w.W * 0.5, y: w.H * 0.72 })
   527→    const gravity = () => w.H * 3.2
   528→    const shelfY = (r: number) => {
   529→      if (w.rows === 1) return 0.34 * w.H
   530→      if (w.rows === 2) return (0.26 + r * 0.32) * w.H
   531→      return (0.22 + r * 0.19) * w.H
   532→    }
   533→    const jarCY = (j: Jar) => shelfY(j.rowIdx) - j.h / 2 - 5
   534→
   535→    const resize = () => {
   536→      const cw = wrap.clientWidth
   537→      const ch = wrap.clientHeight
   538→      if (cw < 40 || ch < 40) return
   539→      dpr = Math.min(window.devicePixelRatio || 1, 2)
   540→      w.W = cw
   541→      w.H = ch
   542→      cv.width = Math.round(cw * dpr)
   543→      cv.height = Math.round(ch * dpr)
   544→      // Les fioles vivantes suivent la nouvelle géométrie (rotation, redimensionnement).
   545→      const jw = jarWidthFor(cw, ch, w.round)
   546→      const jh = jarHeightFor(jw, ch, w.rows)
   547→      for (const j of w.jars) {
   548→        j.w = jw
   549→        j.h = jh
   550→        j.x = clamp(j.x, 12 + jw / 2, Math.max(cw - 12 - jw / 2, 12 + jw / 2))
   551→      }
   552→      bg = document.createElement('canvas')
   553→      bg.width = cv.width
   554→      bg.height = cv.height
   555→      const bctx = bg.getContext('2d')
   556→      if (bctx) {
   557→        bctx.setTransform(dpr, 0, 0, dpr, 0, 0)
   558→        drawBackdrop(bctx, cw, ch)
   559→      }
   560→    }
   561→    resize()
   562→    startRound(w, 1) // fioles de démonstration derrière l'écran d'accueil
   563→
   564→    /* ----- entrées pointeur (un seul doigt ne pilote la visée) ----- */
   565→    const rectPos = (e: PointerEvent) => {
   566→      const r = cv.getBoundingClientRect()
   567→      return { x: e.clientX - r.left, y: e.clientY - r.top }
   568→    }
   569→    const onDown = (e: PointerEvent) => {
   570→      if (phaseRef.current !== 'playing') return
   571→      if (e.button !== 0) return
   572→      if (w.pill.state !== 'ready' || w.pillsLeft <= 0 || w.endTimer > 0) return
   573→      const p = rectPos(e)
   574→      const a = anchor()
   575→      const grab = Math.max(100, w.H * 0.3)
   576→      if (Math.hypot(p.x - a.x, p.y - a.y) < grab) {
   577→        w.pill.state = 'aiming'
   578→        w.aim = { x: clamp(p.x, 4, w.W - 4), y: clamp(p.y, 4, w.H - 4) }
   579→        aimId = e.pointerId
   580→        try {
   581→          cv.setPointerCapture(e.pointerId)
   582→        } catch {
   583→          /* capture best-effort */
   584→        }
   585→      }
   586→    }
   587→    const onMove = (e: PointerEvent) => {
   588→      if (w.pill.state !== 'aiming' || e.pointerId !== aimId) return
   589→      const p = rectPos(e)
   590→      w.aim = { x: clamp(p.x, 4, w.W - 4), y: clamp(p.y, 4, w.H - 4) }
   591→    }
   592→    const pullVec = () => {
   593→      const a = anchor()
   594→      let dx = w.aim.x - a.x
   595→      let dy = w.aim.y - a.y
   596→      const len = Math.hypot(dx, dy)
   597→      const maxPull = w.H * 0.3
   598→      if (len > maxPull) {
   599→        dx *= maxPull / len
   600→        dy *= maxPull / len
   601→      }
   602→      return { dx, dy }
   603→    }
   604→    const launchVec = (px: number, py: number) => {
   605→      const a = anchor()
   606→      return { vx: (a.x - px) * 9, vy: (a.y - py) * 9 }
   607→    }
   608→    const onUp = (e: PointerEvent) => {
   609→      if (w.pill.state !== 'aiming' || e.pointerId !== aimId) return
   610→      aimId = -1
   611→      const a = anchor()
   612→      const { dx, dy } = pullVec()
   613→      if (Math.hypot(dx, dy) < 14) {
   614→        w.pill.state = 'ready'
   615→        return
   616→      }
   617→      const px = a.x + dx
   618→      const py = a.y + dy
   619→      const { vx, vy } = launchVec(px, py)
   620→      w.pill.x = px
   621→      w.pill.y = py
   622→      w.pill.vx = vx
   623→      w.pill.vy = vy
   624→      w.pill.rot = Math.atan2(vy, vx)
   625→      w.pill.vr = 5
   626→      w.pill.state = 'flying'
   627→      w.pillAcc = 0
   628→      w.pillsLeft--
   629→      sfx.play('swish')
   630→      buzz(10)
   631→      w.hudDirty = true
   632→    }
   633→    const cancelAim = (e: PointerEvent) => {
   634→      if (e.pointerId !== aimId) return
   635→      aimId = -1
   636→      if (w.pill.state === 'aiming') w.pill.state = 'ready'
   637→    }
   638→    const onCtx = (e: Event) => e.preventDefault()
   639→    cv.addEventListener('pointerdown', onDown)
   640→    cv.addEventListener('pointermove', onMove)
   641→    cv.addEventListener('pointerup', onUp)
   642→    cv.addEventListener('pointercancel', cancelAim)
   643→    cv.addEventListener('lostpointercapture', cancelAim)
   644→    cv.addEventListener('contextmenu', onCtx)
   645→
   646→    /* ----- pause quand l'onglet est masqué ----- */
   647→    const onVis = () => { pausedRef.current = document.hidden }
   648→    document.addEventListener('visibilitychange', onVis)
   649→
   650→    /* ----- résolution d'un lancer ----- */
   651→    const resolvePill = (hit: boolean) => {
   652→      if (!hit) {
   653→        w.streak = 0
   654→        sfx.play('wrong', { volume: 0.3 })
   655→      }
   656→      w.pill.state = 'ready'
   657→      w.hudDirty = true
   658→      if (w.jars.every((j) => !j.alive)) {
   659→        if (!w.cleared) {
   660→          w.cleared = true
   661→          const bonus = w.pillsLeft * 150
   662→          w.bonusGiven = bonus
   663→          w.roundScore += bonus
   664→          w.totalScore += bonus
   665→          w.pops.push({
   666→            x: w.W / 2, y: w.H * 0.42,
   667→            text: `${L[langRef.current].cleared} +${bonus}`,
   668→            life: 1.4, max: 1.4, color: '#b45309', size: 20,
   669→          })
   670→          sfx.play('win')
   671→          buzz(20)
   672→          setConfetti((k) => k + 1)
   673→          w.endTimer = 1.15
   674→        }
   675→      } else if (w.pillsLeft <= 0) {
   676→        w.endTimer = 0.7
   677→      } else if (hit) {
   678→        // pilule suivante prête (uniquement après un touché — un raté a déjà
   679→        // son feedback « wrong »)
   680→        sfx.play('pop', { volume: 0.35 })
   681→      }
   682→    }
   683→
   684→    /* ----- simulation ----- */
   685→    /** Fioles : glissement physique avec rebond sur les bords et échange de
   686→     *  vitesses au contact (choc élastique 1D) — elles ne se traversent jamais. */
   687→    const moveJars = (dt: number) => {
   688→      for (let r = 0; r < w.rows; r++) {
   689→        const row: Jar[] = []
   690→        for (const j of w.jars) if (j.alive && j.rowIdx === r) row.push(j)
   691→        if (row.length === 0) continue
   692→        for (const j of row) {
   693→          j.x += j.vx * dt
   694→          const xMin = 12 + j.w / 2
   695→          const xMax = Math.max(w.W - 12 - j.w / 2, xMin)
   696→          if (j.x < xMin) {
   697→            j.x = xMin
   698→            j.vx = Math.abs(j.vx)
   699→          } else if (j.x > xMax) {
   700→            j.x = xMax
   701→            j.vx = -Math.abs(j.vx)
   702→          }
   703→        }
   704→        row.sort((p, q) => p.x - q.x)
   705→        for (let i = 0; i < row.length - 1; i++) {
   706→          const p = row[i]
   707→          const q = row[i + 1]
   708→          const minGap = (p.w + q.w) / 2 + 2
   709→          if (q.x - p.x < minGap && p.vx > q.vx) {
   710→            const tv = p.vx
   711→            p.vx = q.vx
   712→            q.vx = tv
   713→          }
   714→        }
   715→      }
   716→    }
   717→
   718→    /* ----- projectile : un pas fixe d'intégration (retourne true si résolu) ----- */
   719→    const integrate = () => {
   720→      const p = w.pill
   721→      p.vy += gravity() * STEP
   722→      p.x += p.vx * STEP
   723→      p.y += p.vy * STEP
   724→      p.rot += p.vr * STEP
   725→      const hl = pillHL()
   726→      for (const j of w.jars) {
   727→        if (!j.alive) continue
   728→        const jx = j.x
   729→        const jy = jarCY(j)
   730→        if (Math.abs(p.x - jx) < j.w / 2 + hl && Math.abs(p.y - jy) < j.h / 2 + hl * 0.75) {
   731→          j.alive = false
   732→          j.fade = 0.45
   733→          w.shelfWob[j.rowIdx] = 1
   734→          w.streak++
   735→          w.maxStreakSession = Math.max(w.maxStreakSession, w.streak)
   736→          const pts = 100 + (w.streak - 1) * 50
   737→          w.roundScore += pts
   738→          w.totalScore += pts
   739→          w.roundHits++
   740→          sfx.play('coin')
   741→          buzz(10)
   742→          const colors = ['#f6e3c2', '#e8b56b', '#d98a4f', '#e2725b']
   743→          if (!reduceRef.current) {
   744→            for (let i = 0; i < 16; i++) {
   745→              const ang = Math.random() * Math.PI * 2
   746→              const sp = 90 + Math.random() * 230
   747→              w.shards.push({
   748→                x: jx, y: jy,
   749→                vx: Math.cos(ang) * sp, vy: Math.sin(ang) * sp - 130,
   750→                life: 0.45 + Math.random() * 0.35, max: 0.8,
   751→                size: 2.5 + Math.random() * 3.5,
   752→                color: colors[i % colors.length],
   753→              })
   754→            }
   755→          }
   756→          w.pops.push({
   757→            x: jx, y: jy - 14,
   758→            text: `+${pts}${w.streak >= 3 ? ` ×${w.streak}` : ''}`,
   759→            life: 0.9, max: 0.9,
   760→            color: w.streak >= 3 ? '#b45309' : '#7c4a1e',
   761→            size: w.streak >= 3 ? 19 : 16,
   762→          })
   763→          if (w.streak >= 8) unlockBadge('toss-combo-8')
   764→          w.hudDirty = true
   765→          resolvePill(true)
   766→          return true
   767→        }
   768→      }
   769→      if (p.y > w.H + 60 || p.x < -90 || p.x > w.W + 90) {
   770→        resolvePill(false)
   771→        return true
   772→      }
   773→      return false
   774→    }
   775→
   776→    const update = (dt: number) => {
   777→      moveJars(dt)
   778→      for (let i = 0; i < 3; i++) w.shelfWob[i] *= Math.exp(-5 * dt)
   779→      for (const j of w.jars) if (!j.alive && j.fade > 0) j.fade -= dt
   780→
   781→      if (phaseRef.current === 'playing' && w.pill.state === 'flying') {
   782→        w.pillAcc += dt
   783→        let guard = 0
   784→        while (w.pillAcc >= STEP && w.pill.state === 'flying' && guard < 24) {
   785→          w.pillAcc -= STEP
   786→          guard++
   787→          if (integrate()) break
   788→        }
   789→      }
   790→
   791→      for (let i = w.shards.length - 1; i >= 0; i--) {
   792→        const s = w.shards[i]
   793→        s.vy += 950 * dt
   794→        s.x += s.vx * dt
   795→        s.y += s.vy * dt
   796→        s.life -= dt
   797→        if (s.life <= 0) w.shards.splice(i, 1)
   798→      }
   799→      for (let i = w.pops.length - 1; i >= 0; i--) {
   800→        const p = w.pops[i]
   801→        p.y -= 34 * dt
   802→        p.life -= dt
   803→        if (p.life <= 0) w.pops.splice(i, 1)
   804→      }
   805→      if (w.endTimer > 0) {
   806→        w.endTimer -= dt
   807→        if (w.endTimer <= 0) {
   808→          w.endTimer = 0
   809→          finishRound()
   810→        }
   811→      }
   812→    }
   813→
   814→    /* ----- rendu ----- */
   815→    const draw = () => {
   816→      const W = w.W, H = w.H
   817→      const hl = pillHL()
   818→      ctx.setTransform(dpr, 0, 0, dpr, 0, 0)
   819→      ctx.clearRect(0, 0, W, H)
   820→      if (bg) ctx.drawImage(bg, 0, 0, W, H)
   821→
   822→      // Étagères
   823→      for (let r = 0; r < w.rows; r++) {
   824→        drawShelf(ctx, shelfY(r) + Math.sin(w.t * 26) * 3 * w.shelfWob[r], W, H)
   825→      }
   826→      // Fioles
   827→      for (const j of w.jars) {
   828→        if (!j.alive && j.fade <= 0) continue
   829→        drawJar(ctx, j.x, jarCY(j), j, j.alive ? 1 : clamp(j.fade / 0.45, 0, 1))
   830→      }
   831→
   832→      // Lance-pierre (socle + pied jusqu'au sol)
   833→      const a = anchor()
   834→      const fw = hl
   835→      const fh = hl * 1.8
   836→      const restY = a.y - fh
   837→      ctx.fillStyle = '#84552e'
   838→      const postTop = a.y + hl * 1.4
   839→      ctx.fillRect(a.x - hl * 0.18, postTop, hl * 0.36, Math.max(0, H - 14 - postTop))
   840→      rr(ctx, a.x - hl * 1.35, a.y + hl * 0.55, hl * 2.7, hl * 0.85, hl * 0.3)
   841→      ctx.fill()
   842→      rr(ctx, a.x - hl * 0.3, a.y - hl * 0.35, hl * 0.6, hl * 1.15, hl * 0.22)
   843→      ctx.fill()
   844→      ctx.strokeStyle = '#84552e'
   845→      ctx.lineWidth = Math.max(3, hl * 0.28)
   846→      ctx.beginPath()
   847→      ctx.moveTo(a.x - hl * 0.24, a.y - hl * 0.3)
   848→      ctx.lineTo(a.x - fw, restY)
   849→      ctx.moveTo(a.x + hl * 0.24, a.y - hl * 0.3)
   850→      ctx.lineTo(a.x + fw, restY)
   851→      ctx.stroke()
   852→
   853→      // Visée / pilule
   854→      if (w.pill.state === 'aiming') {
   855→        const { dx, dy } = pullVec()
   856→        const px = a.x + dx
   857→        const py = a.y + dy
   858→        // élastiques tendus
   859→        ctx.strokeStyle = '#5f3d1e'
   860→        ctx.lineWidth = Math.max(2.5, hl * 0.22)
   861→        ctx.beginPath()
   862→        ctx.moveTo(a.x - fw, restY)
   863→        ctx.lineTo(px, py)
   864→        ctx.lineTo(a.x + fw, restY)
   865→        ctx.stroke()
   866→        // trajectoire prévue — même intégration pas-à-pas que le vol réel
   867→        const { vx, vy } = launchVec(px, py)
   868→        const g = gravity()
   869→        let tx = px, ty = py, tvy = vy
   870→        let dots = 0
   871→        for (let i = 1; i <= 240 && dots < 16; i++) {
   872→          tvy += g * STEP
   873→          tx += vx * STEP
   874→          ty += tvy * STEP
   875→          if (ty > H - 6 || tx < 4 || tx > W - 4) break
   876→          if (i % 8 === 0) {
   877→            dots++
   878→            const k = 1 - dots / 17
   879→            ctx.globalAlpha = 0.9 * k
